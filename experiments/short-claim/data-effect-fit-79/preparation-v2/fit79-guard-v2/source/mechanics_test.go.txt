func TestStorageAccountingUsesDistinctFileBytesAndPreservesRefusedFile(t *testing.T) {
	b := &assetBudget{prior: 5, max: 20}
	dir := t.TempDir()
	path := filepath.Join(dir, "numeric.json")
	if b.create(path, []byte("12345")) != nil {
		t.Fatal("create")
	}
	f, e := os.OpenFile(path, os.O_WRONLY, 0600)
	if e != nil {
		t.Fatal(e)
	}
	defer f.Close()
	if b.update(f, []byte("1234567")) != nil || b.current != 7 || b.written != 12 {
		t.Fatal("logical bytes vs repeated I/O")
	}
	if b.update(f, make([]byte, 16)) == nil {
		t.Fatal("aggregate cap bypass")
	}
	got, e := os.ReadFile(path)
	if e != nil || string(got) != "1234567" || b.current != 7 {
		t.Fatal("cap refusal mutated durable prefix")
	}
	if b.create(path, []byte("overwrite")) == nil {
		t.Fatal("exclusive existing output")
	}
	s := b.snapshot()
	s.Files[0].ID = "mutated-copy"
	if b.files[0].ID == s.Files[0].ID {
		t.Fatal("snapshot ownership")
	}
}

	a := RetainedAudit{Schema: "riido-plan56-covered-retained-serialized-registry-v1", CompleteWithinSnapshot: true, CountPerPath: true, PathCount: 1, PathBytes: 10, Assets: []RetainedAuditAsset{asset}}
	for _, test := range []struct {
		name   string
		mutate func(*RetainedAudit)
	}{
		{"wrong_schema", func(a *RetainedAudit) { a.Schema = "other" }},
		{"incomplete_snapshot", func(a *RetainedAudit) { a.CompleteWithinSnapshot = false }},
		{"inode_dedup_basis", func(a *RetainedAudit) { a.CountPerPath = false }},
		{"missing_assets", func(a *RetainedAudit) { a.Assets = nil }},
		{"false_path_count", func(a *RetainedAudit) { a.PathCount++ }},
		{"false_payload_sum", func(a *RetainedAudit) { a.PathBytes++ }},
		{"duplicate_ids", func(a *RetainedAudit) { a.Assets = append(a.Assets, asset); a.PathCount = 2; a.PathBytes = 20 }},
		{"negative_asset_bytes", func(a *RetainedAudit) { a.Assets[0].Bytes = -1 }},
	} {
		t.Run(test.name, func(t *testing.T) {
			b := a
			b.Assets = slices.Clone(a.Assets)
			test.mutate(&b)
			raw, e := json.Marshal(b)
			if e != nil || os.WriteFile(filepath.Join(root, "audit.json"), raw, 0600) != nil {
				t.Fatal("synthetic invalid audit")
			}
			p := Plan{Evidence: []Pin{{ID: "retained-serialized-audit-registry", Path: "audit.json", Bytes: int64(len(raw)), SHA256: sha(raw)}}}
			if _, e := readRetainedAudit(p, root); e == nil {
				t.Fatal("invalid audit accepted")
			}
		})
	}
