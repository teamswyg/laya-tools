func TestStorageAccountingUsesDistinctFileBytesAndPreservesRefusedFile(t *testing.T) {
	b := &assetBudget{prior: 5, max: 20}
	dir := t.TempDir()
	path := filepath.Join(dir, "numeric.json")
	if b.create(path, []byte("12345")) != nil {
		t.Fatal("create")
	}
	f, e := os.OpenFile(path, os.O_WRONLY, 0600)
	if e != nil {
		t.Fatal(e)
	}
	defer f.Close()
	if b.update(f, []byte("1234567")) != nil || b.current != 7 || b.written != 12 {
		t.Fatal("logical bytes vs repeated I/O")
	}
	if b.update(f, make([]byte, 16)) == nil {
		t.Fatal("aggregate cap bypass")
	}
	got, e := os.ReadFile(path)
	if e != nil || string(got) != "1234567" || b.current != 7 {
		t.Fatal("cap refusal mutated durable prefix")
	}
	if b.create(path, []byte("overwrite")) == nil {
		t.Fatal("exclusive existing output")
	}
	s := b.snapshot()
	s.Files[0].ID = "mutated-copy"
	if b.files[0].ID == s.Files[0].ID {
		t.Fatal("snapshot ownership")
	}
}

func TestRegistryAccountsForBothAlreadyCreatedNewInputs(t *testing.T) {
	root := t.TempDir()
	projection := []byte("toy-projection")
	appendInput := []byte("toy-qualified-input")
	for _, x := range []struct {
		name string
		data []byte
	}{{"projection.json", projection}, {"append.json", appendInput}} {
		if os.WriteFile(filepath.Join(root, x.name), x.data, 0600) != nil {
			t.Fatal("toy file")
		}
	}
	projectionPin := Pin{ID: "projection", Path: "projection.json", Bytes: int64(len(projection)), SHA256: sha(projection)}
	appendPin := Pin{ID: "append", Path: "append.json", Bytes: int64(len(appendInput)), SHA256: sha(appendInput)}
	n := 0 // Toy-only complete empty model registry; never an actual count.
	p := Plan{PriorArtifacts: &n, AssetsMax: maxPayload, NumericFileMax: maxNumericAsset, Projection: projectionPin, QualifiedAppend: appendPin}
	x := Consumption{Schema: "riido-fit79-root-frozen-consumption-v1", Frozen: true, RegistryComplete: true, PriorFitAttempts: 2, Models: []Pin{}, Assets: []Pin{projectionPin}, AssetsBytes: projectionPin.Bytes, RetainedMax: maxPayload, NewStageAssets: []Pin{projectionPin}, NewStageBytes: projectionPin.Bytes}
	writeRegistry := func() {
		raw, e := json.Marshal(x)
		if e != nil {
			t.Fatal("toy encode")
		}
		if os.WriteFile(filepath.Join(root, "registry.json"), raw, 0600) != nil {
			t.Fatal("toy registry")
		}
		p.ConsumptionRegistry = Pin{ID: "registry", Path: "registry.json", Bytes: int64(len(raw)), SHA256: sha(raw)}
	}
	writeRegistry()
	if _, e := consumption(p, root); e == nil || e.Error() != "fit79_new_input_asset_missing_from_consumption" {
		t.Fatal("omitted append must be refused")
	}
	x.Assets = append(x.Assets, appendPin)
	x.AssetsBytes += appendPin.Bytes
	x.NewStageAssets = append(x.NewStageAssets, appendPin)
	x.NewStageBytes += appendPin.Bytes
	writeRegistry()
	b, e := consumption(p, root)
	if e != nil || b.prior != projectionPin.Bytes+appendPin.Bytes {
		t.Fatal("new-data bytes not included")
	}
	x.Assets[0].SHA256 = strings.Repeat("0", 64)
	writeRegistry()
	if _, e := consumption(p, root); e == nil {
		t.Fatal("listed asset must still match bytes")
	}
}

func TestOriginalFloatBitPrefixAndValidationAreNotCoerced(t *testing.T) {
	z := math.Copysign(0, -1)
	old := pairlearn.Dataset{Split: "development", Offsets: []int{0, 1}, Indices: []uint16{2}, Values: []float64{z}, Labels: []float64{1}, Groups: []int{4}, SampleWeights: []float64{0}, Excluded: 39}
	now := pairlearn.Dataset{Split: "development", Offsets: []int{0, 1, 2}, Indices: []uint16{2, 3}, Values: []float64{z, 1}, Labels: []float64{1, 0}, Groups: []int{4, 75}, SampleWeights: []float64{0, 1}, Excluded: 39}
	if !sameDatasetPrefix(old, now, false) || sameDatasetPrefix(old, now, true) {
		t.Fatal("prefix/exact scopes")
	}
	now.Values[0] = 0
	if sameDatasetPrefix(old, now, false) {
		t.Fatal("negative zero erased")
	}
	if floatBitsEqual(nil, []float64{}) {
		t.Fatal("nullable slice coerced")
	}
}
func TestTrainingOriginUsesAllNineRowsWithoutChangingUtilityGate(t *testing.T) {
	p := claimfit.Projection{Parents: make([]claimfit.RuntimeParent, 79)}
	positive := [3]int{1, 0, 2}
	for i := 0; i < 3; i++ {
		x := claimfit.RuntimeParent{Text: shortclaim.Prepared{Request: "toy", Count: 3}, Role: 0, Truth: claimfit.Known}
		for c := 0; c < 3; c++ {
			x.Text.Candidates[c] = shortclaim.PreparedCandidate{ID: "toy", Text: "toy"}
			x.Labels[c] = claimfit.NullableLabel{Known: true, Positive: c == positive[i]}
			x.LossEligible[c] = true
			p.DevelopmentRefs = append(p.DevelopmentRefs, claimfit.RowRef{ParentIndex: 76 + i, CandidateIndex: c})
			p.Development.Offsets = append(p.Development.Offsets, len(p.Development.Values))
			p.Development.Indices = append(p.Development.Indices, 1)
			p.Development.Values = append(p.Development.Values, float64(c))
			label := 0.
			if c == positive[i] {
				label = 1
			}
			p.Development.Labels = append(p.Development.Labels, label)
		}
		p.Parents[76+i] = x
	}
	p.Development.Offsets = append(p.Development.Offsets, 9)
	weights := make([]float64, 8192)
	weights[1] = 1
	var counts Counts
	progressCalls := 0
	d, e := originDiagnostic(p, weights, &counts, fixedToyControls, func(out *OriginDiagnostic) error {
		progressCalls++

			t.Fatal("diagnostic promoted")

	})
	if e != nil || !d.NotUtilityGate || d.Primary == nil || d.Primary.Metric.Parents != 3 || d.Primary.Metric.OracleChecks != 3 || d.Primary.Metric.Top3Correct != 3 || counts.NewTrainControlAttempts != 3 || counts.NewTrainControlReturned != 3 || counts.NewTrainCachedScoreRows != 9 || progressCalls != 6 || counts.ControlParentAttempts != 0 || counts.FitAttempts != 0 {
