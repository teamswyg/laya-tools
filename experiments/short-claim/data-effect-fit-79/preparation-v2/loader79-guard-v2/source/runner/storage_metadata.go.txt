package main

import (
	"encoding/hex"
	"slices"
)

type resourcePolicy struct {
	Schema                    string `json:"schema"`
	State                     string `json:"state"`
	Frozen                    bool   `json:"frozen"`
	HistoricalLimit           uint64 `json:"historical_plan56_original_limit_bytes"`
	HistoricalExcessPreserved bool   `json:"historical_excess_preserved"`
	RetainedMax               uint64 `json:"retained_per_path_logical_bytes_max"`
	StageMax                  uint64 `json:"new_stage_data_models_numeric_bytes_max"`
	CountingBasis             string `json:"counting_basis"`
	ScientificRecipeChanged   bool   `json:"scientific_recipe_changed"`
	RegistrySHA               string `json:"current_registry_sha256"`
	RegistryBytes             int64  `json:"current_registry_bytes"`
	RegistryRetainedBytes     uint64 `json:"current_registry_retained_bytes"`
	ProjectionMax             int64  `json:"projection_file_bytes_max"`
	ProjectionLedgerMax       int64  `json:"projection_ledger_bytes_max"`
	OutsideMax                uint64 `json:"projection_outside_metadata_and_logs_bytes_max"`
	ProjectionCopies          int    `json:"projection_payload_copies_reserved"`
	ProjectReserve            uint64 `json:"projection_stage_reserved_bytes"`
	ResultMax                 uint64 `json:"result_file_bytes_max"`
	LedgerMax                 uint64 `json:"ledger_file_bytes_max"`
	ModelBytes                uint64 `json:"model_file_exact_bytes"`
	FitReserve                uint64 `json:"fit_core_output_reserved_bytes"`
	OriginalCaps              struct {
		Fits   int    `json:"fits"`
		Models int    `json:"model_assets"`
		Sparse uint64 `json:"sparse_payload"`
		CPU    int    `json:"cpu_threads"`
		Heap   uint64 `json:"go_soft_heap"`
		RSS    uint64 `json:"observed_rss"`
		Time   int    `json:"whole_process_timeout_seconds"`
	} `json:"original_caps"`
}
type auditAsset struct {
	ID    string `json:"id"`
	Bytes uint64 `json:"bytes"`
	SHA   string `json:"sha256"`
}
type auditRegistry struct {
	Schema       string       `json:"schema"`
	Complete     bool         `json:"complete_within_fixed_selector_and_snapshot"`
	Count        int          `json:"path_count"`
	Bytes        uint64       `json:"path_payload_bytes"`
	CountsCopies bool         `json:"count_equal_sha_or_equal_inode_per_path"`
	Assets       []auditAsset `json:"assets"`
}

// Validate readiness and accounting claims before creating an attempt directory.
// Root's exact SHA pins still own inventory scope and original-file identity.
func verifyStorageMetadata(s Storage) error {
	var a resourcePolicy
	var r auditRegistry
	if e := read(s.Amendment, 1<<20, &a); e != nil {
		return e
	}
	if e := read(s.Registry, 1<<20, &r); e != nil {
		return e
	}
	return checkStoredStorage(s, a, r)
}
func checkStoredStorage(s Storage, a resourcePolicy, r auditRegistry) error {
	c := a.OriginalCaps
	if !validStorage(s) || a.Schema != "riido-source80-data-effect79-resource-amendment-v1" || !a.Frozen || a.State != "frozen_resource_policy_for_one_data_effect79_stage" || a.HistoricalLimit != stageCap || !a.HistoricalExcessPreserved || a.ScientificRecipeChanged || a.RetainedMax != s.RetainedMax || a.StageMax != s.StageMax || a.CountingBasis != "sum current regular-file bytes per distinct governed path; identical SHA/inode remains counted" || a.RegistrySHA != s.Registry.SHA || a.RegistryBytes != s.Registry.Bytes || a.RegistryRetainedBytes != r.Bytes || a.ProjectionMax != resultCap || a.ProjectionLedgerMax != ledgerCap || a.OutsideMax != s.OutsideMax || a.ProjectionCopies != 1 || a.ProjectReserve != s.ProjectReserve || a.ResultMax != 8<<20 || a.LedgerMax != 8<<20 || a.ModelBytes != 32792 || a.FitReserve != fitOutputReserve || s.FitReserve < a.FitReserve || c.Fits != 8 || c.Models != 16 || c.Sparse != stageCap || c.CPU != 1 || c.Heap != 256<<20 || c.RSS != 256<<20 || c.Time != 300 {
		return safe("resource_policy_not_frozen_or_inconsistent")
	}
	if r.Schema != "riido-plan56-covered-retained-serialized-registry-v1" || !r.Complete || !r.CountsCopies || r.Count != 2776 || len(r.Assets) != r.Count || r.Bytes != 206707066 || r.Bytes > s.RetainedBytes {
		return safe("retained_audit_scope")
	}
	ids := make([]string, len(r.Assets))
	total := uint64(0)
	for i, v := range r.Assets {
		h, e := hex.DecodeString(v.SHA)
		if len(v.ID) < 1 || len(v.SHA) != 64 || e != nil || len(h) != 32 || v.Bytes > s.RetainedMax-total {
			return safe("retained_audit_asset")
		}
		ids[i] = v.ID
		total += v.Bytes
	}
	slices.Sort(ids)
	for i := 1; i < len(ids); i++ {
		if ids[i] == ids[i-1] {
			return safe("retained_audit_duplicate")
		}
	}
	if total != r.Bytes {
		return safe("retained_audit_total")
	}
	return nil
}
