package loader79

import (
	"github.com/teamswyg/laya-tools/internal/pairlearn"
	"testing"
)

// Literal representation of makeDataset's positive-weight diagnostic rows.
// This regression deliberately keeps explicit unit weights; v1 assumed nil.
func TestDiagnosticRowsRetainExplicitUnitWeights(t *testing.T) {
	d := pairlearn.Dataset{Split: "development", Offsets: []int{0, 1, 2}, Indices: []uint16{1, 2}, Values: []float64{.5, -.5}, Labels: []float64{0, 1}, Groups: []int{7, 7}, SampleWeights: []float64{1, 1}}
	if !shape(d, false) {
		t.Fatal("rejected explicit positive-weight diagnostic rows")
	}
	t.Run("missing_weights", func(t *testing.T) {
		x := d
		x.SampleWeights = nil
		if shape(x, false) {
			t.Fatal("accepted incomplete diagnostic weights")
		}
	})
	t.Run("masked_row_in_positive_view", func(t *testing.T) {
		x := d
		x.SampleWeights = []float64{1, 0}
		if shape(x, false) {
			t.Fatal("accepted masked diagnostic row")
		}
	})
}
