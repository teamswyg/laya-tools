// These controls invoke owned fake functions and synthetic reference shapes.
// They perform no original imports, source getters, filesystem Sync or native
// execution. The copied testdata are the exact existing owned frozen carriers.
func frozenControl(t *testing.T) ([]byte, []byte, []byte) {
	t.Helper()
	read := func(n string) []byte {
		b, e := os.ReadFile("testdata/" + n + ".json")
		if e != nil {
			t.Fatal(e)
		}
		return b
	}
	return read("freeze"), read("fixtures"), read("captions")
}

type evidenceTransport struct {
	bytes.Buffer
	badStage frame.Stage
	badD     uint8
	frames   [frame.MaxFrames]frame.Frame
	n        uint16
}

func newEvidenceTransport() *evidenceTransport {
	return &evidenceTransport{badStage: 255}
}

func (x *evidenceTransport) ack() (frame.Ack, error) {
	b := x.Bytes()
	if len(b) == 0 || b[len(b)-1] != '\n' || x.n >= frame.MaxFrames {
		return frame.Ack{}, frame.ErrAck
	}
	end := len(b) - 1
	start := bytes.LastIndexByte(b[:end], '\n') + 1
	f, e := frame.DecodeFrame(b[start:])
	if e != nil {
		return frame.Ack{}, e
	}
	x.frames[x.n] = f
	x.n++
	h := sha256.Sum256(b[start:end])
	s := hex.EncodeToString(h[:])
	if f.K == x.badStage && (f.D == x.badD || f.D == 255) {
		s = string(bytes.Repeat([]byte{'0'}, 64))
	}
	return frame.Ack{V: frame.Version, Q: f.Q, H: s}, nil
}

func (x *evidenceTransport) has(k frame.Stage) bool {
	for i := uint16(0); i < x.n; i++ {
		if x.frames[i].K == k {
			return true
		}
	}
	return false
}

type evidenceFakeSink struct {
	rows             [39]RowResult
	refs             [39][5]uint16
	attempts         uint8
	accepted         uint8
	finishes         uint8
	finishedComplete bool
	commitMode       string
	finishMode       string
	onCommit         func(RowResult, [5]uint16)
}

func (s *evidenceFakeSink) CommitRow(r RowResult, q [5]uint16) (frame.RowCommit, error) {
	s.attempts++
	if s.onCommit != nil {
		s.onCommit(r, q)
	}
	switch s.commitMode {
	case "error":
		return frame.RowCommit{}, errors.New("owned synthetic commit failure")
	case "nil-panic":
		panic(nil)
	}
	if r.Ordinal != s.accepted || s.accepted >= 39 {
		return frame.RowCommit{}, ErrObserver
	}
	s.rows[s.accepted], s.refs[s.accepted] = r, q
	s.accepted++
	ref := frame.RowCommit{O: 200 + uint32(r.Ordinal)*100, N: 100, H: FreezeSHA}
	if s.commitMode == "bad-ref" {
		ref.O = 0
	}
	return ref, nil
}

func (s *evidenceFakeSink) FinishEvidence(complete bool) (frame.FileCommit, error) {
	s.finishes++
	s.finishedComplete = complete
	switch s.finishMode {
	case "error":
		return frame.FileCommit{}, errors.New("owned synthetic footer failure")
	case "nil-panic":
		panic(nil)
	case "bad-ref":
		return frame.FileCommit{N: 1, H: "invalid", C: s.accepted}, nil
	}
	return frame.FileCommit{N: 200 + uint32(s.accepted)*100 + 30, H: FixtureSHA, C: s.accepted}, nil
}

type evidenceFakeCalls struct{ factories, directories, valid int }
type evidenceFakeDir struct {
	calls *evidenceFakeCalls
	fault func()
	cause error
	mode  string
}

func (d *evidenceFakeDir) Readdir(_ int, mark func(bool)) InfoReturn {
	mark(false)
	d.calls.directories++
	if d.mode == "source-nil-panic" {
		panic(nil)
	}
	mark(true)
	if d.mode == "owned-copy-nil-panic" {
		d.fault()
		panic(nil)
	}
	return InfoReturn{Entries: InfoList{}, Error: d.cause}
}

type evidenceFakeAPI struct {
	d     *evidenceFakeDir
	calls *evidenceFakeCalls
}

func (a *evidenceFakeAPI) NewUnion(Provider, Provider, Merge) Dir {
	if a.d.mode == "owned-candidate-nil-panic" {
		panic(nil)
	}
	return a.d
}
func (a *evidenceFakeAPI) Valid(_ string, mark func(bool)) bool {
	mark(false)
	a.calls.valid++
	mark(true)
	return false
}
func (*evidenceFakeAPI) Parse(string, func(bool)) Value { panic(ErrObserver) }
func (*evidenceFakeAPI) Map(Value, func(bool)) MapList  { panic(ErrObserver) }
func (*evidenceFakeAPI) ForEach(Value, func(func() (Value, Value)) bool, func(bool)) {
	panic(ErrObserver)
}

func evidenceFactory(c *evidenceFakeCalls, mode string) Factory {
	return func(fault func()) API {
		d := uint8(c.factories)
		c.factories++
		if mode == "factory-nil-panic" {
			panic(nil)
		}
		x, _ := frame.Schedule(d)
		dir := &evidenceFakeDir{calls: c, fault: fault, mode: mode}
		// The fake returns a finite owned cause for the fixture with a one-call
		// upper bound. It does not run providers or mimic an original Union.
		if x[0] == 0 && x[1] == 5 {
			dir.cause = ErrLayer
		}
		return &evidenceFakeAPI{d: dir, calls: c}
	}
}

func TestRunEvidenceFullScheduleCapturesBeforeAfterAndStaysCompact(t *testing.T) {
	f, i, c := frozenControl(t)
	out := newEvidenceTransport()
	sink := new(evidenceFakeSink)
	calls := new(evidenceFakeCalls)
	sink.onCommit = func(row RowResult, q [5]uint16) {
		if row.AckKnown || row.Invocation != (frame.Facts{}) || row.ObserverFault || row.SourcePanic.K {
			t.Fatal("captured late lifecycle or manufactured known false", row.Ordinal)
		}
		if out.frames[out.n-1].K != frame.AfterMethod {
			t.Fatal("candidate AFTER preceded its owned capture")
		}
		for j := uint8(0); j < row.APIN; j++ {
			if q[j] == 0 || !row.API[j].DurableOriginalOutcomeAvailable {
				t.Fatal("missing matching method ACK reference")
			}
		}
		for j := row.APIN; j < 5; j++ {
			if q[j] != 0 {
				t.Fatal("nonneutral reference padding")
			}
		}
	}
	r, e := RunEvidence(out, out.ack, evidenceFactory(calls, ""), sink, f, i, c)
	if e != nil || !r.Complete || r.Completed != 39 || r.CommittedRows != 39 || r.Prefix.Rows != 39 ||
		!r.FileAttempted || !r.FileReferenceAvailable || r.File.C != 39 || sink.accepted != 39 || sink.finishes != 1 || !sink.finishedComplete {
		t.Fatal(e, r, sink.accepted, sink.finishes)
	}
	if calls.factories != 39 || calls.directories != 30 || calls.valid != 21 || out.n != 260 {
		t.Fatal("owned schedule changed", calls, out.n)
	}
	for d := uint8(0); d < 39; d++ {
		x, _ := frame.Schedule(d)
		if sink.rows[d].Ordinal != d || sink.rows[d].IDs != x {
			t.Fatal("dispatch order changed", d)
		}
	}
	for j := uint16(0); j < out.n; j++ {
		x := out.frames[j]
		if x.K == frame.AfterCandidate && (x.R == nil || x.R.O != 200+uint32(x.D)*100 || x.F.Source.A) {
			t.Fatal("unbound candidate or method-only source escaped")
		}
	}
	last := out.frames[out.n-1]
	if last.K != frame.Final || last.E == nil || *last.E != r.File || !last.Z.Complete || last.Z.Startup != nil || last.Z.Nested != nil {
		t.Fatal("terminal correspondence or unknown native counters")
	}
	b, e := json.Marshal(r)
	if e != nil || len(b) > 2048 {
		t.Fatal("summary retained padded result", len(b), e)
	}
}

func TestMethodReferencesStayAtReservationIndexWithNestedCallback(t *testing.T) {
	w, _ := controlWriter(t, frame.Final)
	x, _ := frame.Schedule(0)
	c := context{w: w, d: 0, methodMax: 2, callbackMax: 1, row: RowResult{Ordinal: 0, IDs: x}}
	var captured [5]uint16
	_, e := w.Call(frame.Row, 0, 0, 0, func() frame.Facts {
		_, ce := w.CallCandidate(0, func() frame.Facts {
			c.direct(frame.Readdir, func(mark func(bool)) frame.Facts {
				mark(false)
				c.callback(frame.Merger, func() frame.Facts {
					c.direct(frame.Readdir, func(inner func(bool)) frame.Facts { inner(false); inner(true); return frame.Facts{} })
					return frame.Facts{}
				})
				mark(true)
				return frame.Facts{}
			})
			return frame.Facts{}
		}, func() (frame.RowCommit, error) {
			captured = c.methodAfter
			if c.row.Invocation != (frame.Facts{}) || c.row.AckKnown {
				t.Fatal("late nested capture")
			}
			return frame.RowCommit{O: 200, N: 100, H: FreezeSHA}, nil
		})
		if ce != nil {
			t.Fatal(ce)
		}
		return frame.Facts{}
	})
	if e != nil || c.fail != nil || c.row.APIN != 2 || captured[1] == 0 || captured[0] <= captured[1] ||
		!c.row.API[0].DurableOriginalOutcomeAvailable || !c.row.API[1].DurableOriginalOutcomeAvailable {
		t.Fatal("reservation order was confused with AFTER order", e, c.fail, captured)
	}
}
