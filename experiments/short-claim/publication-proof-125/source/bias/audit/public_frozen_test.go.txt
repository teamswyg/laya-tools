package audit

import (
	"bytes"
	"encoding/json"
	"math"
	"os"
	"reflect"
	"testing"
)

// Reproduce frozen development evidence. Scores allow small architecture
// roundoff; IDs, labels, orders, ties, counts, styles and all other values are exact.
func TestPublicDevelopment33FrozenReproduction(t *testing.T) {
	expectedPath, dataPath := os.Getenv("RIIDOLAYA_BIAS_FROZEN"), os.Getenv("RIIDOLAYA_BIAS_DATA")
	if expectedPath == "" || dataPath == "" {
		t.Fatal("explicit frozen evidence and data required")
	}
	raw, err := os.ReadFile(expectedPath)
	if err != nil {
		t.Fatal(err)
	}
	var expected Report
	if err := json.Unmarshal(raw, &expected); err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile(dataPath)
	if err != nil {
		t.Fatal(err)
	}
	actual, err := Run(bytes.NewReader(data), expected.DataSHA256)
	if err != nil {
		t.Fatal(err)
	}
	if len(actual.Cases) != len(expected.Cases) {
		t.Fatal("case count differs")
	}
	for c := range actual.Cases {
		rawRanks := actual.Cases[c].Ranks
		for r := range actual.Cases[c].Ranks {
			for s := range actual.Cases[c].Ranks[r].Scores {
				got, want := actual.Cases[c].Ranks[r].Scores[s], expected.Cases[c].Ranks[r].Scores[s]
				bound := 1e-12 * math.Max(1, math.Abs(want))
				if math.IsNaN(got) || math.IsInf(got, 0) || math.Abs(got-want) > bound {
					t.Fatal("score outside roundoff bound")
				}
				actual.Cases[c].Ranks[r].Scores[s] = want
			}
		}
		if !reflect.DeepEqual(actual.Cases[c], expected.Cases[c]) {
			for r := range actual.Cases[c].Ranks {
				if !reflect.DeepEqual(actual.Cases[c].Ranks[r], expected.Cases[c].Ranks[r]) {
					var gotBits, wantBits [3]uint64
					for s := range gotBits {
						gotBits[s] = math.Float64bits(rawRanks[r].Scores[s])
						wantBits[s] = math.Float64bits(expected.Cases[c].Ranks[r].Scores[s])
					}
					t.Fatalf("case=%d parent=%d primary=%t mapping=%v kind=%s raw_actual=%+v expected=%+v actual_score_bits=%x expected_score_bits=%x", c, actual.Cases[c].ParentIndex, actual.Cases[c].Primary, actual.Cases[c].DisplayToSelected, rawRanks[r].Kind, rawRanks[r], expected.Cases[c].Ranks[r], gotBits, wantBits)
				}
			}
			t.Fatalf("case=%d non-ranking metadata differs", c)
		}
	}
	if !reflect.DeepEqual(actual, expected) {
		t.Fatalf("global non-score evidence differs; counts_actual=%+v counts_expected=%+v primary_actual=%+v primary_expected=%+v permuted_actual=%+v permuted_expected=%+v", actual.Counts, expected.Counts, actual.Primary, expected.Primary, actual.Permuted, expected.Permuted)
	}
}
