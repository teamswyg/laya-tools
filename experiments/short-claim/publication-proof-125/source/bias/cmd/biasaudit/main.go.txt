package main

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"io"
	"os"
	"path/filepath"

	"github.com/teamswyg/laya-tools/experiments/developmentbiasaudit/audit"
)

func failed()                 { os.Stderr.WriteString("bias_audit_failed\n"); os.Exit(1) }
func canonical(p string) bool { return filepath.IsAbs(p) && filepath.Clean(p) == p }
func syncDir(p string) error {
	f, e := os.Open(p)
	if e != nil {
		return e
	}
	e = f.Sync()
	c := f.Close()
	if e != nil {
		return e
	}
	return c
}
func main() {
	var values [3]string
	var seen [3]bool
	a := os.Args[1:]
	if len(a) != 6 {
		failed()
	}
	for i := 0; i < len(a); i += 2 {
		idx := -1
		switch a[i] {
		case "--data":
			idx = 0
		case "--data-sha256":
			idx = 1
		case "--out":
			idx = 2
		}
		if idx < 0 || seen[idx] || a[i+1] == "" {
			failed()
		}
		seen[idx] = true
		values[idx] = a[i+1]
	}
	input, out := values[0], values[2]
	if !canonical(input) || !canonical(out) {
		failed()
	}
	resolved, e := filepath.EvalSymlinks(input)
	if e != nil || resolved != input {
		failed()
	}
	before, e := os.Lstat(input)
	if e != nil || !before.Mode().IsRegular() || before.Size() > audit.MaxDataBytes {
		failed()
	}
	parent := filepath.Dir(out)
	resolved, e = filepath.EvalSymlinks(parent)
	if e != nil || resolved != parent {
		failed()
	}
	if e = os.Mkdir(out, 0700); e != nil {
		failed()
	}
	if syncDir(parent) != nil {
		failed()
	}
	f, e := os.Open(input)
	if e != nil {
		failed()
	}
	opened, e := f.Stat()
	if e != nil || !os.SameFile(before, opened) {
		f.Close()
		failed()
	}
	report, e := audit.Run(io.LimitReader(f, audit.MaxDataBytes+1), values[1])
	closed := f.Close()
	after, statErr := os.Lstat(input)
	if e != nil || closed != nil || statErr != nil || !os.SameFile(before, after) || before.Size() != after.Size() || !before.ModTime().Equal(after.ModTime()) {
		failed()
	}
	raw, e := json.Marshal(report)
	raw = append(raw, '\n')
	if e != nil || len(raw) > audit.MaxReportBytes {
		failed()
	}
	target := filepath.Join(out, "results.json")
	result, e := os.OpenFile(target, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		failed()
	}
	_, e = result.Write(raw)
	if e == nil {
		e = result.Sync()
	}
	c := result.Close()
	if e != nil || c != nil || syncDir(out) != nil {
		failed()
	}
	h := sha256.Sum256(raw)
	// Only after final file/file-close/directory acknowledgments: scalar stdout.
	// Root still decides what the simulated development result means.
	ack := struct {
		State    string `json:"state"`
		Bytes    int    `json:"bytes"`
		SHA      string `json:"sha256"`
		Calls    int    `json:"baselines_calls"`
		Rankings int    `json:"rankings"`
	}{"complete", len(raw), hex.EncodeToString(h[:]), report.Counts.BaselinesReturned, report.Counts.RankingRecords}
	if json.NewEncoder(os.Stdout).Encode(ack) != nil {
		failed()
	}
}
