// Validates Unicode scalars before Go's decoder can replace an unpaired escape.
// Derived from the project's Apache-2.0 shortclaimdata reader, source pinned in
// the preparation ledger. json.Valid is a precondition for the indexed slices.
func scalarEscapes(raw []byte) bool {
	for i := 0; i < len(raw); i++ {
		if raw[i] != '"' {
			continue
		}
		for i++; i < len(raw) && raw[i] != '"'; i++ {
			if raw[i] != '\\' {
				continue
			}
			i++
			if raw[i] != 'u' {
				continue
			}
			u := hex4(raw[i+1 : i+5])
			i += 4
			if u >= 0xdc00 && u <= 0xdfff {
				return false
			}
			if u >= 0xd800 && u <= 0xdbff {
				if i+6 >= len(raw) || raw[i+1] != '\\' || raw[i+2] != 'u' {
					return false
				}
				lo := hex4(raw[i+3 : i+7])
				if lo < 0xdc00 || lo > 0xdfff {
					return false
				}
				i += 6
			}
		}
	}
	return true
}
func hex4(b []byte) uint16 {
	var v uint16
	for _, c := range b {
		v <<= 4
		switch {
		case c >= '0' && c <= '9':
			v |= uint16(c - '0')
		case c >= 'a' && c <= 'f':
			v |= uint16(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			v |= uint16(c - 'A' + 10)
		}
	}
	return v
}

// The complete JSON is checked for decoded duplicate/case-alias keys and bounded
// shape before typed decoding. Arrays/slices track keys; no label-keyed maps.
func checkedJSON(raw []byte, dst any, exactFields int) error {
	if len(raw) < 1 || len(raw) > maxInputBytes || !utf8.Valid(raw) || !json.Valid(raw) || !scalarEscapes(raw) {
		return failure("input_json_unicode_bounds")
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.UseNumber()
	nodes := 0
	fields, e := walk(d, 0, &nodes)
	if e != nil {
		return e
	}
	if exactFields >= 0 && fields != exactFields {
		return failure("input_fields")
	}
	if _, e = d.Token(); !errors.Is(e, io.EOF) {
		return failure("input_trailing")
	}
	if json.Unmarshal(raw, dst) != nil {
		return failure("input_types")
	}
	return nil
}
func walk(d *json.Decoder, depth int, nodes *int) (int, error) {
	*nodes++
	if depth > 32 || *nodes > 65536 {
		return 0, failure("input_shape_bounds")
	}
	t, e := d.Token()
	if e != nil {
		return 0, failure("input_json")
	}
	delim, ok := t.(json.Delim)
	if !ok {
		return -1, nil
	}
	if delim == '{' {
		keys := make([]string, 0, 16)
		for d.More() {
			if len(keys) >= 128 {
				return 0, failure("input_shape_bounds")
			}
			kt, e := d.Token()
			key, ok := kt.(string)
			if e != nil || !ok {
				return 0, failure("input_json")
			}
			for _, old := range keys {
				if strings.EqualFold(old, key) {
					return 0, failure("input_duplicate")
				}
			}
			keys = append(keys, key)
			if _, e = walk(d, depth+1, nodes); e != nil {
				return 0, e
			}
		}
		end, e := d.Token()
		if e != nil || end != json.Delim('}') {
			return 0, failure("input_json")
		}
		return len(keys), nil
	}
	if delim == '[' {
		n := 0
		for d.More() {
			n++
			if n > 1024 {
				return 0, failure("input_shape_bounds")
			}
			if _, e = walk(d, depth+1, nodes); e != nil {
				return 0, e
			}
		}
		end, e := d.Token()
		if e != nil || end != json.Delim(']') {
			return 0, failure("input_json")
		}
		return -1, nil
	}
	return 0, failure("input_json")
}
func exactObject(raw []byte, keys []string) error {
	var fields []struct {
		key   string
		value json.RawMessage
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	t, e := d.Token()
	if e != nil || t != json.Delim('{') {
		return failure("row_fields")
	}
	for d.More() {
		kt, e := d.Token()
		k, ok := kt.(string)
		if e != nil || !ok {
			return failure("row_fields")
		}
		var value json.RawMessage
		if d.Decode(&value) != nil {
			return failure("row_fields")
		}
		fields = append(fields, struct {
			key   string
			value json.RawMessage
		}{k, value})
	}
	if len(fields) != len(keys) {
		return failure("row_fields")
	}
	for _, k := range keys {
		found := false
		for _, f := range fields {
			if f.key == k {
				found = true
				break
			}
		}
		if !found {
			return failure("row_fields")
		}
	}
	return nil
}
func rowDecode(raw []byte) (Row, error) {
	var r Row
	if len(raw) > maxRowBytes {
		return r, failure("row_bounds")
	}
	if e := checkedJSON(raw, &r, 11); e != nil {
		return Row{}, e
	}
	if e := exactObject(raw, []string{"schema", "stable_id", "request", "candidates", "role", "whole_group", "source_family", "source_revision", "finite_scope", "text_revision", "feature_policy"}); e != nil {
		return Row{}, e
	}
	var partial struct {
		Candidates []json.RawMessage `json:"candidates"`
		Scope      json.RawMessage   `json:"finite_scope"`
	}
	if json.Unmarshal(raw, &partial) != nil {
		return Row{}, failure("row_fields")
	}
	for _, c := range partial.Candidates {
		if e := exactObject(c, []string{"metadata_id", "text", "label", "sample_weight"}); e != nil {
			return Row{}, e
		}
	}
	if e := exactObject(partial.Scope, []string{"input_count", "candidate_observations", "description", "all_input_guarantee", "unseen_source_guarantee"}); e != nil {
		return Row{}, e
	}
	if r.Schema != "riido-finite-development-request-row-v1" || r.Role != "development_train" || r.Policy != "request_and_candidates_text_only" || len(r.Candidates) < 1 || len(r.Candidates) > 8 || r.Scope.Inputs < 1 || r.Scope.Inputs > 4096 || r.Scope.Observations != r.Scope.Inputs*len(r.Candidates) || r.Scope.All == nil || *r.Scope.All || r.Scope.Unseen == nil || *r.Scope.Unseen {
		return Row{}, failure("row_metadata_scope")
	}
	if !identifier(r.ID, 64) || !identifier(r.Family, 128) || !identifier(r.TextRevision, 128) || !revision(r.Revision) || r.Group < 1 || r.Group > 65535 || !textOK(r.Request) || !textOK(r.Scope.Description) {
		return Row{}, failure("row_metadata_text")
	}
	for _, c := range r.Candidates {
		if !identifier(c.ID, 64) || !textOK(c.Text) || c.Label == nil || c.Weight != 1 {
			return Row{}, failure("row_candidate")
		}
	}
	return r, nil
}
func textOK(s string) bool {
	return len(s) > 0 && len(s) <= 512 && strings.TrimSpace(s) != "" && !strings.ContainsRune(s, 0)
}
func identifier(s string, max int) bool {
	if len(s) < 1 || len(s) > max {
		return false
	}
	for i, c := range []byte(s) {
		if !(c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9') && (i == 0 || c != '.' && c != '_' && c != '-') {
			return false
		}
	}
	return true
}
func revision(s string) bool {
	if len(s) != 40 {
		return false
	}
	for _, c := range s {
		if !(c >= '0' && c <= '9' || c >= 'a' && c <= 'f') {
			return false
		}
	}
	return true
}
func intsEqual(a, b []int) bool {
	if a == nil || b == nil || len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}
func positions(a []int, n int) bool {
	if a == nil {
		return false
	}
	prev := -1
	for _, i := range a {
		if i < 0 || i >= n || i <= prev {
			return false
		}
		prev = i
	}
	return true
}
