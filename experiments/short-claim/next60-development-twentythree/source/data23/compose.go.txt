func isTrue(v *bool) bool           { return v != nil && *v }
func isFalse(v *bool) bool          { return v != nil && !*v }
func isNull(v json.RawMessage) bool { return bytes.Equal(bytes.TrimSpace(v), []byte("null")) }
func stringsEqual(a, b []string) bool {
	if a == nil || b == nil || len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}

// All public entry points bind fixed historical inputs before decoding or output.
func bind(b Bundle) error {
	for _, p := range []struct {
		raw  []byte
		size int
		sha  string
	}{
		{b.Previous, PreviousBytes, PreviousSHA}, {b.Fixtures, FixtureBytes, FixtureSHA},
		{b.Adoption, AdoptionBytes, AdoptionSHA}, {b.Comparison, ComparisonBytes, ComparisonSHA},
	} {
		if len(p.raw) != p.size || Digest(p.raw) != p.sha {
			return failure("input_pin")
		}
	}
	return nil
}

func inspectPrefix(raw []byte) (Summary, error) {
	s := InitialSummary()
	if len(raw) < 1 || len(raw) > maxDataBytes || raw[len(raw)-1] != '\n' {
		return s, failure("prefix_newline_bounds")
	}
	scan := bufio.NewScanner(bytes.NewReader(raw))
	scan.Buffer(make([]byte, 4096), maxRowBytes+1)
	for scan.Scan() {
		i := s.Requests
		if i >= len(priorIDs) || len(scan.Bytes()) == 0 {
			return s, failure("prefix_rows")
		}
		r, e := rowDecode(scan.Bytes())
		if e != nil {
			return s, e
		}
		if r.ID != priorIDs[i] || r.Group < 76 || r.Group > 78 {
			return s, failure("prefix_identity")
		}
		if e = countRow(r, &s); e != nil {
			return s, e
		}
	}
	if scan.Err() != nil || s.Requests != 21 || s.Labels != 61 || s.Positive != 21 || s.Negative != 40 || s.Inputs != 103 || s.Selected != 301 || s.PositiveIndex1 != 18 {
		return s, failure("prefix_counts")
	}
	s.Original = 305
	s.PrefixExact = true
	return s, nil
}
func countRow(r Row, s *Summary) error {
	positive := 0
	for i, c := range r.Candidates {
		for j := 0; j < i; j++ {
			if r.Candidates[j].ID == c.ID {
				return failure("duplicate_candidate_id")
			}
		}
		s.Labels++
		if *c.Label {
			s.Positive++
			positive++
			if i == 1 {
				s.PositiveIndex1++
			}
		} else {
			s.Negative++
		}
	}
	if positive != 1 {
		return failure("row_positive_count")
	}
	s.Requests++
	s.Inputs += r.Scope.Inputs
	s.Selected += r.Scope.Observations
	return nil
}

				return f, q, s, failure("Root_selected_correspondence")

			prefix := "bounded-slice-"

// Arithmetic over Root's saved finite predicate states only. No Wanted/Got
// semantic predicate is recomputed, and the reader cannot create a label.
func inspectComparison(raw []byte, f fixtures, q qualification, s *Summary) error {
	var c comparison
	if e := checkedJSON(raw, &c, -1); e != nil {
		return e
	}
	r := c.Report
	if c.Schema != "riido-extra-two-saved-comparison-envelope-v1" || c.State != "complete" || !isNull(c.Failure) || c.Workers != 2 || !isTrue(c.Started) || c.Original != 0 || c.Oracle != 0 || c.Model != 0 || c.Fit != 0 || !isFalse(c.Labels) || !isFalse(c.Roles) || !isFalse(c.Weights) || !isFalse(c.Qualified) || r.Schema != "riido-extra-two-saved-semantic-comparison-v1" || r.State != "complete" || len(r.Observations) != 30 || len(r.Candidates) != 6 || len(r.References) != 2 || !isTrue(r.References[0]) || !isTrue(r.References[1]) || !isNull(r.Inner) || !isNull(r.Startup) || !isFalse(r.Labels) || !isFalse(r.Roles) || !isFalse(r.Weights) || !isFalse(r.Qualified) || r.Model != 0 || r.Fit != 0 {
		return failure("comparison_header")
	}
	var pc, oc countTFU
	for i, o := range r.Observations {
		fi := i / 15
		k := (i % 15) / 3
		j := i % 3
		if o.Family != fi || o.Fixture != k || o.Candidate != j || o.CaseID != f.Records[fi*5+k].Original.CaseID || o.Count < 1 || o.Count > 12 || len(o.Predicates) != 12 || !isTrue(o.ReturnKnown) || !isTrue(o.Returned) || !isTrue(o.PanicKnown) || !isFalse(o.Panic) || !isFalse(o.ErrorText) {
			return failure("saved_observation_identity")
		}
		var counts countTFU
		for z, p := range o.Predicates {
			if z >= o.Count {
				if p != (predicate{}) {
					return failure("saved_predicate_unused")
				}
				continue
			}
			if p.Field == "" || p.Reason == "" {
				return failure("saved_predicate_shape")
			}
			for a := 0; a < z; a++ {
				if o.Predicates[a].Field == p.Field {
					return failure("saved_predicate_duplicate")
				}
			}
			switch p.State {
			case "true":
				counts.True++
			case "false":
				counts.False++
			case "unknown":
				counts.Unknown++
			default:
				return failure("saved_predicate_state")
			}
		}
		state := "true"
		if counts.False > 0 {
			state = "false"
		} else if counts.Unknown > 0 {
			state = "unknown"
		}
		if counts != o.Counts || o.State != state {
			return failure("saved_predicate_counts")
		}
		pc.True += counts.True
		pc.False += counts.False
		pc.Unknown += counts.Unknown
		switch state {
		case "true":
			oc.True++
		case "false":
			oc.False++
		case "unknown":
			oc.Unknown++
		}
	}
	if pc != (countTFU{125, 15, 1}) || oc != (countTFU{23, 7, 0}) || r.Predicates != pc || r.Counts != oc {
		return failure("saved_comparison_totals")
	}
	unknowns := 0
	for fi, ar := range q.Requests {
		for j, ac := range ar.Selected {
			known := []int{}
			unknown := []int{}
			retained := []unknownPredicate{}
			var fc countTFU
			for k := 0; k < 5; k++ {
				ordinal := fi*15 + k*3 + j
				o := r.Observations[ordinal]
				v := "satisfied"
				switch o.State {
				case "false":
					v = "unsatisfied"
					fc.False++
				case "unknown":
					v = "unknown"
					fc.Unknown++
				default:
					fc.True++
				}
				if ac.Ordinals[k] != ordinal || ac.Vector[k] != v {
					return failure("Root_saved_vector")
				}
				if o.Counts.False > 0 {
					known = append(known, k)
				}
				if o.State == "unknown" {
					unknown = append(unknown, k)
				}
				for _, p := range o.Predicates[:o.Count] {
					if p.State == "unknown" {
						retained = append(retained, unknownPredicate{k, p.Field, p.Reason})
					}
				}
			}
			if !intsEqual(ac.Known, known) || !intsEqual(ac.Unknown, unknown) || len(ac.Retained) != len(retained) {
				return failure("Root_counterexamples_unknown")
			}
			for k, p := range retained {
				if ac.Retained[k] != p {
					return failure("Root_retained_unknown")
				}
			}
			ev := r.Candidates[fi*3+j]
			state := "true"
			if fc.False > 0 {
				state = "false"
			} else if fc.Unknown > 0 {
				state = "unknown"
			}
			if ev.Family != fi || ev.Candidate != j || ev.Counts != fc || ev.State != state {
				return failure("saved_candidate_evidence")
			}
			if *ac.Label {
				if state != "true" || len(retained) != 0 {
					return failure("Root_positive_evidence")
				}
			} else if state != "false" || len(known) == 0 {
				return failure("Root_negative_evidence")
			}
			unknowns += len(retained)
			if len(retained) > 0 {
				s.Unknowns = append(s.Unknowns, retained...)
			}
		}
	}
	if unknowns != 1 || len(s.Unknowns) != 1 || s.Unknowns[0] != (unknownPredicate{0, "B_code", "observable_unavailable"}) {
		return failure("retained_unknown_identity")
	}
	s.UnknownPredicateCount = unknowns
	return nil
}

func newRow(a adoptedRequest) Row {
	no1, no2 := false, false
	r := Row{Schema: "riido-finite-development-request-row-v1", ID: a.ID, Request: a.Request, Role: a.Selected[0].Role, Group: a.Group, Family: a.Family, Revision: a.Revision, TextRevision: a.TextRevision, Policy: "request_and_candidates_text_only", Scope: finiteScope{a.Count, a.Observations, a.Scope, &no1, &no2}}
	for _, c := range a.Selected {
		b := *c.Label
		r.Candidates = append(r.Candidates, candidate{c.ID, c.Text, &b, c.Weight})
	}
	return r
}

// composeBound is private and accepts typed synthetic fixtures for controls.
// Real callers must pass through ComposePinned/Materialize and their fixed pins.
func composeBound(b Bundle) ([]byte, Summary, error) {
	_, q, s, e := inspectInputs(b)
	if e != nil {
		return nil, s, e
	}
	data := bytes.Clone(b.Previous)
	for _, a := range q.Requests {
		r := newRow(a)
		raw, e := json.Marshal(r)
		if e != nil {
			return nil, s, failure("row_encode")
		}
		checked, e := rowDecode(raw)
		if e != nil {
			return nil, s, e
		}
		if e = countRow(checked, &s); e != nil {
			return nil, s, e
		}
		if len(data)+len(raw)+1 > maxDataBytes {
			return nil, s, failure("data_bounds")
		}
		data = append(data, raw...)
		data = append(data, '\n')
	}
	if s.Requests != 23 || s.Labels != 67 || s.Positive != 23 || s.Negative != 44 || s.Inputs != 113 || s.Selected != 331 || s.PositiveIndex1 != 20 {
		return nil, s, failure("combined_counts")
	}
	s.Original = 335
	s.CopiedRootSupervision = true
	s.DataSHA = Digest(data)
	s.DataBytes = len(data)
	s.State = "composed_Root_adopted_data_only"
	return data, s, nil
}
func ComposePinned(b Bundle) ([]byte, Summary, error) {
	if e := bind(b); e != nil {
		return nil, InitialSummary(), e
	}
	return composeBound(b)
}
