func boolPtr(v bool) *bool { return &v }
func intPtr(v int) *int    { return &v }
func marshal(t *testing.T, v any) []byte {
	t.Helper()
	b, e := json.Marshal(v)
	if e != nil {
		t.Fatal(e)
	}
	return b
}

func fakeView(raw []byte) (LoadedView, error) {
	r, e := rowDecode(raw)
	if e != nil {
		return LoadedView{}, e
	}
	v := LoadedView{Valid: true, UnusedZero: true, ID: r.ID, Request: r.Request, Schema: "riido-short-behavior-claim-v1", Provenance: "finite-development-v1", Role: r.Role, Policy: r.Policy, Family: r.Family, Revision: r.Revision, TextRevision: r.TextRevision, ScopeDescription: r.Scope.Description, Group: r.Group, Inputs: r.Scope.Inputs, Observations: r.Scope.Observations, Count: len(r.Candidates), SupervisionCount: len(r.Candidates)}
	for j, c := range r.Candidates {
		v.Candidates[j] = LoadedCandidate{c.ID, c.Text, *c.Label, c.Weight}
	}
	return v, nil
}

func TestSyntheticPrefixAppendAndReaderCorrespondence(t *testing.T) {
	b := synthetic(t)
	data, s, e := composeBound(b)
	if e != nil {
		t.Fatal(e)
	}
	if !bytes.HasPrefix(data, b.Previous) || len(data) <= len(b.Previous) || s.Labels != 67 || s.Requests != 23 || s.UnknownPredicateCount != 1 || s.UnknownFalse || s.Original != 335 || s.Selected != 331 {
		t.Fatal("prefix/counts/unknown lost")
	}
	second, s2, e := composeBound(b)
	if e != nil || !bytes.Equal(data, second) || s.DataSHA != s2.DataSHA {
		t.Fatal("non-deterministic bytes")
	}
	var progress []Summary
	calls := 0
	e = validateBound(b, data, func(raw []byte) (LoadedView, error) { calls++; return fakeView(raw) }, func(s Summary) error { progress = append(progress, s); return nil }, &s)
	if e != nil || calls != 23 || len(progress) != 46 || s.ReaderMatches != 23 || s.PositiveIndex1 != 20 || s.LabelsAssigned || s.Qualified {
		t.Fatal("fake reader correspondence", e)
	}
	if progress[0].ReaderAttempts != 1 || progress[0].ReaderReturns != 0 || progress[1].ReaderReturns != 1 {
		t.Fatal("dispatch/return collapsed")
	}
	// Real fixed-pin APIs reject this fake corpus, even though its structure passes.
	if raw, _, e := ComposePinned(b); e == nil || raw != nil {
		t.Fatal("fake corpus crossed fixed-pin gate")
	}
}

	for _, name := range []string{"order", "labelnull", "unknownerase", "vector", "weight", "group", "scope", "revision", "fixturetruth", "comparisonstate", "predicateunused", "missingfalse", "unknown_to_false"} {

func TestAllRowsCheckedBeforeReaderAndFailureStops(t *testing.T) {
	b := synthetic(t)
	data, _, e := composeBound(b)
	if e != nil {
		t.Fatal(e)
	}
	cases := []struct {
		name string
		raw  []byte
	}{{"missingLF", bytes.TrimSuffix(data, []byte{'\n'})}, {"blanktail", append(bytes.Clone(data), '\n')}, {"same_length_prefix", bytes.Replace(data, []byte("Owned synthetic request"), []byte("Other synthetic request"), 1)}, {"extra_candidate_field", bytes.Replace(data, []byte(`"sample_weight":1`), []byte(`"sample_weight":1,"extra":0`), 1)}}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			calls := 0
			s := InitialSummary()
			e := validateBound(b, tc.raw, func(raw []byte) (LoadedView, error) { calls++; return fakeView(raw) }, func(Summary) error { return nil }, &s)
			if e == nil || calls != 0 {
				t.Fatal("invalid corpus dispatched")
			}
		})
	}
	t.Run("before_dispatch_checkpoint", func(t *testing.T) {
		calls := 0
		s := InitialSummary()
		e := validateBound(b, data, func(raw []byte) (LoadedView, error) { calls++; return fakeView(raw) }, func(Summary) error { return errors.New("private callback message") }, &s)
		if FailureCode(e) != "reader_checkpoint_failed" || calls != 0 || s.ReaderAttempts != 1 || s.ReaderReturns != 0 {
			t.Fatal("checkpoint refusal not atomic")
		}
	})
	t.Run("reader_value_error", func(t *testing.T) {
		calls := 0
		s := InitialSummary()
		e := validateBound(b, data, func(raw []byte) (LoadedView, error) {
			calls++
			v, e := fakeView(raw)
			v.Candidates[0].Weight = 0
			return v, e
		}, func(Summary) error { return nil }, &s)
		if e == nil || calls != 1 || s.ReaderReturns != 1 || s.ReaderMatches != 0 {
			t.Fatal("forged loaded weight accepted")
		}
	})
	t.Run("reader_panic", func(t *testing.T) {
		calls := 0
		s := InitialSummary()
		e := validateBound(b, data, func([]byte) (LoadedView, error) { calls++; panic(errors.New("private panic text")) }, func(Summary) error { return nil }, &s)
		if e == nil || calls != 1 || s.ReaderReturns != 0 || s.Failure != "reader_panic" {
			t.Fatal("panic invented a return")
		}
	})
	t.Run("callback_ownership", func(t *testing.T) {
		s := InitialSummary()
		e := validateBound(b, data, fakeView, func(v Summary) error { v.Unknowns[0].Field = "changed"; return nil }, &s)
		if e != nil || s.Unknowns[0].Field != "B_code" {
			t.Fatal("callback changed live evidence")
		}
	})
}

func TestStrictJSONNullSurrogatesDuplicatesAndOptions(t *testing.T) {
	for _, raw := range []string{`{"truth":false,"truth":true}`, `{"truth":false,"TRUTH":true}`, `{"tr\u0075th":false,"truth":true}`, `{"x":"\ud800"}`, `{"x":"\udc00"}`, `{"x":0} {}`, `{"x":[}`} {
		var q qualification
		if checkedJSON([]byte(raw), &q, -1) == nil {
			t.Fatal("bad JSON accepted")
		}
	}
	b := synthetic(t)
	line := bytes.Split(b.Previous, []byte{'\n'})[0]
	for _, raw := range [][]byte{bytes.Replace(line, []byte(`"label":true`), []byte(`"label":null`), 1), bytes.Replace(line, []byte(`"all_input_guarantee":false`), []byte(`"all_input_guarantee":null`), 1)} {
		if _, e := rowDecode(raw); e == nil {
			t.Fatal("null treated as label/scope")
		}
	}
	names := []string{"--one", "--two"}
	if _, e := ParseOptions([]string{"--two", "b", "--one", "a"}, names); e != nil {
		t.Fatal(e)
	}
	for _, args := range [][]string{{"--one", "a", "--one", "b"}, {"--one=a", "a", "--two", "b"}, {"--one", "a"}, {"--one", "a", "--two", "b", "extra"}} {
		if _, e := ParseOptions(args, names); e == nil {
			t.Fatal("ambiguous options accepted")
		}
	}
}

func TestOwnedPinnedFilesExclusiveOutputAndRecorder(t *testing.T) {
	dir := t.TempDir()
	raw := []byte("owned input\n")
	input := filepath.Join(dir, "input")
	if e := os.WriteFile(input, raw, 0600); e != nil {
		t.Fatal(e)
	}
	if _, e := ReadPinned(input, Digest(raw), len(raw)); e != nil {
		t.Fatal(e)
	}
	if _, e := ReadPinned(input, strings.Repeat("0", 64), len(raw)); e == nil {
		t.Fatal("wrong hash accepted")
	}
	if e := os.Symlink(input, filepath.Join(dir, "link")); e != nil {
		t.Fatal(e)
	}
	if _, e := ReadPinned(filepath.Join(dir, "link"), Digest(raw), len(raw)); e == nil {
		t.Fatal("symlink accepted")
	}
	path := filepath.Join(dir, "output")
	s := InitialSummary()
	if e := WriteExclusive(path, raw, 128, &s); e != nil {
		t.Fatal(e)
	}
	if !s.Reserved || !s.FileSynced || !s.DirectorySynced {
		t.Fatal("sync acknowledgment missing")
	}
	if e := WriteExclusive(path, []byte("replace"), 128, nil); e == nil {
		t.Fatal("overwrite accepted")
	}
	got, _ := os.ReadFile(path)
	if !bytes.Equal(got, raw) {
		t.Fatal("failed overwrite changed bytes")
	}
	t.Run("forged_completed_without_journal", func(t *testing.T) {
		w, e := OpenReaderRecorder(filepath.Join(dir, "forged"))
		if e != nil {
			t.Fatal(e)
		}
		s := InitialSummary()
		s.State = "complete_reader_value_correspondence_only"
		s.ReaderAttempts = 23
		s.ReaderReturns = 23
		s.ReaderMatches = 23
		if e = w.Finish(s); FailureCode(e) != "reader_finish_unacknowledged" {
			t.Fatal("unacknowledged completion accepted", e)
		}
	})
	t.Run("real_fake_lifecycle", func(t *testing.T) {
		w, e := OpenReaderRecorder(filepath.Join(dir, "owned-reader"))
		if e != nil {
			t.Fatal(e)
		}
		b := synthetic(t)
		data, _, e := composeBound(b)
		if e != nil {
			t.Fatal(e)
		}
		s := InitialSummary()
		if e = validateBound(b, data, fakeView, w.Checkpoint, &s); e != nil {
			t.Fatal(e)
		}
		if e = w.Finish(s); e != nil {
			t.Fatal(e)
		}
		if e = w.Finish(s); e == nil {
			t.Fatal("double Finish")
		}
		got, e := os.ReadFile(filepath.Join(dir, "owned-reader", "results.json"))
		if e != nil || !bytes.Contains(got, []byte(`"LoadDevelopmentRow_value_matches":23`)) {
			t.Fatal("final metadata absent")
		}
	})
}
