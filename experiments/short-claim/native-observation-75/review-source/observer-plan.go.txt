package pure

type Ref struct {
	ID     string `json:"id"`
	Bytes  int    `json:"bytes"`
	SHA256 string `json:"sha256"`
}
type ClosureRef struct {
	Base, Path string
	Ref
}
type ExecutionPlan struct {
	Schema                                                                                                        string
	Frozen                                                                                                        bool
	ObserverRoot, OriginalSourceRoot                                                                              string
	Want, Binary                                                                                                  Ref
	Closure                                                                                                       []ClosureRef
	Toolchain, GOOS, GOARCH                                                                                       string
	CPU, SoftGoHeapBytes, OutsideTimeoutSeconds, OutputLimitBytes, PlannedOriginalAPICalls, DistinctBehaviorGoals int
	OutsideReservationRequired, NoAutomaticRetry                                                                  bool
	TrainingReady, BroadTruthProven, CaptionsQualified, ParentLabelsQualified                                     bool
}
