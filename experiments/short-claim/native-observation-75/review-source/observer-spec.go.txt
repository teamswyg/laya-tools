package pure

import (
	"errors"
	"reflect"

const MaxProbes = 32
const OutputLimit = (1 << 20) - 1

type Pair struct{ Key, Value string }
type MapState struct {
	Nil     bool
	Entries []Pair
}
type Observation struct {
	Text                    string
	Error                   *string
	InputBefore, InputAfter *MapState
	Panicked                bool
}
type Goal struct{ ID, API, Repository, Revision, SourceSHA256, Derivation, ExcludedScope string }
type Probe struct {
	ID, Goal, Text string
	Width          uint
	MapNil         bool
	Entries        []Pair
	Want           Observation
}
type Specification struct {
	Schema                                                                    string
	Goals                                                                     []Goal
	Probes                                                                    []Probe
	BroadTruthProven, CaptionsQualified, ParentLabelsQualified, TrainingReady bool
	NewParents, NewLabels, Models, Fits, ProtectedFinalReads                  int
}

func clonePairs(v []Pair) []Pair {
	if v == nil {
		return nil
	}
	return append([]Pair{}, v...)
}
func cloneMap(v *MapState) *MapState {
	if v == nil {
		return nil
	}
	return &MapState{Nil: v.Nil, Entries: clonePairs(v.Entries)}
}
func CloneObservation(v Observation) Observation {
	v.InputBefore = cloneMap(v.InputBefore)
	v.InputAfter = cloneMap(v.InputAfter)
	if v.Error != nil {
		x := *v.Error
		v.Error = &x
	}
	return v
}
func cloneProbe(p Probe) Probe {
	p.Entries = clonePairs(p.Entries)
	p.Want = CloneObservation(p.Want)
	return p
}
func invoke(p Probe, call func(Probe) Observation) (out Observation) {
	defer func() {
		if recover() != nil {
			msg := "upstream_panic_redacted"
			out = Observation{Panicked: true, Error: &msg}
		}
	}()
	return CloneObservation(call(cloneProbe(p)))
}

type Record struct {
	Probe   Probe
	Got     Observation
	Matches bool
}
type Result struct {
	Schema, State, WantSHA256, PlanSHA256, Failure, ActiveProbeID                                                  string
	DistinctBehaviorGoals, PlannedAPICalls, DispatchReserved, OriginalReturns, WrapperPanics, Matches, Differences int
	Records                                                                                                        []Record
	PackageInitializationScope                                                                                     string
	OriginalInitializationDynamicallyInstrumented                                                                  bool
	CaptionsQualified, ParentLabelsQualified, BroadTruthProven, TrainingReady                                      bool
	NewParents, NewLabels, ModelCalls, Fits, ProtectedFinalReads                                                   int
}

func Observe(s Specification, wantSHA, planSHA string, call func(Probe) Observation, checkpoint func(Result) error) (Result, error) {
	r := Result{Schema: "riido-wrap-marshal75-finite-observation-v1", State: "running_finite_observations", WantSHA256: wantSHA, PlanSHA256: planSHA, DistinctBehaviorGoals: len(s.Goals), PlannedAPICalls: len(s.Probes), Records: []Record{}, PackageInitializationScope: "One original-import child initializes Go packages before main. Source-inspected godotenv parser globals are three errors.New and three regexp.MustCompile expressions; wordwrap has no dynamic package globals. Initializer counts/types are not dynamically instrumented and are distinct from 24 direct API dispatches."}
	if !reflect.DeepEqual(s, LiteralSpecification()) || len(s.Probes) > MaxProbes || call == nil || checkpoint == nil {
		return r, errors.New("fixed_spec_refused")
	}
	for _, p := range s.Probes {
		r.ActiveProbeID = p.ID
		r.DispatchReserved++
		if err := checkpoint(r); err != nil {
			r.Failure = "checkpoint_failed_before_dispatch"
			return r, err
		}
		got := invoke(p, call)
		if got.Panicked {
			r.WrapperPanics++
		} else {
			r.OriginalReturns++
		}
		match := reflect.DeepEqual(got, p.Want)
		if match {
			r.Matches++
		} else {
			r.Differences++
		}
		r.Records = append(r.Records, Record{Probe: cloneProbe(p), Got: got, Matches: match})
		r.ActiveProbeID = ""
		if err := checkpoint(r); err != nil {
			r.Failure = "checkpoint_failed_after_dispatch"
			return r, err
		}
	}
	r.State = "completed_finite_observations_all_match"
	if r.Differences != 0 {
		r.State = "completed_finite_observations_with_differences"
	}
	if err := checkpoint(r); err != nil {
		r.Failure = "checkpoint_failed_final"
		return r, err
	}
	return r, nil
}
