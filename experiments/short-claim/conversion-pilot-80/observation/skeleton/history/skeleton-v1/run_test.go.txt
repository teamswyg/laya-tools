func uuidReturn() Return { u := [16]byte{1}; return Return{UUID: &u, ErrorChannel: true} }
func TestSyntheticCoordinatesAndRegistryOwnership(t *testing.T) {
	cs, parents, goals := Candidates(), Parents(), Goals()
	rows := make([]Literal, 72)
	for g := 0; g < 3; g++ {
		for c := 0; c < 3; c++ {
			for f := 0; f < 8; f++ {
				idx := g*24 + c*8 + f
				rows[idx] = Literal{Ordinal: idx, GoalIndex: g, CandidateIndex: c, FixtureIndex: f, OriginalFixtureOrdinal: g*8 + f, ParentID: parents[g], GoalID: goals[g], CandidateID: cs[g][c].ID, FunctionName: cs[g][c].Function}
			}
		}
	}
	if e := Check72Coordinates(rows); e != nil {
		t.Fatal(e)
	}
	for _, mutate := range []func([]Literal){func(v []Literal) { v[1], v[8] = v[8], v[1] }, func(v []Literal) { v[0].FunctionName = "AnyOther" }, func(v []Literal) { v[7].OriginalFixtureOrdinal = 8 }, func(v []Literal) { v[30].ParentID = "other" }} {
		v := append([]Literal(nil), rows...)
		mutate(v)
		if Check72Coordinates(v) == nil {
			t.Fatal("changed binding accepted")
		}
	}
	if Check72Coordinates(rows[:71]) == nil {
		t.Fatal("cardinality accepted")
	}
	cs[0][0].ID = "mutated"
	parents[0] = "mutated"
	if Candidates()[0][0].ID == "mutated" || Parents()[0] == "mutated" {
		t.Fatal("registry ownership lost")
	}
}

type methodTrap struct{ ErrorCalls, StringCalls int }

func (e *methodTrap) Error() string  { e.ErrorCalls++; return "trap error" }
func (e *methodTrap) String() string { e.StringCalls++; return "trap string" }
func TestSyntheticPrimaryPanicNeverFormatsPayload(t *testing.T) {
	trap := &methodTrap{}
	before := [16]byte{255, 238}
	c := toy()
	c.Literal.GoalIndex = 1
	c.ReceiverBefore = &before
	c.Literal.Want = Channels{ReceiverBeforeHex: hexValue(&before), PanicPresent: true, PanicType: ptr("*pure.methodTrap")}
	c.Literal.WantObserved.NonStringPanicTextUnobserved = true
	s, e := Run([]Case{c}, Budget{1, 48}, func(Literal) Return { panic(trap) }, func(RunState) error { return nil })
	got := s.Records[0].Got
	if e != nil || trap.ErrorCalls != 0 || trap.StringCalls != 0 || s.Counters.CandidatePanicked != 1 || s.Counters.CandidateReturned != 0 || s.Counters.ErrorDispatched != 0 || s.Counters.Matches != 1 {
		t.Fatal("panic payload dispatched methods or counters lost")
	}
	if got.UUIDHex != nil || got.ReceiverAfterHex != nil || got.ExactOutputString != nil || got.ErrorNil != nil || got.ErrorType != nil || got.ErrorMessage != nil || got.PanicText != nil || got.ReturnedNormally || !s.Records[0].NonStringPanicTextUnobserved {
		t.Fatal("unreturned channels forged")
	}
}
func TestSyntheticReturnedErrorInvokesOnlyOneError(t *testing.T) {

func TestSyntheticCheckpointFailureStopsBeforeCalls(t *testing.T) {
	for _, phase := range []string{"before_candidate", "after_candidate", "before_Error", "after_record", "completed"} {
		t.Run(phase, func(t *testing.T) {
			c := toy()
			trap := &methodTrap{}
			candidateCalls := 0
			s, e := Run([]Case{c}, Budget{1, 1}, func(Literal) Return { candidateCalls++; return Return{ErrorChannel: true, Err: trap} }, func(s RunState) error {
				if s.Phase == phase {
					return errors.New("private-path-and-token-must-not-escape")
				}
				return nil
			})
			if e == nil || strings.Contains(e.Error(), "private") || strings.Contains(s.Failure, "private") || s.Failure == "" {
				t.Fatal("fixed failure missing")
			}
			if phase == "before_candidate" {
				if candidateCalls != 0 || s.Counters.CandidateDispatched != 0 || s.Counters.CandidateReserved != 1 || len(s.Records) != 1 || s.Records[0].Got != nil {
					t.Fatal("candidate dispatched before sync")
				}
			} else if candidateCalls != 1 {
				t.Fatal("candidate re-run or lost")
			}
			if phase == "before_Error" || phase == "after_candidate" {
				if trap.ErrorCalls != 0 || s.Counters.ErrorDispatched != 0 || s.Records[0].Got.ErrorNil == nil || *s.Records[0].Got.ErrorNil {
					t.Fatal("Error dispatched or primary return lost")
				}
			}
		})
	}
}

	a, b, c := toy(), toy(), toy()

	calls := 0
	s, e := Run(cases, Budget{3, 0}, func(l Literal) Return { calls++; l.Input[0] = 'x'; return Return{ErrorChannel: true} }, func(snapshot RunState) error {

	}
}
