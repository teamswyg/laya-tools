func bp(v bool) *bool { return &v }
func ip(v int) *int   { return &v }
func marshal(t *testing.T, v any) []byte {
	t.Helper()
	b, e := json.Marshal(v)
	if e != nil {
		t.Fatal(e)
	}
	return b
}

func shapeCheck(b bundle) (summary, error) {
	out := initialSummary()
	e := validateBound(b, func([]byte) error { return nil }, &out)
	return out, e
}
func changeRow(t *testing.T, b *bundle, index int, f func(*row)) {
	t.Helper()
	lines := bytes.Split(bytes.TrimSuffix(b.Data, []byte{'\n'}), []byte{'\n'})
	var r row
	if json.Unmarshal(lines[index], &r) != nil {
		t.Fatal("row decode")
	}
	f(&r)
	lines[index] = marshal(t, r)
	b.Data = append(bytes.Join(lines, []byte{'\n'}), '\n')
}
func changeAdoption(t *testing.T, b *bundle, f func(*adoption)) {
	t.Helper()
	var a adoption
	if json.Unmarshal(b.Adoption, &a) != nil {
		t.Fatal("adoption decode")
	}
	f(&a)
	b.Adoption = marshal(t, a)
}

	}
}

func TestAdversarialRows(t *testing.T) {
	cases := []struct {
		name   string
		change func(*row)
	}{
		{"caption_edit", func(r *row) { r.Candidates[1].Text += " edit" }},
		{"request_edit", func(r *row) { r.Request += " edit" }},
		{"candidate_reorder", func(r *row) { r.Candidates[0], r.Candidates[1] = r.Candidates[1], r.Candidates[0] }},
		{"candidate_id_edit", func(r *row) { r.Candidates[1].ID = "other" }},
		{"false_label", func(r *row) { r.Candidates[1].Label = bp(false) }},
		{"missing_false_label", func(r *row) { r.Candidates[0].Label = nil }},
		{"weighted", func(r *row) { r.Candidates[0].Weight = 2 }},
		{"new_family", func(r *row) { r.Family = "new-source" }},
		{"wrong_group", func(r *row) { r.Group = 76 }},
		{"wrong_revision", func(r *row) { r.Revision = strings.Repeat("a", 40) }},
		{"wrong_text_revision", func(r *row) { r.TextRevision = "v-other" }},
		{"overclaimed_scope", func(r *row) { r.Scope.All = bp(true) }},
		{"unseen_scope", func(r *row) { r.Scope.Unseen = bp(true) }},
		{"scope_count", func(r *row) { r.Scope.Inputs++ }},
		{"heldout_role", func(r *row) { r.Role = "evaluation" }},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			b := fakeBundle(t)
			changeRow(t, &b, 7, tc.change)
			if _, e := shapeCheck(b); e == nil {
				t.Fatal("bad row accepted")
			}
		})
	}
}

		})
	}

func TestPinnedRegularBoundsAndDigest(t *testing.T) {
	d := t.TempDir()
	p := filepath.Join(d, "input")
	raw := []byte("{}\n")
	if os.WriteFile(p, raw, 0600) != nil {
		t.Fatal("write")
	}
	got, e := pinned(p, digest(raw), len(raw))
	if e != nil || !bytes.Equal(got, raw) {
		t.Fatal(e)
	}
	if _, e = pinned(p, strings.Repeat("0", 64), len(raw)); e == nil {
		t.Fatal("bad hash")
	}
	if _, e = pinned(p, digest(raw), len(raw)+1); e == nil {
		t.Fatal("bad size")
	}
	link := filepath.Join(d, "link")
	if os.Symlink(p, link) != nil {
		t.Fatal("link")
	}
	if _, e = pinned(link, digest(raw), len(raw)); e == nil {
		t.Fatal("symlink")
	}
	if _, e = pinned(d, digest(raw), 0); e == nil {
		t.Fatal("directory")
	}
}
