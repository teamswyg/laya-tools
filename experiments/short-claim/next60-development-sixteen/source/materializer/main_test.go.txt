func TestOwnedComposePreservesPrefixExactTextAndAdoptedOrder(t *testing.T) {
	previous, c, q := owned(t)
	got, s, e := compose(previous, c, q)
	if e != nil {
		t.Fatal(e)
	}
	if !bytes.HasPrefix(got, previous) || s.RowsAppended != 9 || s.Requests != 16 || s.Labels != 47 || s.Positive != 16 || s.Negative != 31 || s.Inputs != 80 || s.Observations != 236 || s.SHA != digest(got) {
		t.Fatal("wrong composition")
	}
	lines := bytes.Split(bytes.TrimSuffix(got, []byte{'\n'}), []byte{'\n'})
	if len(lines) != 16 {
		t.Fatal("wrong row count")
	}
	for i := range requestIDs {
		var r Row
		if json.Unmarshal(lines[7+i], &r) != nil {
			t.Fatal("bad JSON row")
		}
		if r.Request != c.Requests[i].Request || r.StableID != q.Rows[i].StableID || r.Role != role || r.Group != q.Rows[i].Group || r.Family != q.Rows[i].Family || r.Revision != q.Rows[i].Revision || r.TextRevision != "catalog9-finite-v2" || r.FeaturePolicy != featurePolicy || r.Scope.Description != description || r.Scope.AllInputs || r.Scope.Unseen {
			t.Fatal("text/metadata changed")
		}
		for j, cid := range candidateIDs {
			a := r.Candidates[j]
			if a.ID != cid || a.Text != c.Requests[i].Captions[j].Text || a.Label != *q.Rows[i].Candidates[j].Label || a.Weight != q.Rows[i].Candidates[j].Weight {
				t.Fatal("candidate mapping changed")
			}
		}
	}
	if bytes.Contains(got, []byte("owned-HOLD-not-copied")) {
		t.Fatal("appended tenth hold")
	}
	if s.NewQualification || s.UnknownToFalse || s.ReaderCalls+s.OriginalCalls+s.OracleCalls+s.Features+s.Score+s.Project+s.Fit+s.Model != 0 || s.TrainingReady {
		t.Fatal("granted unrelated authority")
	}
}
func TestOwnedComposeRejectsMisbindings(t *testing.T) {
	for _, kind := range []string{"request", "catalog-order", "caption-index", "qualification-index", "stable-id", "group", "family", "revision", "fixtures", "missing-label", "changed-label", "weight", "unknown-to-false", "aggregate", "role", "catalog-sha", "short-order", "extra-candidate", "text-cap", "previous-count", "previous-missing-lf"} {
		t.Run(kind, func(t *testing.T) {
			previous, c, q := owned(t)
			switch kind {
			case "request":
				c.Requests[0].ID = "other"
			case "catalog-order":
				c.Requests[0].Order[0], c.Requests[0].Order[1] = c.Requests[0].Order[1], c.Requests[0].Order[0]
			case "caption-index":
				c.Requests[0].Captions[0].Index = 1
			case "qualification-index":
				q.Rows[0].Candidates[0].Index = 1
			case "stable-id":
				q.Rows[0].StableID = "next60-other"
			case "group":
				q.Rows[0].Group = 76
			case "family":
				q.Rows[0].Family = "other"
			case "revision":
				q.Rows[0].Revision = strings.Repeat("b", 40)
			case "fixtures":
				q.Rows[0].Fixtures++
			case "missing-label":
				q.Rows[0].Candidates[0].Label = nil
			case "changed-label":
				v := true
				q.Rows[0].Candidates[0].Label = &v
			case "weight":
				q.Rows[0].Candidates[0].Weight = 2
			case "unknown-to-false":
				q.UnknownToFalse = true
			case "aggregate":
				q.Combined.Requests++
			case "role":
				q.Role = "validation"
			case "catalog-sha":
				q.CatalogSHA = strings.Repeat("a", 64)
			case "short-order":
				c.Requests[0].Order = c.Requests[0].Order[:2]
			case "extra-candidate":
				q.Rows[0].Candidates = append(q.Rows[0].Candidates, q.Rows[0].Candidates[0])
			case "text-cap":
				c.Requests[0].Request = strings.Repeat("x", 513)
			case "previous-count":
				previous = append(previous, []byte("{}\n")...)
			case "previous-missing-lf":
				previous = previous[:len(previous)-1]
			}
			if _, _, e := compose(previous, c, q); e == nil {
				t.Fatal("accepted altered binding")
			}
		})
	}
}
func TestOwnedPinAndRegularBounds(t *testing.T) {
	dir := t.TempDir()
	path := filepath.Join(dir, "owned")
	raw := []byte("owned")
	if os.WriteFile(path, raw, 0600) != nil {
		t.Fatal("write fixture")
	}
	got, _, e := pinned(path, digest(raw), len(raw))
	if e != nil || !bytes.Equal(got, raw) {
		t.Fatal("exact pin failed", e)
	}
	if _, _, e := pinned(path, strings.Repeat("a", 64), len(raw)); e == nil {
		t.Fatal("changed SHA accepted")
	}
	if _, _, e := pinned(path, digest(raw), len(raw)-1); e == nil {
		t.Fatal("wrong size accepted")
	}
	if _, _, e := pinned(dir, digest(raw), len(raw)); e == nil {
		t.Fatal("directory accepted")
	}
	link := filepath.Join(dir, "link")
	if os.Symlink(path, link) != nil {
		t.Fatal("link fixture")
	}
	if _, _, e := pinned(link, digest(raw), len(raw)); e == nil {
		t.Fatal("symlink leaf accepted")
	}
}
func TestOwnedExclusiveWriteSyncAndExistingPreservation(t *testing.T) {
	dir := t.TempDir()
	file := filepath.Join(dir, "owned.jsonl")
	raw := []byte("owned\n")
	s := Summary{}
	if e := writeExclusive(file, raw, &s); e != nil {
		t.Fatal(e)
	}
	if !s.OutputReserved || s.BytesWritten != len(raw) || !s.FileSynced || !s.DirectorySynced {
		t.Fatal("lost durable write facts")
	}
	if e := writeExclusive(file, []byte("replacement\n"), &Summary{}); e == nil {
		t.Fatal("overwrote existing output")
	}
	got, e := os.ReadFile(file)
	if e != nil || !bytes.Equal(got, raw) {
		t.Fatal("changed previous output")
	}
	if e := writeExclusive(filepath.Join(dir, "oversize"), make([]byte, maxData+1), &Summary{}); e == nil {
		t.Fatal("oversize accepted")
	}
	if _, e := os.Lstat(filepath.Join(dir, "oversize")); !os.IsNotExist(e) {
		t.Fatal("created oversize output")
	}
}
