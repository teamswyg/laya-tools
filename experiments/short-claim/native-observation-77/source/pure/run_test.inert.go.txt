type fakeDecoder struct {
	record, at                     int
	key, value                     []byte
	buf                            [2]byte
	infiniteRecords, infinitePairs bool
	err                            error
}

func (d *fakeDecoder) ScanRecord() bool {
	d.buf = [2]byte{'X', 'Y'}
	d.record++
	d.at = 0
	return d.infiniteRecords || d.record <= 2
}
func (d *fakeDecoder) ScanKeyval() bool {
	n := 2
	if d.record == 2 {
		n = 1
	}
	if d.record > 2 || d.at >= n && !d.infinitePairs {
		d.key = nil
		d.value = nil
		return false
	}
	d.at++
	d.buf = [2]byte{'a', byte('0' + d.at)}
	if d.record == 2 {
		d.buf = [2]byte{'b', '3'}
	}
	d.key = d.buf[:1]
	d.value = d.buf[1:]
	return true
}
func (d *fakeDecoder) Key() []byte   { return d.key }
func (d *fakeDecoder) Value() []byte { return d.value }
func (d *fakeDecoder) Err() error    { return d.err }
func stubs() Ops {
	return Ops{New: func(string, int) Decoder { return &fakeDecoder{} }, ErrorInfo: func(e error) ErrorState { return ErrorState{Type: "toy_error", Message: e.Error()} }, Escape: func(b []byte) string { return string(b) }, Unescape: func(s string) ([]byte, error) {
		if s == "" {
			return nil, nil
		}
		return []byte(s), nil
	}}
}
func runToy(w WantSet, o Ops) (Result, error) {
	r := NewResult(w, "toy-only")
	e := Run(w, o, &r, func(*Result) error { return nil })
	return r, e
}

func TestStubOrderAndBorrowedSnapshots(t *testing.T) {
	r, e := runToy(Specifications(), stubs())
	if e != nil {
		t.Fatal(e)
	}
	if r.Started != 24 || r.Finished != 24 || r.State != "completed_with_differences_no_qualification" {
		t.Fatal("all fixture differences must survive")
	}
	l := r.Cases[2].Got.Log
	if len(l.Records) != 2 || l.Records[0].Steps[0].Key.Hex != "61" || l.Records[0].Steps[0].Value.Hex != "31" || l.Records[0].Steps[1].Key.Hex != "61" || l.Records[0].Steps[1].Value.Hex != "32" || l.Records[1].Steps[0].Key.Hex != "62" {
		t.Fatal("order, duplicates or borrowed materialization")
	}
	if l.Records[0].Steps[2].Key.Nil != true || l.StickyKeyval != nil || l.StickyRecord != nil || l.SkippedReason != "no_current_record_after_EOF" {
		t.Fatal("terminal nil")
	}
	if r.Calls[0].Returned != 11 || r.Calls[1].Returned != 1 || r.Calls[8].Returned != 4 || r.Calls[9].Returned != 8 {
		t.Fatal("constructors and percent calls are not24API calls")
	}
}

	o := stubs()

			panic("private panic text must not be copied")
		}

	}
	r, e := runToy(Specifications(), o)

		t.Fatal("panic leaked")
	}
}

	for _, which := range []string{"records", "pairs"} {
		t.Run(which, func(t *testing.T) {

				return &fakeDecoder{infiniteRecords: which == "records", infinitePairs: which == "pairs"}

			r, e := runToy(Specifications(), o)
			if e == nil || r.Finished != 24 || r.Cases[0].Failure == "" {
				t.Fatal("bound accepted")
			}
			for _, c := range r.Cases {
				for i, v := range c.Calls {
					if v.Reserved > caps[i] {
						t.Fatal("API cap")
					}
				}
			}
		})
	}
}

type eofOnlyDecoder struct{}

func (*eofOnlyDecoder) ScanRecord() bool { return false }
func (*eofOnlyDecoder) ScanKeyval() bool { panic("no current record") }
func (*eofOnlyDecoder) Key() []byte      { panic("no current record") }
func (*eofOnlyDecoder) Value() []byte    { panic("no current record") }
func (*eofOnlyDecoder) Err() error       { return nil }
func TestEOFSkipsNonexistentRecordAndReservesNoPostEOFCalls(t *testing.T) {
	o := stubs()
	o.New = func(string, int) Decoder { return &eofOnlyDecoder{} }
	r, e := runToy(Specifications(), o)
	if e != nil {
		t.Fatal(e)
	}
	for _, c := range r.Cases[:12] {
		l := c.Got.Log
		if l.StickyRecord != nil || l.StickyKeyval != nil || l.SkippedReason != "no_current_record_after_EOF" || c.Calls[2].Returned != 1 || c.Calls[3].Reserved != 0 || c.Calls[4].Reserved != 0 || c.Calls[5].Reserved != 0 || c.Calls[6].Returned != 1 || c.Got.Panic {
			t.Fatal("post-EOF call or false materialization")
		}
	}
}

type stickyErrorDecoder struct{ e error }

func (*stickyErrorDecoder) ScanRecord() bool { return false }
func (*stickyErrorDecoder) ScanKeyval() bool { return false }
func (*stickyErrorDecoder) Key() []byte      { return nil }
func (*stickyErrorDecoder) Value() []byte    { return nil }
func (d *stickyErrorDecoder) Err() error     { return d.e }
func TestOnlyNonNilTerminalErrorAllowsStickyScans(t *testing.T) {
	o := stubs()
	o.New = func(string, int) Decoder { return &stickyErrorDecoder{e: errors.New("toy sticky error")} }
	r, e := runToy(Specifications(), o)
	if e != nil {
		t.Fatal(e)
	}
	for _, c := range r.Cases[:12] {
		l := c.Got.Log
		if l.TerminalError == nil || l.StickyRecord == nil || *l.StickyRecord || l.StickyKeyval == nil || l.StickyKeyval.ScanKeyval || l.SkippedReason != "" || c.Calls[2].Returned != 2 || c.Calls[3].Returned != 1 || c.Calls[4].Returned != 1 || c.Calls[5].Returned != 1 || c.Calls[6].Returned != 3 || c.Calls[7].Returned != 3 {
			t.Fatal("sticky-error counters")
		}
	}
}
