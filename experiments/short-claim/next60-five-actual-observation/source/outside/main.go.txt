func failure(s string) error { return errors.New(s) }
func fixedCode(e error) string {
	if e == nil {
		return ""
	}
	s := e.Error()
	allowed := [14]string{"nextfive_outside_config_invalid", "nextfive_outside_pin_invalid", "nextfive_outside_build_invalid",
		"nextfive_outside_path_invalid", "nextfive_outside_output_invalid", "nextfive_outside_start_failed",
		"nextfive_outside_wait_failed", "nextfive_outside_deadline", "nextfive_outside_capture_cap",
		"nextfive_outside_worker_artifacts_invalid", "nextfive_outside_worker_protocol_invalid",
		"nextfive_outside_metrics_invalid", "nextfive_outside_rss_limit", "nextfive_outside_bindings_changed"}
	for _, a := range allowed {
		if s == a {
			return a
		}
	}
	return "nextfive_outside_output_invalid"
}
func path(s string) bool { return s != "" && filepath.IsAbs(s) && filepath.Clean(s) == s }
func contains(parent, child string) bool {
	rel, e := filepath.Rel(parent, child)
	return e == nil && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
}
func regularRead(path string, cap int) ([]byte, os.FileInfo, error) {
	st, e := os.Lstat(path)
	if e != nil || !st.Mode().IsRegular() || st.Size() < 0 || st.Size() > int64(cap) {
		return nil, nil, failure("nextfive_outside_pin_invalid")
	}
	f, e := os.Open(path)
	if e != nil {
		return nil, nil, failure("nextfive_outside_pin_invalid")
	}
	defer f.Close()
	now, e := f.Stat()
	if e != nil || !now.Mode().IsRegular() || !os.SameFile(st, now) || now.Size() != st.Size() {
		return nil, nil, failure("nextfive_outside_pin_invalid")
	}
	raw, e := io.ReadAll(io.LimitReader(f, int64(cap)+1))
	after, se := f.Stat()
	if e != nil || se != nil || len(raw) > cap || int64(len(raw)) != st.Size() || after.Size() != st.Size() || !os.SameFile(st, after) {
		return nil, nil, failure("nextfive_outside_pin_invalid")
	}
	return raw, st, nil
}
func sum(raw []byte) string { s := sha256.Sum256(raw); return hex.EncodeToString(s[:]) }
func verify(p Pin, cap int) (PinFact, os.FileInfo, error) {
	fact := PinFact{ID: p.ID, Bytes: p.Bytes, SHA: p.SHA}
	if p.ID == "" || !path(p.Path) || p.Bytes <= 0 || p.Bytes > cap || !protocol.SHA(p.SHA) {
		return fact, nil, failure("nextfive_outside_pin_invalid")
	}
	raw, st, e := regularRead(p.Path, cap)
	if e != nil || len(raw) != p.Bytes || sum(raw) != p.SHA {
		return fact, nil, failure("nextfive_outside_pin_invalid")
	}
	fact.Verified = true
	return fact, st, nil
}
func pin(raw []byte) (Pin, error) {
	v, e := protocol.Object(raw, []string{"id", "path", "bytes", "sha256"})
	if e != nil {
		return Pin{}, failure("nextfive_outside_config_invalid")
	}
	var p Pin
	p.ID, e = protocol.ReadString(v[0])
	if e != nil {
		return p, failure("nextfive_outside_config_invalid")
	}
	p.Path, e = protocol.ReadString(v[1])
	if e != nil {
		return p, failure("nextfive_outside_config_invalid")
	}
	p.Bytes, e = protocol.ReadInt(v[2])
	if e != nil {
		return p, failure("nextfive_outside_config_invalid")
	}
	p.SHA, e = protocol.ReadString(v[3])
	if e != nil {
		return p, failure("nextfive_outside_config_invalid")
	}
	return p, nil
}
func readConfig(file, sha, out string) (Config, error) {
	var c Config
	raw, _, e := regularRead(file, configCap)
	if e != nil || !protocol.SHA(sha) || sum(raw) != sha {
		return c, failure("nextfive_outside_config_invalid")
	}
	v, e := protocol.Object(raw, []string{"schema", "controller_directory", "worker_attempt_directory", "worker_binary",
		"controller_binary", "execution_plan", "time_binary", "root_receipts", "max_wall_seconds",
		"max_rss_bytes", "heap_limit_bytes", "max_stdout_bytes", "max_stderr_bytes"})
	if e != nil {
		return c, failure("nextfive_outside_config_invalid")
	}
	s, e := protocol.ReadString(v[0])
	if e != nil || s != schema {
		return c, failure("nextfive_outside_config_invalid")
	}
	c.Directory, e = protocol.ReadString(v[1])
	if e != nil || c.Directory != out || !path(c.Directory) {
		return c, failure("nextfive_outside_config_invalid")
	}
	c.Attempt, e = protocol.ReadString(v[2])
	if e != nil || !path(c.Attempt) || contains(c.Directory, c.Attempt) || contains(c.Attempt, c.Directory) {
		return c, failure("nextfive_outside_path_invalid")
	}
	pins := [4]*Pin{&c.Worker, &c.Self, &c.Plan, &c.Time}
	for i, p := range pins {
		*p, e = pin(v[3+i])
		if e != nil {
			return c, e
		}
	}
	if c.Worker.ID != "nextfive/worker" || c.Self.ID != "nextfive/controller" ||
		c.Plan.ID != "nextfive/execution-plan" || c.Time.ID != "nextfive/system-time" ||
		c.Plan.Bytes <= 0 || c.Plan.Bytes > configCap || !protocol.SHA(c.Plan.SHA) || c.Time.Path != "/usr/bin/time" {
		return c, failure("nextfive_outside_config_invalid")
	}
	var evidence []json.RawMessage
	if json.Unmarshal(v[7], &evidence) != nil || len(evidence) != 4 {
		return c, failure("nextfive_outside_config_invalid")
	}
	required := [4]string{"nextfive/root-source", "nextfive/root-compiler", "nextfive/root-rights", "nextfive/root-resource"}
	c.Evidence = make([]Pin, 4)
	for i, r := range evidence {
		c.Evidence[i], e = pin(r)
		if e != nil || c.Evidence[i].ID != required[i] {
			return c, failure("nextfive_outside_config_invalid")
		}
	}
	c.Wall, e = protocol.ReadInt(v[8])
	if e != nil || c.Wall != 60 {
		return c, failure("nextfive_outside_config_invalid")
	}
	expected := [4]int{heapLimit, heapLimit, stdoutCap, stderrCap}
	for i, want := range expected {
		n, e := protocol.ReadInt(v[9+i])
		if e != nil || n != want {
			return c, failure("nextfive_outside_config_invalid")
		}
	}
	executable, e := os.Executable()
	if e != nil || filepath.Clean(executable) != c.Self.Path {
		return c, failure("nextfive_outside_pin_invalid")
	}
	all := []Pin{c.Worker, c.Self, c.Plan, c.Time}
	all = append(all, c.Evidence...)
	for i, p := range all {
		if contains(c.Directory, p.Path) || contains(c.Attempt, p.Path) {
			return c, failure("nextfive_outside_path_invalid")
		}
		for j := 0; j < i; j++ {
			if p.Path == all[j].Path {
				return c, failure("nextfive_outside_path_invalid")
			}
		}
	}
	if contains(c.Directory, file) || contains(c.Attempt, file) {
		return c, failure("nextfive_outside_path_invalid")
	}
	return c, nil
}
func build(file, module, pkg string) error {
	m, e := macho.Open(file)
	if e != nil {
		return failure("nextfive_outside_build_invalid")
	}
	ok := m.Magic == macho.Magic64 && m.Cpu == macho.CpuArm64 && m.Type == macho.TypeExec
	m.Close()
	if !ok {
		return failure("nextfive_outside_build_invalid")
	}
	info, e := buildinfo.ReadFile(file)
	if e != nil || info.GoVersion != "go1.27.1" || info.Path != pkg || info.Main.Path != module {
		return failure("nextfive_outside_build_invalid")
	}
	required := [5][2]string{{"GOOS", "darwin"}, {"GOARCH", "arm64"}, {"CGO_ENABLED", "0"}, {"-trimpath", "true"}, {"-buildmode", "exe"}}
	for _, r := range required {
		found := false
		for _, s := range info.Settings {
			if s.Key == r[0] {
				if s.Value != r[1] {
					return failure("nextfive_outside_build_invalid")
				}
				found = true
			}
		}
		if !found {
			return failure("nextfive_outside_build_invalid")
		}
	}
	return nil
}
func verifyAll(c Config) ([]PinFact, error) {
	pins := []Pin{c.Worker, c.Self, c.Plan, c.Time}
	pins = append(pins, c.Evidence...)
	facts := make([]PinFact, 0, len(pins))
	info := make([]os.FileInfo, 0, len(pins))
	for i, p := range pins {
		cap := binaryCap
		if i == 2 {
			cap = configCap
		}
		if i >= 4 {
			cap = evidenceCap
		}
		f, st, e := verify(p, cap)
		facts = append(facts, f)
		if e != nil {
			return facts, e
		}
		for _, old := range info {
			if os.SameFile(st, old) {
				return facts, failure("nextfive_outside_path_invalid")
			}
		}
		info = append(info, st)
	}
	return facts, nil
}

// Bind the canonical plan and its four exact input files before package init.
// Full source/compiler/notice conclusions remain Root-owned frozen evidence;
// this is not an exhaustive module/cache/generated/OS closure proof.
func bindPlan(c Config) (p protocol.Plan, cases [protocol.CaseCount]string, facts []PinFact, retErr error) {
	raw, _, e := regularRead(c.Plan.Path, configCap)
	if e != nil || len(raw) != c.Plan.Bytes || sum(raw) != c.Plan.SHA {
		return p, cases, facts, failure("nextfive_outside_pin_invalid")
	}
	p, e = protocol.ReadPlan(raw, c.Worker.SHA)
	if e != nil || p.WallSeconds != c.Wall || contains(p.SourceRoot, c.Directory) || contains(p.SourceRoot, c.Attempt) || contains(c.Directory, p.SourceRoot) || contains(c.Attempt, p.SourceRoot) {
		return p, cases, facts, failure("nextfive_outside_path_invalid")
	}
	real, e := filepath.EvalSymlinks(p.SourceRoot)
	st, se := os.Lstat(p.SourceRoot)
	if e != nil || se != nil || real != p.SourceRoot || !st.IsDir() || st.Mode()&os.ModeSymlink != 0 {
		return p, cases, facts, failure("nextfive_outside_path_invalid")
	}
	inputs := [4]protocol.RelativePin{p.Manifest, p.Fixtures, p.Wants, p.Readiness}
	var physical [4]os.FileInfo
	for i, input := range inputs {
		cur := p.SourceRoot
		for _, part := range strings.Split(input.Path, "/") {
			cur = filepath.Join(cur, part)
			st, e := os.Lstat(cur)
			if e != nil || st.Mode()&os.ModeSymlink != 0 {
				return p, cases, facts, failure("nextfive_outside_pin_invalid")
			}
		}
		pin := Pin{ID: input.ID, Path: cur, Bytes: int(input.Bytes), SHA: input.SHA256}
		fact, st, e := verify(pin, 1<<20)
		facts = append(facts, fact)
		if e != nil {
			return p, cases, facts, e
		}
		for j := 0; j < i; j++ {
			if os.SameFile(st, physical[j]) {
				return p, cases, facts, failure("nextfive_outside_path_invalid")
			}
		}
		physical[i] = st
		if i == 1 {
			raw, _, e := regularRead(cur, 1<<20)
			if e != nil {
				return p, cases, facts, e
			}
			cases, e = protocol.ReadFixtures(raw)
			if e != nil {
				return p, cases, facts, failure("nextfive_outside_worker_protocol_invalid")
			}
		}
	}
	return p, cases, facts, nil
}

type capture struct {
	file     *os.File
	cap, n   int
	exceeded bool
	signal   chan<- struct{}
}

func (w *capture) Write(p []byte) (int, error) {
	if len(p) > w.cap-w.n {
		left := w.cap - w.n
		if left > 0 {
			n, e := w.file.Write(p[:left])
			w.n += n
			if e != nil {
				return n, e
			}
		}
		w.exceeded = true
		select {
		case w.signal <- struct{}{}:
		default:
		}
		return left, failure("nextfive_outside_capture_cap")
	}
	n, e := w.file.Write(p)
	w.n += n
	return n, e
}
func write(f *os.File, v any, cap int) error {
	raw, e := json.Marshal(v)
	if e != nil {
		return failure("nextfive_outside_output_invalid")
	}
	raw = append(raw, '\n')
	if len(raw) > cap {
		return failure("nextfive_outside_output_invalid")
	}
	n, e := f.Write(raw)
	if e != nil || n != len(raw) || f.Sync() != nil {
		return failure("nextfive_outside_output_invalid")
	}
	return nil
}
func syncDirectory(path string) error {
	before, e := os.Lstat(path)
	if e != nil || !before.IsDir() || before.Mode()&os.ModeSymlink != 0 {
		return failure("nextfive_outside_output_invalid")
	}
	f, e := os.Open(path)
	if e != nil {
		return failure("nextfive_outside_output_invalid")
	}
	defer f.Close()
	now, e := f.Stat()
	if e != nil || !now.IsDir() || !os.SameFile(before, now) || f.Sync() != nil {
		return failure("nextfive_outside_output_invalid")
	}
	return nil
}

type waitOutcome struct {
	Err                   error
	Deadline, Cap, Killed bool
	KillOK                *bool
}

func await(done <-chan error, deadline <-chan time.Time, capSignal <-chan struct{}, kill func() bool) waitOutcome {
	var out waitOutcome
	select {
	case out.Err = <-done:
		return out
	case <-deadline:
		out.Deadline = true
	case <-capSignal:
		out.Cap = true
	}
	out.Killed = true
	ok := kill()
	out.KillOK = &ok
	// One existing Wait goroutine reaps the started wrapper after termination.
	// SIGKILL/reaping is an OS operation, not a formal whole-kernel time proof.
	out.Err = <-done
	return out
}
func parseMetrics(raw []byte) Metrics {
	out := Metrics{Scope: "Darwin /usr/bin/time -l worker child process, including startup; not model or GPU"}
	rssN, timeN := 0, 0
	for _, line := range strings.Split(string(raw), "\n") {
		f := strings.Fields(line)
		if len(f) == 5 && f[1] == "maximum" && f[2] == "resident" && f[3] == "set" && f[4] == "size" {
			n, e := strconv.ParseInt(f[0], 10, 64)
			if e != nil || n <= 0 {
				return out
			}
			out.MaximumRSSBytes = &n
			rssN++
		}
		if len(f) == 6 && f[1] == "real" && f[3] == "user" && f[5] == "sys" {
			var vals [3]float64
			for i := range vals {
				n, e := strconv.ParseFloat(f[i*2], 64)
				if e != nil || math.IsNaN(n) || math.IsInf(n, 0) || n < 0 {
					return out
				}
				vals[i] = n
			}
			out.RealSeconds = &vals[0]
			out.UserSeconds = &vals[1]
			out.SystemSeconds = &vals[2]
			timeN++
		}
	}
	out.Valid = rssN == 1 && timeN == 1
	return out
}
func artifact(file, name string, cap int) ([]byte, Artifact, error) {
	raw, _, e := regularRead(file, cap)
	a := Artifact{Name: name}
	if e != nil {
		return nil, a, failure("nextfive_outside_worker_artifacts_invalid")
	}
	a.Bytes = len(raw)
	a.SHA = sum(raw)
	a.ReadValid = true
	return raw, a, nil
}
func run(configFile, configSHA, outPath string) (r Receipt, retErr error) {
	r = Receipt{Schema: "riido-next-five-outside-receipt-v1", State: "failed", ConfigSHA: configSHA,
		IncompleteActualCountsUnknown: true,
		Controls:                      "GOMAXPROCS1; GOMEMLIMIT256MiB soft; process-group wall deadline60s within300s batch ceiling; post-run Darwin maxRSS<=256MiB"}
	if !path(outPath) || !path(configFile) || !protocol.SHA(configSHA) {
		return r, failure("nextfive_outside_config_invalid")
	}
	// This fresh directory/tombstone is invocation-scoped; it is not a global
	// guarantee against an operator supplying a different config/attempt.
	if os.Mkdir(outPath, 0700) != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	if syncDirectory(outPath) != nil || syncDirectory(filepath.Dir(outPath)) != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	root, e := os.OpenRoot(outPath)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	defer root.Close()
	tomb, e := root.OpenFile("RESERVED.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	if e = write(tomb, struct {
		Schema    string `json:"schema"`
		ConfigSHA string `json:"config_sha256"`
		Started   bool   `json:"started"`
	}{
		"riido-next-five-outside-reservation-v1", configSHA, false}, 4096); e != nil {
		tomb.Close()
		return r, e
	}
	tomb.Close()
	if syncDirectory(outPath) != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	report, e := root.OpenFile("result.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	defer report.Close()
	defer func() {
		if retErr != nil {
			s := fixedCode(retErr)
			r.FailureCode = &s
			r.State = "failed"
			r.ActualObservationCount = nil
			r.IncompleteActualCountsUnknown = true
		}
		if e := write(report, r, receiptCap); e != nil {
			retErr = e
			r.State = "failed"
			s := fixedCode(e)
			r.FailureCode = &s
			r.StructuralCompletion = false
			r.ActualObservationCount = nil
			r.IncompleteActualCountsUnknown = true
		}
	}()
	c, e := readConfig(configFile, configSHA, outPath)
	if e != nil {
		return r, e
	}
	r.Pins, e = verifyAll(c)
	if e != nil {
		return r, e
	}
	r.EvidenceHashesVerified = true
	if build(c.Worker.Path, "riido.local/nextfive", "riido.local/nextfive/cmd/observe") != nil ||
		build(c.Self.Path, "example.com/riido-nextfive-outside", "example.com/riido-nextfive-outside") != nil {
		return r, failure("nextfive_outside_build_invalid")
	}
	r.StaticBuildInfoValid = true
	_, cases, planFacts, e := bindPlan(c)
	r.PlanBindings = planFacts
	if e != nil {
		return r, e
	}
	if _, e = os.Lstat(c.Attempt); !os.IsNotExist(e) {
		return r, failure("nextfive_outside_path_invalid")
	}
	stdout, e := root.OpenFile("stdout.private.bin", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	defer stdout.Close()
	stderr, e := root.OpenFile("stderr-time.private.bin", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	defer stderr.Close()
	startFile, e := root.OpenFile("START-RESERVED.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	if e = write(startFile, struct {
		ConfigSHA     string `json:"config_sha256"`
		StartAttempts int    `json:"start_attempts"`
		WorkerSHA     string `json:"worker_sha256"`
		PlanSHA       string `json:"plan_sha256"`
	}{
		configSHA, 1, c.Worker.SHA, c.Plan.SHA}, 4096); e != nil {
		startFile.Close()
		return r, e
	}
	startFile.Close()
	if stdout.Sync() != nil || stderr.Sync() != nil || report.Sync() != nil || syncDirectory(outPath) != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	// Repeat the pinned regular-file byte checks immediately before Start.
	if _, e = verifyAll(c); e != nil {
		return r, e
	}
	if _, _, _, e = bindPlan(c); e != nil {
		return r, e
	}
	capSignal := make(chan struct{}, 2)
	o := capture{file: stdout, cap: stdoutCap, signal: capSignal}
	d := capture{file: stderr, cap: stderrCap, signal: capSignal}
	cmd := exec.Command(c.Time.Path, "-l", c.Worker.Path, "--plan", c.Plan.Path, "--plan-sha256", c.Plan.SHA, "--attempt-dir", c.Attempt)
	cmd.Env = []string{"PATH=/usr/bin:/bin", "LANG=C", "LC_ALL=C", "GOMAXPROCS=1", "GOMEMLIMIT=268435456"}
	cmd.Dir = filepath.Dir(c.Worker.Path)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.WaitDelay = 2 * time.Second // bound pipe-copy delay after the process exits
	cmd.Stdout = &o
	cmd.Stderr = &d
	began := time.Now()
	r.Process.StartAttempts = 1
	startErr := cmd.Start()
	r.Process.StartSucceeded = startErr == nil
	r.Process.WaitAttempts = 1
	if startErr != nil {
		_ = cmd.Wait() // exactly one Wait call, returns not-started; no reap claim
		r.Process.WaitReturned = true
		r.Process.WallSeconds = time.Since(began).Seconds()
		return r, failure("nextfive_outside_start_failed")
	}
	done := make(chan error, 1)
	go func() { done <- cmd.Wait() }()
	timer := time.NewTimer(time.Duration(c.Wall) * time.Second)
	w := await(done, timer.C, capSignal, func() bool { return syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL) == nil })
	waitErr := w.Err
	r.Process.DeadlineExceeded = w.Deadline
	r.Process.CaptureCapExceeded = w.Cap
	r.Process.GroupKillAttempted = w.Killed
	r.Process.GroupKillSucceeded = w.KillOK
	timer.Stop()
	r.Process.WaitReturned = true
	r.Process.Reaped = true
	r.Process.WallSeconds = time.Since(began).Seconds()
	if cmd.ProcessState != nil {
		n := cmd.ProcessState.ExitCode()
		r.Process.ExitCode = &n
	}
	r.CaptureStdoutBytes = o.n
	r.CaptureStderrBytes = d.n
	if o.exceeded || d.exceeded {
		r.Process.CaptureCapExceeded = true
	}
	// Command.Wait has finished both capture-copy goroutines; files/counters
	// are now read sequentially. Each writer owns its stream; no shared locks.
	if stdout.Sync() != nil || stderr.Sync() != nil {
		return r, failure("nextfive_outside_output_invalid")
	}
	savedOut, a, e := artifact(filepath.Join(outPath, "stdout.private.bin"), "stdout.private.bin", stdoutCap)
	r.Artifacts = append(r.Artifacts, a)
	if e != nil {
		return r, e
	}
	savedErr, a, e := artifact(filepath.Join(outPath, "stderr-time.private.bin"), "stderr-time.private.bin", stderrCap)
	r.Artifacts = append(r.Artifacts, a)
	if e != nil {
		return r, e
	}
	r.Metrics = parseMetrics(savedErr)
	attemptInfo, attemptErr := os.Lstat(c.Attempt)
	attemptDirectoryValid := attemptErr == nil && attemptInfo.IsDir() && attemptInfo.Mode()&os.ModeSymlink == 0
	// Preserve parseable saved prefixes on failure. They are not complete
	// actual call totals, qualification, or an acknowledgement of success.
	rawJournal, a, je := artifact(filepath.Join(c.Attempt, "journal.jsonl"), "worker/journal.jsonl", protocol.JournalCap)
	r.Artifacts = append(r.Artifacts, a)
	if je == nil {
		j, e := protocol.ReadJournal(rawJournal, cases)
		r.Journal = &j
		r.JournalValid = e == nil
	}
	rawResult, a, re := artifact(filepath.Join(c.Attempt, "results.json"), "worker/results.json", protocol.FinalCap)
	r.Artifacts = append(r.Artifacts, a)
	var savedResult *protocol.Result
	if re == nil {
		v, e := protocol.ReadResult(rawResult, cases, c.Plan.SHA)
		if e == nil {
			savedResult = &v
			s := protocol.Summarize(v)
			r.WorkerResult = &s
		}
	}
	r.WorkerStdoutMatches = re == nil && bytes.Equal(rawResult, savedOut)
	rawReservation, a, rve := artifact(filepath.Join(c.Attempt, "reservation.json"), "worker/reservation.json", protocol.ReservationCap)
	r.Artifacts = append(r.Artifacts, a)
	_, pe := verifyAll(c)
	r.SourceBindingsStillMatch = pe == nil
	_, _, _, pbe := bindPlan(c)
	r.PlanBindingsStillMatch = pbe == nil
	if r.Process.DeadlineExceeded {
		return r, failure("nextfive_outside_deadline")
	}
	if r.Process.CaptureCapExceeded {
		return r, failure("nextfive_outside_capture_cap")
	}
	if waitErr != nil || r.Process.ExitCode == nil || *r.Process.ExitCode != 0 {
		return r, failure("nextfive_outside_wait_failed")
	}
	if pe != nil || pbe != nil {
		return r, failure("nextfive_outside_bindings_changed")
	}
	if !attemptDirectoryValid || je != nil || re != nil || rve != nil || !bytes.Equal(rawReservation, []byte(protocol.Reservation)) || len(rawJournal)+len(rawResult)+len(rawReservation) > protocol.CombinedCap {
		return r, failure("nextfive_outside_worker_artifacts_invalid")
	}
	if !r.JournalValid || r.Journal == nil || savedResult == nil || !r.WorkerStdoutMatches ||
		!protocol.Complete(*savedResult, *r.Journal) {
		return r, failure("nextfive_outside_worker_protocol_invalid")
	}
	if !r.Metrics.Valid || r.Process.WallSeconds > float64(c.Wall) {
		return r, failure("nextfive_outside_metrics_invalid")
	}
	if *r.Metrics.MaximumRSSBytes > heapLimit {
		return r, failure("nextfive_outside_rss_limit")
	}
	r.StructuralCompletion = true
	r.State = "complete"
	n := protocol.Planned
	r.ActualObservationCount = &n
	r.IncompleteActualCountsUnknown = false
	return r, nil
}
func main() {
	runtime.GOMAXPROCS(1)
	debug.SetMemoryLimit(heapLimit)
	f := flag.NewFlagSet("riido-nextfive-outside", flag.ContinueOnError)
	f.SetOutput(io.Discard)
	config := f.String("config", "", "frozen exact config")
	sha := f.String("config-sha256", "", "frozen config SHA")
	out := f.String("out", "", "fresh exclusive controller directory")
	if f.Parse(os.Args[1:]) != nil || f.NArg() != 0 || *config == "" || *sha == "" || *out == "" {
		fmt.Fprintln(os.Stderr, "nextfive_outside_arguments_invalid")
		os.Exit(2)
	}
	r, e := run(*config, *sha, *out)
	if e != nil {
		s := fixedCode(e)
		r.FailureCode = &s
	}
	summary := struct {
		Schema        string  `json:"schema"`
		State         string  `json:"state"`
		FailureCode   *string `json:"failure_code"`
		StartAttempts int     `json:"start_attempts"`
		WaitAttempts  int     `json:"wait_attempts"`
		Actual        *int    `json:"actual_completed_observation_count"`
	}{"riido-next-five-outside-summary-v1", r.State, r.FailureCode, r.Process.StartAttempts, r.Process.WaitAttempts, r.ActualObservationCount}
	raw, me := json.Marshal(summary)
	if me != nil {
		os.Exit(1)
	}
	if _, we := os.Stdout.Write(append(raw, '\n')); we != nil {
		os.Exit(1)
	}
	if e != nil {
		os.Exit(1)
	}
}
