func TestStrictCanonicalRejectsAliasesAndDuplicates(t *testing.T) {
	type fixture struct {
		Sequence int `json:"sequence"`
	}
	for _, s := range []string{"{\"Sequence\":1}\n", "{\"sequence\":0,\"sequence\":1}\n", "{\"sequence\":1,\"extra\":0}\n", "{\"sequence\":1}\n{}\n"} {
		var f fixture
		if strictCanonical([]byte(s), &f) == nil {
			t.Fatal("accepted alias/duplicate/extra/trailing")
		}
	}
	var f fixture
	if strictCanonical([]byte("{\"sequence\":1}\n"), &f) != nil || f.Sequence != 1 {
		t.Fatal("canonical valid rejected")
	}
}
