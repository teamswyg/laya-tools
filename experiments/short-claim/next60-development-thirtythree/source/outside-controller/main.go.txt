type OutsideResult struct {
	Schema                                string         `json:"schema"`
	State                                 string         `json:"state"`
	Failure                               string         `json:"failure"`
	ConfigSHA256                          string         `json:"config_sha256"`
	PlanSHA256                            string         `json:"plan_sha256"`
	WorkerSHA256                          string         `json:"worker_sha256"`
	ControllerSHA256                      string         `json:"controller_sha256"`
	Process                               ProcessReceipt `json:"process"`
	DurablePrefix                         DurableReceipt `json:"durable_prefix"`
	FramesACKWritten                      int            `json:"frames_ack_written"`
	LastDurableFrameCounts                *core.Counts   `json:"last_durable_frame_counts"`
	LastACKWrittenCounts                  *core.Counts   `json:"last_ack_written_counts"`
	TerminalRowsACKWritten                int            `json:"terminal_rows_ack_written"`
	RemainingCallsUnknown                 bool           `json:"remaining_calls_unknown"`
	FinalACKWritten                       bool           `json:"final_ack_written"`
	FinalChildFileIdentity                bool           `json:"final_child_file_identity"`
	CompleteCountersKnown                 bool           `json:"complete_counters_known"`
	CompleteRows                          *int           `json:"complete_rows"`
	InitializationCalls                   any            `json:"initialization_calls"`
	NestedOriginalCalls                   any            `json:"nested_original_calls"`
	SourceRightsCompilerResourceAuthority any            `json:"source_rights_compiler_resource_authority"`
	TruthAssigned                         int            `json:"truth_assigned"`
	LabelsAssigned                        int            `json:"labels_assigned"`
	FitOrModelCalls                       int            `json:"fit_or_model_calls"`
	OutputBytesRetained                   *int64         `json:"output_bytes_retained"`
	AggregateCap                          int            `json:"aggregate_cap_bytes"`
	NoAutomaticRetry                      bool           `json:"no_automatic_retry"`
}

func fixedFailure(e error) string {
	if e == nil {
		return ""
	}
	// All controller errors are owned static codes; never format worker errors.
	// Unknown injected errors receive a fixed fallback code.
	owned, ok := e.(failureCode)
	if !ok {
		return "controller_failed"
	}
	s := string(owned)
	if len(s) < 1 || len(s) > 96 {
		return "controller_failed"
	}
	for _, c := range s {
		if !(c >= 'a' && c <= 'z' || c >= '0' && c <= '9' || c == '_') {
			return "controller_failed"
		}
	}
	return s
}
func savedOutputBytes(dir string) (int64, error) {
	// Fixed owned filenames only; unexpected regular files or links fail closed.
	allowed := [4]string{"reservation.json", "journal.jsonl", "stderr.txt", "outside.json"}
	caps := [4]int64{16384, journalCap, stderrCap, 16384}
	var total int64
	entries, e := os.ReadDir(dir)
	if e != nil {
		return 0, errCode("output_census_failed")
	}
	for _, entry := range entries {
		if entry.Name() == "worker-attempt" {
			continue
		}
		found := -1
		for i, name := range allowed {
			if entry.Name() == name {
				found = i
			}
		}
		if found < 0 {
			return 0, errCode("unexpected_output_entry")
		}
		n, e := ownedOutputSize(filepath.Join(dir, entry.Name()), caps[found])
		if e != nil {
			return 0, e
		}
		total += n
	}
	child := filepath.Join(dir, "worker-attempt")
	childNames := [3]string{"reservation.json", "results.json", "failure.json"}
	childCaps := [3]int64{16384, 65536, 16384}
	st, e := os.Lstat(child)
	if os.IsNotExist(e) {
		return total, nil
	}
	if e != nil || !st.IsDir() || st.Mode()&os.ModeSymlink != 0 || st.Mode().Perm() != 0o700 {
		return 0, errCode("child_output_not_directory")
	}
	entries, e = os.ReadDir(child)
	if e != nil {
		return 0, errCode("child_output_census_failed")
	}
	for _, entry := range entries {
		found := -1
		for i, name := range childNames {
			if entry.Name() == name {
				found = i
			}
		}
		if found < 0 {
			return 0, errCode("unexpected_child_output")
		}
		n, e := ownedOutputSize(filepath.Join(child, entry.Name()), childCaps[found])
		if e != nil {
			return 0, e
		}
		total += n
	}
	return total, nil
}

func collect(c Config, configRaw []byte) (OutsideResult, error) {
	r := OutsideResult{Schema: "riido-three-native-outside-result-v1", State: "not_started", ConfigSHA256: digest(configRaw), PlanSHA256: c.Plan.SHA256, WorkerSHA256: c.Worker.Pin.SHA256, ControllerSHA256: c.Controller.Pin.SHA256, AggregateCap: aggregateCap, NoAutomaticRetry: true, RemainingCallsUnknown: true}
	if e := validateConfig(c); e != nil {
		return r, e
	}
	p, e := loadCountPlan(c)
	if e != nil {
		return r, e
	}
	initial, e := loadInitialSnapshots(p)
	if e != nil {
		return r, e
	}
	if e = requireAbsentOutput(c.Output); e != nil {
		return r, e
	}
	if e = os.Mkdir(c.Output, 0o700); e != nil {
		return r, errCode("exclusive_output_directory_failed")
	}
	if e = syncDir(filepath.Dir(c.Output)); e != nil {
		return r, e
	}
	reserve := struct {
		Schema     string `json:"schema"`
		Config     string `json:"config_sha256"`
		Plan       string `json:"plan_sha256"`
		Worker     string `json:"worker_sha256"`
		Controller string `json:"controller_sha256"`
		Inputs     int    `json:"inputs"`
		Dispatches int    `json:"candidate_dispatches"`
		Methods    int    `json:"tracked_method_maximum"`
		Frames     int    `json:"wire_frame_maximum"`
		Aggregate  int    `json:"aggregate_cap_bytes"`
		Retry      bool   `json:"automatic_retry"`
	}{"riido-three-native-outside-reservation-v1", digest(configRaw), c.Plan.SHA256, c.Worker.Pin.SHA256, c.Controller.Pin.SHA256, 14, 42, 795, wireFrameBound, aggregateCap, false}
	b, e := json.Marshal(reserve)
	if e != nil {
		return r, errCode("reservation_encode")
	}
	b = append(b, '\n')
	if e = durableFile(c.Output, "reservation.json", b, 16384); e != nil {
		return r, e
	}
	j, e := openJournal(c.Output, wireFrameBound)
	if e != nil {
		return r, e
	}
	state := ProtocolState{FrozenInitial: initial, FrozenInitialKnown: true}
	attempt := filepath.Join(c.Output, "worker-attempt")
	if e = requireAbsentOutput(attempt); e != nil {
		j.Close()
		return r, e
	}
	args := []string{"--plan", c.Plan.Path, "--plan-sha256", c.Plan.SHA256, "--attempt-dir", attempt}
	process, stderr, runErr := runChild(c, args, func(reader io.Reader, writer io.Writer) error {
		return consumeProtocol(reader, writer, j, &state, c.Plan.SHA256, func() error { return checkChildReservation(attempt, p, c.Plan.SHA256) })
	})
	r.Process = process
	r.DurablePrefix = j.Receipt
	r.FramesACKWritten = state.FramesACKWritten
	r.FinalACKWritten = state.Final != nil && state.FramesACKWritten == j.Receipt.FramesSynced
	r.LastDurableFrameCounts = state.LastDurableCounts
	r.TerminalRowsACKWritten = int(state.NextDispatch)
	if state.FramesACKWritten > 0 {
		counts := state.Counts
		r.LastACKWrittenCounts = &counts
	}
	closeErr := j.Close()
	if e = durableFile(c.Output, "stderr.txt", stderr, stderrCap); e != nil && runErr == nil {
		runErr = e
	}
	if closeErr != nil && runErr == nil {
		runErr = closeErr
	}
	if runErr == nil {
		if e = finalPlanBinding(state.Final, p); e != nil {
			runErr = e
		}
	}
	if runErr == nil {
		if e = exactFinalResult(filepath.Join(attempt, "results.json"), state); e != nil {
			runErr = e
		} else {
			r.FinalChildFileIdentity = true
		}
	}
	// Re-pin all inputs after the joined child, including Root evidence. No
	// changed source/control/binary is silently treated as the frozen trial.
	if e = validateConfig(c); e != nil && runErr == nil {
		runErr = e
	}
	if _, e = loadCountPlan(c); e != nil && runErr == nil {
		runErr = e
	}
	if runErr == nil && r.FinalACKWritten {
		r.State = "complete_observation_transport"
		r.CompleteCountersKnown = true
		r.RemainingCallsUnknown = false
		n := 42
		r.CompleteRows = &n
	} else {
		r.State = "incomplete_observation_transport"
	}
	r.Failure = fixedFailure(runErr)
	current, e := savedOutputBytes(c.Output)
	if e != nil {
		r.State = "incomplete_observation_transport"
		r.CompleteCountersKnown = false
		r.CompleteRows = nil
		r.RemainingCallsUnknown = true
		if runErr == nil {
			runErr = e
		}
		r.Failure = fixedFailure(runErr)
		return r, runErr
	}
	// Outside result is <=16KiB; reserves are actually checked before the write.
	retained := current
	r.OutputBytesRetained = &retained
	for i := 0; i < 4; i++ {
		b, e = json.Marshal(r)
		if e != nil {
			failure := errCode("outside_encode")
			return downgradeOutside(r, failure), failure
		}
		b = append(b, '\n')
		n := current + int64(len(b))
		if n == retained {
			break
		}
		retained = n
	}
	if len(b) > 16384 || current+int64(len(b)) > aggregateCap {
		r.State = "incomplete_observation_transport"
		r.CompleteCountersKnown = false
		r.CompleteRows = nil
		r.RemainingCallsUnknown = true
		r.OutputBytesRetained = nil
		r.Failure = "aggregate_saved_output_cap"
		return r, errCode("aggregate_saved_output_cap")
	}
	var saveErr error
	r, saveErr = persistOutsideResult(c.Output, b, r, durableFile)
	if saveErr != nil {
		return r, saveErr
	}
	return r, runErr
}

// Failure of the final outer file invalidates the outer completion claim.
// Already-durable frames, written ACKs and joined-process evidence survive.
// No worker or original function is retried.
func downgradeOutside(r OutsideResult, e error) OutsideResult {
	r.State = "incomplete_observation_transport"
	r.Failure = fixedFailure(e)
	r.CompleteCountersKnown = false
	r.CompleteRows = nil
	r.RemainingCallsUnknown = true
	r.OutputBytesRetained = nil
	return r
}
func persistOutsideResult(dir string, b []byte, r OutsideResult, persist func(string, string, []byte, int) error) (OutsideResult, error) {
	if persist == nil {
		e := errCode("outside_persistence_missing")
		return downgradeOutside(r, e), e
	}
	if e := persist(dir, "outside.json", b, 16384); e != nil {
		return downgradeOutside(r, e), e
	}
	return r, nil
}

func run(args []string) int {
	if len(args) != 2 || args[0] != "--config" {
		fmt.Fprintln(os.Stderr, "usage_config_only")
		return 2
	}
	runtime.GOMAXPROCS(1)
	debug.SetMemoryLimit(256 << 20)
	raw, e := readRegular(args[1], metadataCap)
	if e != nil {
		fmt.Fprintln(os.Stderr, "config_read_failed")
		return 1
	}
	var c Config
	if strictCanonical(raw, &c) != nil {
		fmt.Fprintln(os.Stderr, "config_decode_failed")
		return 1
	}
	r, e := collect(c, raw)
	b, enc := json.Marshal(r)
	if enc != nil {
		fmt.Fprintln(os.Stderr, "outside_summary_encode_failed")
		return 1
	}
	b = append(b, '\n')
	if len(b) > 16384 {
		fmt.Fprintln(os.Stderr, "outside_summary_cap")
		return 1
	}
	if n, w := os.Stdout.Write(b); w != nil || n != len(b) {
		return 1
	}
	if e != nil {
		return 1
	}
	return 0
}
func main() { os.Exit(run(os.Args[1:])) }
