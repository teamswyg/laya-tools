type Error string

func (e Error) Error() string { return string(e) }
func code(s string) error     { return Error(s) }
func FailureCode(e error) string {
	if e == nil {
		return ""
	}
	if v, ok := e.(Error); ok {
		return string(v)
	}
	return "data33_callback_failed"
}
