// Source-only proposed CLI; future original initialization precedes main.
package main

import (
	"os"
	"riido.local/next60gjson-two-nativeprep/native"
	"riido.local/next60gjson-two-nativeprep/twoliteral"
	"riido.local/next60gjson-two-nativeprep/workerio"
)

func options(a []string) (plan, sha, attempt string, ok bool) {
	if len(a) != 6 {
		return
	}
	seen := [3]bool{}
	for i := 0; i < 6; i += 2 {
		k := -1
		switch a[i] {
		case "--plan":
			k = 0
			plan = a[i+1]
		case "--plan-sha256":
			k = 1
			sha = a[i+1]
		case "--attempt-dir":
			k = 2
			attempt = a[i+1]
		}
		if k < 0 || seen[k] || a[i+1] == "" {
			return "", "", "", false
		}
		seen[k] = true
	}
	return plan, sha, attempt, seen[0] && seen[1] && seen[2]
}
func run() int {
	planPath, sha, attempt, ok := options(os.Args[1:])
	if !ok {
		return 2
	}
	p, fixtures, wants, e := workerio.LoadPlan(planPath, sha)
	if e != nil {
		return 2
	}
	files, e := workerio.Open(attempt, sha, p)
	if e != nil {
		return 2
	}
	transport := workerio.NewTransport(os.Stdin, os.Stdout)
	result := twoliteral.Observe(native.Operations(), fixtures, wants, transport.Exchange, sha)
	if result.State != "complete" {
		files.SaveFailure(result, "observer_stopped", transport)
		return 1
	}
	// Child canonical file is durable before final binding; a failed final ACK cannot certify completion.
	if files.SaveResult(result) != nil {
		files.SaveFailure(result, "result_persistence_failed", transport)
		return 1
	}
	if transport.Finish(result) != nil {
		files.SaveFailure(result, "final_ack_failed", transport)
		return 1
	}
	return 0
}
func main() { os.Exit(run()) }
