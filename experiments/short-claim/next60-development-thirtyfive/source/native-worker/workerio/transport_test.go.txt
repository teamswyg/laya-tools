type dynamicACK struct {
	line []byte
	bad  bool
}

func (a *dynamicACK) Write(b []byte) (int, error) {
	a.line = append([]byte(nil), b...)
	return len(b), nil
}
func (a *dynamicACK) Read(b []byte) (int, error) {
	if len(a.line) == 0 {
		return 0, io.EOF
	}
	var f twoliteral.Frame
	json.Unmarshal(a.line, &f)
	h := SHA(a.line)
	if a.bad {
		h = twoliteral.ZeroSHA
	}
	v, _ := json.Marshal(twoliteral.Ack{Version: 2, Sequence: f.Sequence, SHA256: h})
	v = append(v, '\n')
	a.line = nil
	return copy(b, v), nil
}
func TestMissingOrWrongACKStops(t *testing.T) {
	for _, missing := range []bool{true, false} {
		a := &dynamicACK{bad: true}
		var input io.Reader = a
		if missing {
			input = bytes.NewReader(nil)
		}
		tr := NewTransport(input, a)
		f := twoliteral.Frame{Version: 2, Sequence: 1, Previous: twoliteral.ZeroSHA, Stage: twoliteral.Ready}
		if tr.Exchange(f) == nil || tr.Exchange(f) == nil {
			t.Fatal("ACK failure reusable")
		}
		s, _ := tr.Prefix()
		if s != 0 {
			t.Fatal("unacknowledged line promoted")
		}
	}
}
func TestFinalCannotForgeCompletedRows(t *testing.T) {
	tr := NewTransport(bytes.NewReader(nil), io.Discard)
	r := twoliteral.Result{Schema: twoliteral.ResultSchema, State: "complete", Completed: 33, LastACKSHA256: twoliteral.ZeroSHA, Counts: twoliteral.Counts{Row: [4]int{33, 33, 33, 0}, Candidate: [4]int{33, 33, 33, 0}}}
	if tr.Finish(r) == nil {
		t.Fatal("no row ACKs accepted")
	}
}
func TestCanonicalACKRejectsCaseAliasAndDuplicate(t *testing.T) {
	for _, s := range []string{`{"V":2,"s":1,"h":"x"}`, `{"v":2,"v":2,"s":1,"h":"x"}`} {
		var a twoliteral.Ack
		if DecodeCanonical([]byte(s+"\n"), &a) == nil {
			t.Fatal("noncanonical ACK")
		}
	}
}

func TestExclusivePersistencePreservesExisting(t *testing.T) {
	dir := t.TempDir()
	p := filepath.Join(dir, "reservation.json")
	if writeExclusive(p, []byte("first"), 16) != nil {
		t.Fatal("initial write")
	}
	if writeExclusive(p, []byte("second"), 16) == nil {
		t.Fatal("overwrite")
	}
	b, e := os.ReadFile(p)
	if e != nil || string(b) != "first" {
		t.Fatal("old bytes changed")
	}
}
