package observation

import (
	"encoding/json"
	"strings"
	"testing"
)

func iwants() Wants {return Wants{VersionInputs:[3]string{"plain","prefix","short"},StrictAccepted:[3]bool{true,false,false},NewAccepted:[3]bool{true,true,true},NewString:[3]string{"ok","ok","ok"},GlobNames:[2]string{"flat","nested"},GlobPatterns:[3]string{"single","nested","middle"},GlobMatched:[3][2]bool{{true,false},{true,true},{true,false}},PlannedEntries:12,PlannedStrings:3}}
func icallbacks() Callbacks {return Callbacks{Flags:func()Flags{return Flags{true,true}},Strict:func(s string)Parsed{return Parsed{VersionPresent:s=="plain",ErrorPresent:s!="plain"}},New:func(string)Parsed{return Parsed{VersionPresent:true,String:func()string{return "ok"}}},Match:func(p,n string)(bool,bool){return n=="flat"||p=="nested",false}}}
func TestIndependentNilObjectAndObjectWithErrorPreserveAcceptanceDefinition(t *testing.T){w:=iwants();c:=icallbacks();c.Strict=func(s string)Parsed{if s=="prefix"{return Parsed{}};if s=="short"{return Parsed{VersionPresent:true,ErrorPresent:true}};return Parsed{VersionPresent:true}}
	r:=Run(w,Flags{true,true},c)
	for _,i:=range []int{1,2}{v:=r.Versions[i];if !v.Returned||v.Panicked||!v.GotDefined||v.Got||!v.Compared||!v.Matches{t.Fatal("false acceptance presence bits lost")}}
	if r.Versions[1].ErrorPresent||r.Versions[1].VersionPresent||!r.Versions[2].ErrorPresent||!r.Versions[2].VersionPresent{t.Fatal("nil vs object+error collapsed")}
	if r.NewObservedSupervision||r.TrainingReady||r.Compared!=15||r.Uncompared!=0{t.Fatal("finite acceptance comparison was promoted")}
}
func TestIndependentPanicNilIsUndefinedAndNotSuccessfulReject(t *testing.T){w:=iwants();c:=icallbacks();c.Strict=func(s string)Parsed{if s=="prefix"{panic(nil)};return Parsed{VersionPresent:s=="plain",ErrorPresent:s!="plain"}}
	r:=Run(w,Flags{true,true},c);v:=r.Versions[1]
	if !v.Attempted||v.Returned||!v.Panicked||v.GotDefined||v.Compared||v.Matches||r.Entries[0].Attempts!=3||r.Entries[0].Returned!=2||r.Entries[0].Panicked!=1||r.Uncompared!=1||r.Compared+r.Uncompared!=15{t.Fatal("nil panic counter/undefined state")}
}
func TestIndependentFinalFlagPanicKeepsCompletedRows(t *testing.T){w:=iwants();c:=icallbacks();reads:=0;c.Flags=func()Flags{reads++;if reads==2{panic("private-independent-flag")};return Flags{true,true}}
	r:=Run(w,Flags{true,true},c)
	if r.State!="completed_observation_failure"||r.FlagsExpectedAndUnchanged||r.FlagReadAttempts!=2||r.FlagReadReturns!=1||r.FlagReadPanics!=1||r.Compared!=15||r.Matches!=15||r.Strings.Attempts!=3||r.Entries[2].Attempts!=6{t.Fatal("final flag failure overwrote prior rows")}
	b,e:=json.Marshal(r);if e!=nil||strings.Contains(string(b),"private-independent"){t.Fatal("private panic text retained")}
}
