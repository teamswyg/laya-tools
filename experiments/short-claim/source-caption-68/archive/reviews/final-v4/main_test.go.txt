package main

import "testing"

func TestWholeRawSpanRejectsBoundaryHashAndQuoteChanges(t *testing.T) {
	b:=[]byte("header\n  - Optionally work with a `v` prefix\nfooter\n")
	text:="  - Optionally work with a `v` prefix\n"
	s:=Span{7,45,2,2,hash([]byte(text))}
	if !validSpan(b,s,text) { t.Fatal("complete bullet rejected") }
	bad:=s;bad.End--;if validSpan(b,bad,text) { t.Fatal("truncated LF accepted") }
	bad=s;bad.SHA=hash([]byte("different"));if validSpan(b,bad,text) { t.Fatal("incorrect hash accepted") }
	if validSpan(b,s,"- Optionally work with a v prefix") { t.Fatal("rewritten quote accepted") }
	bad=s;bad.First=1;if validSpan(b,bad,text) { t.Fatal("incorrect line accepted") }
}
func TestWordBudgetKeepsWholeBulletAndCamelBoundaries(t *testing.T) {
	if words("  - Optionally work with a `v` prefix\n")!=6 { t.Fatal("whole bullet words") }
	if words("StrictNewVersion HTTPServer v1.2.3")!=8 { t.Fatal("normalization boundary drift") }
}
