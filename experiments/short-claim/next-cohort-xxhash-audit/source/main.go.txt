package main

import (
	"bytes"
	"crypto/sha256"
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"os"
	"unicode/utf8"

	"github.com/cespare/xxhash/v2"
)

const (
	inputLimit         = 32 * 1024
	outputLimit        = 96 * 1024
	wireSize           = 76
	prime1      uint64 = 11400714785074694791
	prime2      uint64 = 14029467366897019727
)

var implementationIDs = [5]string{"direct", "reconstruct", "legacy", "roundtrip", "digest"}

type Fixture struct {
	ID          string `json:"id"`
	InputHex    string `json:"input_hex"`
	WantSuccess *bool  `json:"want_success"`
	Reason      string `json:"reason"`
}

type Candidate struct {
	ID      string `json:"id"`
	Caption string `json:"caption"`
}

type Frozen struct {
	Schema         string      `json:"schema"`
	OldReceiverHex string      `json:"old_receiver_hex"`
	Fixtures       []Fixture   `json:"fixtures"`
	Candidates     []Candidate `json:"candidates"`
}

type decodedFixture struct {
	Fixture
	Input []byte
}

func badInput(reason string) (Frozen, []decodedFixture, []byte, []byte, error) {
	return Frozen{}, nil, nil, nil, errors.New(reason)
}

type Event struct {
	Phase        string `json:"p"`
	Method       string `json:"m"`
	Attempted    bool   `json:"a"`
	Returned     bool   `json:"r"`
	Panicked     bool   `json:"x"`
	ErrorPresent *bool  `json:"e,omitempty"`
	ErrorText    string `json:"err,omitempty"`
	PanicText    string `json:"panic,omitempty"`
	NilResult    *bool  `json:"nil,omitempty"`
	WriteN       *int   `json:"n,omitempty"`
	BytesLen     *int   `json:"len,omitempty"`
	BytesSHA     string `json:"sha,omitempty"`
	SumHex       string `json:"sum,omitempty"`
}

type observer struct {
	phase  string
	events []Event
}

type apiAbort struct{ Phase, Method, Message string }

func boolValue(v bool) *bool       { return &v }
func intValue(v int) *int          { return &v }
func stringValue(v string) *string { return &v }
func digestHex(v uint64) string    { return fmt.Sprintf("%016x", v) }
func shaHex(v []byte) string {
	s := sha256.Sum256(v)
	return hex.EncodeToString(s[:])
}

func (o *observer) start(method string) int {
	o.events = append(o.events, Event{Phase: o.phase, Method: method, Attempted: true})
	return len(o.events) - 1
}

// Direct defer; r=false detects even panic(nil).
func (o *observer) finish(i int) {
	if o.events[i].Returned {
		return
	}
	v := recover()
	e := &o.events[i]
	e.Panicked = true
	e.PanicText = fmt.Sprint(v)
	panic(apiAbort{e.Phase, e.Method, e.PanicText})
}

func (o *observer) newDigest(seed uint64) (d *xxhash.Digest) {
	i := o.start("new")
	defer o.finish(i)
	d = xxhash.NewWithSeed(seed)
	o.events[i].Returned = true
	o.events[i].NilResult = boolValue(d == nil)
	return
}

func (o *observer) write(d *xxhash.Digest, b []byte) (n int, err error) {
	i := o.start("write")
	defer o.finish(i)
	n, err = d.Write(b)
	e := &o.events[i]
	e.Returned = true
	e.WriteN = intValue(n)
	e.ErrorPresent = boolValue(err != nil)
	if err != nil {
		e.ErrorText = err.Error()
	}
	return
}

func (o *observer) unmarshal(d *xxhash.Digest, b []byte) (err error) {
	i := o.start("unmarshal")
	defer o.finish(i)
	err = d.UnmarshalBinary(b)
	e := &o.events[i]
	e.Returned = true
	e.ErrorPresent = boolValue(err != nil)
	if err != nil {
		e.ErrorText = err.Error()
	}
	return
}

func (o *observer) marshal(d *xxhash.Digest) (b []byte, err error) {
	i := o.start("marshal")
	defer o.finish(i)
	b, err = d.MarshalBinary()
	e := &o.events[i]
	e.Returned = true
	e.ErrorPresent = boolValue(err != nil)
	e.NilResult = boolValue(b == nil)
	e.BytesLen = intValue(len(b))
	if o.phase == "candidate" {
		e.BytesSHA = shaHex(b)
	}
	if err != nil {
		e.ErrorText = err.Error()
	}
	return
}

func (o *observer) sum(d *xxhash.Digest) (v uint64) {
	i := o.start("sum")
	defer o.finish(i)
	v = d.Sum64()
	o.events[i].Returned = true
	o.events[i].SumHex = digestHex(v)
	return
}

// Each mechanism validates independently; the observer imposes no predicates.
func direct(o *observer, dst *xxhash.Digest, b []byte) error {
	if len(b) != wireSize {
		return errors.New("direct: size")
	}
	if string(b[:4]) != "xxh\x06" {
		return errors.New("direct: magic")
	}
	total := binary.LittleEndian.Uint64(b[36:44])
	if total > 32 {
		return errors.New("direct: total")
	}
	n := int(total % 32)
	for _, v := range b[44+n:] {
		if v != 0 {
			return errors.New("direct: unused tail")
		}
	}
	if total < 32 {
		seed := binary.LittleEndian.Uint64(b[20:28])
		v1 := binary.LittleEndian.Uint64(b[4:12])
		v2 := binary.LittleEndian.Uint64(b[12:20])
		v4 := binary.LittleEndian.Uint64(b[28:36])
		if v1 != seed+prime1+prime2 || v2 != seed+prime2 || v4 != seed-prime1 {
			return errors.New("direct: short lanes")
		}
	}
	var imported xxhash.Digest
	if err := o.unmarshal(&imported, b); err != nil {
		return err
	}
	*dst = imported
	return nil
}

func reconstruct(o *observer, dst *xxhash.Digest, b []byte) error {
	if len(b) != wireSize {
		return errors.New("reconstruct: size")
	}
	if string(b[:4]) != "xxh\x06" {
		return errors.New("reconstruct: magic")
	}
	total := binary.LittleEndian.Uint64(b[36:44])
	if total > 32 {
		return errors.New("reconstruct: total")
	}
	var imported xxhash.Digest
	if err := o.unmarshal(&imported, b); err != nil {
		return err
	}
	var state []byte
	var err error
	if total < 32 {
		seed := binary.LittleEndian.Uint64(b[20:28])
		rebuilt := o.newDigest(seed)
		if rebuilt == nil {
			return errors.New("reconstruct: nil new")
		}
		n, writeErr := o.write(rebuilt, b[44:44+int(total)])
		if writeErr != nil {
			return writeErr
		}
		if n != int(total) {
			return errors.New("reconstruct: short write")
		}
		state, err = o.marshal(rebuilt)
	} else {
		// Boundary32 uses canonical structure, not seed recovery.
		state, err = o.marshal(&imported)
	}
	if err != nil {
		return err
	}
	if !bytes.Equal(state, b) {
		return errors.New("reconstruct: full state")
	}
	*dst = imported
	return nil
}

func legacy(o *observer, dst *xxhash.Digest, b []byte) error {
	var imported xxhash.Digest
	if err := o.unmarshal(&imported, b); err != nil {
		return err
	}
	*dst = imported
	return nil
}

func roundtrip(o *observer, dst *xxhash.Digest, b []byte) error {
	var imported xxhash.Digest
	if err := o.unmarshal(&imported, b); err != nil {
		return err
	}
	state, err := o.marshal(&imported)
	if err != nil {
		return err
	}
	if !bytes.Equal(state, b) {
		return errors.New("roundtrip: full state")
	}
	*dst = imported
	return nil
}

func digest(o *observer, dst *xxhash.Digest, b []byte) error {
	var imported xxhash.Digest
	if err := o.unmarshal(&imported, b); err != nil {
		return err
	}
	total := binary.LittleEndian.Uint64(b[36:44])
	if total < 32 {
		seed := binary.LittleEndian.Uint64(b[20:28])
		rebuilt := o.newDigest(seed)
		if rebuilt == nil {
			return errors.New("digest: nil new")
		}
		n, err := o.write(rebuilt, b[44:44+int(total)])
		if err != nil {
			return err
		}
		if n != int(total) {
			return errors.New("digest: short write")
		}
		importedSum := o.sum(&imported)
		rebuiltSum := o.sum(rebuilt)
		if importedSum != rebuiltSum {
			return errors.New("digest: mismatch")
		}
	}
	// Declared legacy fallback at total>=32.
	*dst = imported
	return nil
}

func candidateCall(id string, o *observer, d *xxhash.Digest, b []byte) error {
	switch id {
	case "direct":
		return direct(o, d, b)
	case "reconstruct":
		return reconstruct(o, d, b)
	case "legacy":
		return legacy(o, d, b)
	case "roundtrip":
		return roundtrip(o, d, b)
	case "digest":
		return digest(o, d, b)
	default:
		return errors.New("unknown implementation")
	}
}

type Trial struct {
	FixtureID          string  `json:"f"`
	CandidateID        string  `json:"c"`
	WantSuccess        bool    `json:"want"`
	InputSHA           string  `json:"input_sha"`
	SetupOK            bool    `json:"setup_ok"`
	BeforeHex          *string `json:"b,omitempty"`
	BeforeSum          *string `json:"bs,omitempty"`
	OldMatches         *bool   `json:"old,omitempty"`
	CandidateAttempted bool    `json:"ca"`
	CandidateReturned  bool    `json:"cr"`
	CandidateError     *bool   `json:"ce,omitempty"`
	CandidateErrorText string  `json:"err,omitempty"`
	CandidatePanic     bool    `json:"cx"`
	CandidatePanicText string  `json:"panic,omitempty"`
	AfterHex           *string `json:"a,omitempty"`
	AfterSum           *string `json:"as,omitempty"`
	InputEqual         bool    `json:"input_equal"`
	BytesEqual         *bool   `json:"bytes_equal,omitempty"`
	SumEqual           *bool   `json:"sum_equal,omitempty"`
	AfterInputEqual    *bool   `json:"after_input_equal,omitempty"`
	Failure            string  `json:"failure,omitempty"`
	Calls              []Event `json:"calls"`
}

func runTrial(f decodedFixture, c Candidate, old []byte) (t Trial) {
	input := append([]byte(nil), f.Input...)
	original := append([]byte(nil), input...)
	o := observer{phase: "setup", events: make([]Event, 0, 12)}
	var beforeBytes, afterBytes []byte
	var beforeMarshalOK, afterMarshalOK bool
	finished := false
	t.FixtureID, t.CandidateID = f.ID, c.ID
	t.WantSuccess = *f.WantSuccess
	t.InputSHA = shaHex(original)
	defer func() {
		if !finished {
			v := recover()
			t.Failure = o.phase + ": panic"
			message := fmt.Sprint(v)
			if a, ok := v.(apiAbort); ok {
				message = a.Message
			}
			if o.phase == "candidate" && t.CandidateAttempted && !t.CandidateReturned {
				t.CandidatePanic = true
				t.CandidatePanicText = message
			}
		}
		t.InputEqual = bytes.Equal(original, input)
		if beforeMarshalOK && afterMarshalOK {
			t.BytesEqual = boolValue(bytes.Equal(beforeBytes, afterBytes))
		}
		if afterMarshalOK {
			t.AfterInputEqual = boolValue(bytes.Equal(original, afterBytes))
		}
		if t.BeforeSum != nil && t.AfterSum != nil {
			t.SumEqual = boolValue(*t.BeforeSum == *t.AfterSum)
		}
		t.Calls = o.events
	}()
	d := o.newDigest(19)
	if d == nil {
		t.Failure = "setup: nil new"
		finished = true
		return
	}
	n, err := o.write(d, []byte{0x6f, 0x6c, 0x64})
	if err != nil || n != 3 {
		t.Failure = "setup: write"
		finished = true
		return
	}
	o.phase = "before"
	beforeBytes, err = o.marshal(d)
	t.BeforeHex = stringValue(hex.EncodeToString(beforeBytes))
	if err != nil {
		t.Failure = "before: marshal error"
		finished = true
		return
	}
	beforeMarshalOK = true
	t.OldMatches = boolValue(bytes.Equal(beforeBytes, old))
	if !*t.OldMatches {
		t.Failure = "before: frozen old mismatch"
		finished = true
		return
	}
	t.BeforeSum = stringValue(digestHex(o.sum(d)))
	t.SetupOK = true
	o.phase = "candidate"
	t.CandidateAttempted = true
	err = candidateCall(c.ID, &o, d, input)
	t.CandidateReturned = true
	t.CandidateError = boolValue(err != nil)
	if err != nil {
		t.CandidateErrorText = err.Error()
	}
	// Normal error returns receive after probes; panics abort the trial.
	o.phase = "after"
	afterBytes, err = o.marshal(d)
	t.AfterHex = stringValue(hex.EncodeToString(afterBytes))
	afterMarshalOK = err == nil
	if err != nil {
		t.Failure = "after: marshal error"
	}
	t.AfterSum = stringValue(digestHex(o.sum(d)))
	finished = true
	return
}

type FixtureView struct {
	ID          string `json:"id"`
	InputSHA    string `json:"input_sha256"`
	WantSuccess bool   `json:"want_success"`
	Reason      string `json:"reason"`
}

type Counts struct {
	FixtureEntries     int `json:"fixture_entries"`
	CandidateEntries   int `json:"candidate_entries"`
	ExpectedTrials     int `json:"expected_trials"`
	TrialEntries       int `json:"trial_entries"`
	SetupCompleted     int `json:"setup_completed"`
	CandidateAttempted int `json:"candidate_attempted"`
	CandidateReturned  int `json:"candidate_returned"`
	CandidateErrors    int `json:"candidate_errors"`
	CandidatePanics    int `json:"candidate_panics"`
	TrialFailures      int `json:"trial_failures"`
	CallAttempted      int `json:"call_attempted"`
	CallReturned       int `json:"call_returned"`
	CallErrors         int `json:"call_errors"`
	CallPanics         int `json:"call_panics"`
}

type Report struct {
	Schema                 string        `json:"schema"`
	Status                 string        `json:"status"`
	FrozenSHA              string        `json:"frozen_sha256"`
	OldReceiverHex         string        `json:"old_receiver_hex"`
	LedgerSchema           string        `json:"ledger_schema"`
	QualificationPerformed bool          `json:"qualification_performed"`
	FitCalls               int           `json:"fit_calls"`
	ModelCalls             int           `json:"model_calls"`
	Fixtures               []FixtureView `json:"fixtures"`
	Candidates             []Candidate   `json:"candidates"`
	Counts                 Counts        `json:"counts"`
	Trials                 []Trial       `json:"trials"`
}

func summarize(r *Report) {
	r.Counts.FixtureEntries = len(r.Fixtures)
	r.Counts.CandidateEntries = len(r.Candidates)
	r.Counts.ExpectedTrials = len(r.Fixtures) * len(r.Candidates)
	r.Counts.TrialEntries = len(r.Trials)
	for _, t := range r.Trials {
		if t.SetupOK {
			r.Counts.SetupCompleted++
		}
		if t.CandidateAttempted {
			r.Counts.CandidateAttempted++
		}
		if t.CandidateReturned {
			r.Counts.CandidateReturned++
		}
		if t.CandidateError != nil && *t.CandidateError {
			r.Counts.CandidateErrors++
		}
		if t.CandidatePanic {
			r.Counts.CandidatePanics++
		}
		if t.Failure != "" {
			r.Counts.TrialFailures++
		}
		for _, e := range t.Calls {
			if e.Attempted {
				r.Counts.CallAttempted++
			}
			if e.Returned {
				r.Counts.CallReturned++
			}
			if e.ErrorPresent != nil && *e.ErrorPresent {
				r.Counts.CallErrors++
			}
			if e.Panicked {
				r.Counts.CallPanics++
			}
		}
	}
}

func canonicalHex(s string) ([]byte, error) {
	b, err := hex.DecodeString(s)
	if err != nil || hex.EncodeToString(b) != s {
		return nil, errors.New("invalid literal hex")
	}
	return b, nil
}

func readFrozen(path string) (Frozen, []decodedFixture, []byte, []byte, error) {
	var f Frozen
	file, err := os.Open(path)
	if err != nil {
		return badInput("frozen input open failed")
	}
	raw, readErr := io.ReadAll(io.LimitReader(file, inputLimit+1))
	closeErr := file.Close()
	if readErr != nil || closeErr != nil {
		return badInput("frozen input read failed")
	}
	if len(raw) > inputLimit || !utf8.Valid(raw) {
		return badInput("frozen input bound or UTF8")
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.DisallowUnknownFields()
	if d.Decode(&f) != nil {
		return badInput("frozen input JSON")
	}
	var extra any
	if d.Decode(&extra) != io.EOF {
		return badInput("frozen input trailing JSON")
	}
	if f.Schema != "riido-xxhash-short-state-frozen-v1" || len(f.Fixtures) != 15 || len(f.Candidates) != 5 {
		return badInput("frozen input schema or counts")
	}
	old, err := canonicalHex(f.OldReceiverHex)
	if err != nil || len(old) != wireSize {
		return badInput("frozen old literal")
	}
	decoded := make([]decodedFixture, len(f.Fixtures))
	for i, x := range f.Fixtures {
		if x.ID == "" || x.Reason == "" || x.WantSuccess == nil {
			return badInput("frozen fixture missing field")
		}
		for j := 0; j < i; j++ {
			if f.Fixtures[j].ID == x.ID {
				return badInput("frozen fixture duplicate ID")
			}
		}
		b, err := canonicalHex(x.InputHex)
		if err != nil {
			return badInput("frozen fixture literal")
		}
		decoded[i] = decodedFixture{Fixture: x, Input: b}
	}
	for i, c := range f.Candidates {
		if c.ID != implementationIDs[i] || c.Caption == "" {
			return badInput("frozen candidate order or caption")
		}
	}
	return f, decoded, old, raw, nil
}

func run(path string) error {
	f, fixtures, old, raw, err := readFrozen(path)
	if err != nil {
		return err
	}
	r := Report{
		Schema:    "riido-xxhash-actual-development-audit-v1",
		Status:    "actual_development_audit_unqualified",
		FrozenSHA: shaHex(raw), OldReceiverHex: f.OldReceiverHex,
		LedgerSchema: "phase-public-api-call-v1",
		Candidates:   f.Candidates, Trials: make([]Trial, 0, 75),
	}
	for _, x := range fixtures {
		r.Fixtures = append(r.Fixtures, FixtureView{x.ID, shaHex(x.Input), *x.WantSuccess, x.Reason})
		for _, c := range f.Candidates {
			r.Trials = append(r.Trials, runTrial(x, c, old))
		}
	}
	summarize(&r)
	if r.Counts.TrialFailures != 0 {
		r.Status = "actual_development_audit_partial_unqualified"
	}
	out, err := json.Marshal(r)
	if err != nil {
		return errors.New("report encoding failed")
	}
	if len(out)+1 > outputLimit {
		return errors.New("report exceeds 96KiB; no JSON emitted")
	}
	out = append(out, '\n')
	n, err := os.Stdout.Write(out)
	if err != nil || n != len(out) {
		return errors.New("report stdout write failed")
	}
	if r.Counts.TrialFailures != 0 {
		return errors.New("trial observation failure; JSON retained on stdout")
	}
	return nil
}

func main() {
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: xxhash-observer frozen.json")
		os.Exit(2)
	}
	if err := run(os.Args[1]); err != nil {
		fmt.Fprintln(os.Stderr, err.Error())
		os.Exit(1)
	}
}
