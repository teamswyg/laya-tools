package goldenreader

import (
	"bytes"
	"encoding/json"
	"io"
	"reflect"
	"strconv"
	"unicode/utf8"
)

// A bounded lexical pass precedes allocation by json.Unmarshal. Every decoded
// exact key is mandatory, duplicate/escaped duplicate/case alias is rejected.
// Candidate arity is exactly8; predicate slices are separately bounded at64.
func strict(raw []byte, out any) error {
	if !utf8.Valid(raw) || !json.Valid(raw) || !scalarEscapes(raw) {
		return ErrJSON
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.UseNumber()
	typ := reflect.TypeOf(out)
	if typ == nil || typ.Kind() != reflect.Pointer {
		return ErrJSON
	}
	if err := walk(d, typ.Elem(), 0); err != nil {
		return err
	}
	if _, err := d.Token(); err != io.EOF {
		return ErrJSON
	}
	if json.Unmarshal(raw, out) != nil {
		return ErrJSON
	}
	return nil
}
func walk(d *json.Decoder, typ reflect.Type, depth int) error {
	if depth > 12 {
		return ErrJSON
	}
	if typ.Kind() == reflect.Pointer {
		var raw json.RawMessage
		if d.Decode(&raw) != nil {
			return ErrJSON
		}
		if bytes.Equal(bytes.TrimSpace(raw), []byte("null")) {
			return nil
		}
		n := json.NewDecoder(bytes.NewReader(raw))
		n.UseNumber()
		if err := walk(n, typ.Elem(), depth+1); err != nil {
			return err
		}
		if _, err := n.Token(); err != io.EOF {
			return ErrJSON
		}
		return nil
	}
	token, err := d.Token()
	if err != nil {
		return ErrJSON
	}
	switch typ.Kind() {
	case reflect.Struct:
		if token != json.Delim('{') || typ.NumField() > 16 {
			return ErrJSON
		}
		var seen [16]bool
		for d.More() {
			key, err := d.Token()
			if err != nil {
				return ErrJSON
			}
			s, ok := key.(string)
			if !ok {
				return ErrJSON
			}
			at := -1
			for i := 0; i < typ.NumField(); i++ {
				if typ.Field(i).Tag.Get("json") == s {
					at = i
					break
				}
			}
			if at < 0 || seen[at] {
				return ErrJSON
			}
			seen[at] = true
			if err := walk(d, typ.Field(at).Type, depth+1); err != nil {
				return err
			}
		}
		if t, e := d.Token(); e != nil || t != json.Delim('}') {
			return ErrJSON
		}
		for i := 0; i < typ.NumField(); i++ {
			if !seen[i] {
				return ErrJSON
			}
		}
	case reflect.Slice:
		if token != json.Delim('[') {
			return ErrJSON
		}
		limit := MaxPredicates
		candidate := typ.Elem() == reflect.TypeOf(wireCandidate{})
		if candidate {
			limit = 8
		}
		n := 0
		for d.More() {
			if n == limit {
				return ErrBounds
			}
			n++
			if err := walk(d, typ.Elem(), depth+1); err != nil {
				return err
			}
		}
		if t, e := d.Token(); e != nil || t != json.Delim(']') {
			return ErrJSON
		}
		if candidate && n != 8 {
			return ErrShape
		}
	case reflect.String:
		if _, ok := token.(string); !ok {
			return ErrJSON
		}
	case reflect.Bool:
		if _, ok := token.(bool); !ok {
			return ErrJSON
		}
	case reflect.Int:
		n, ok := token.(json.Number)
		if !ok {
			return ErrJSON
		}
		i, e := strconv.ParseInt(n.String(), 10, 32)
		if e != nil || strconv.FormatInt(i, 10) != n.String() {
			return ErrJSON
		}
	default:
		return ErrJSON
	}
	return nil
}

// Derived from the owned v1 Reader (Apache-2.0). json.Valid precedes this scan.
func scalarEscapes(raw []byte) bool {
	for i := 0; i < len(raw); i++ {
		if raw[i] != '"' {
			continue
		}
		for i++; i < len(raw) && raw[i] != '"'; i++ {
			if raw[i] != '\\' {
				continue
			}
			i++
			if raw[i] != 'u' {
				continue
			}
			u := hex4(raw[i+1 : i+5])
			i += 4
			if u >= 0xdc00 && u <= 0xdfff {
				return false
			}
			if u >= 0xd800 && u <= 0xdbff {
				if i+6 >= len(raw) || raw[i+1] != '\\' || raw[i+2] != 'u' {
					return false
				}
				low := hex4(raw[i+3 : i+7])
				if low < 0xdc00 || low > 0xdfff {
					return false
				}
				i += 6
			}
		}
	}
	return true
}
func hex4(b []byte) uint16 {
	var v uint16
	for _, c := range b {
		v <<= 4
		switch {
		case c >= '0' && c <= '9':
			v |= uint16(c - '0')
		case c >= 'a' && c <= 'f':
			v |= uint16(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			v |= uint16(c - 'A' + 10)
		}
	}
	return v
}
