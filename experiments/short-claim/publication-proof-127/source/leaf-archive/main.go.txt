//go:build darwin || linux

package main

import (
	"encoding/json"
	"os"
)

func main() {
	if len(os.Args) != 5 || os.Args[1] != "--config" || os.Args[3] != "--config-sha256" {
		os.Stderr.WriteString("leaf_archive_args\n")
		os.Exit(2)
	}
	r, e := run(os.Args[2], os.Args[4])
	// Full records remain in private FINAL/FAILURE files. Captured stdout is
	// only a bounded binding summary and is charged within control allowance.
	b, me := json.Marshal(struct {
		Schema    string `json:"schema"`
		Complete  bool   `json:"complete"`
		Completed int    `json:"completed_leaves"`
		Reclaimed *int64 `json:"reclaimed_logical_bytes_lower_bound"`
		Final     *pin   `json:"final"`
		Failure   string `json:"failure"`
	}{r.Schema, r.Complete, r.CompletedLeaves, r.ReclaimedLogicalBytesLowerBound, r.Final, r.Failure})
	if me != nil || len(b)+1 > 2048 {
		os.Exit(1)
	}
	if _, we := os.Stdout.Write(append(b, '\n')); we != nil {
		os.Exit(1)
	}
	if e != nil {
		os.Exit(1)
	}
}
