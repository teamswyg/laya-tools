func fixed(s string)error{return errors.New(s)}
func validSHA(s string)bool{if len(s)!=64{return false};for _,c:=range s{if !(c>='0'&&c<='9')&&!(c>='a'&&c<='f'){return false}};return true}
func read(path string,cap int64)([]byte,error) {
	a,e:=os.Lstat(path);if e!=nil||!a.Mode().IsRegular()||a.Size()<=0||a.Size()>cap{return nil,fixed("input_type_or_cap")}
	f,e:=os.Open(path);if e!=nil{return nil,fixed("input_open")};defer f.Close()
	s,e:=f.Stat();if e!=nil||!os.SameFile(a,s)||s.Size()!=a.Size(){return nil,fixed("input_identity")}
	b,e:=io.ReadAll(io.LimitReader(f,cap+1));if e!=nil||int64(len(b))!=a.Size(){return nil,fixed("input_read")}
	z,e:=os.Lstat(path);if e!=nil||!os.SameFile(a,z)||z.Size()!=a.Size()||!z.ModTime().Equal(a.ModTime()){return nil,fixed("input_changed")};return b,nil
}
func uniquePlan(b []byte) error {
	if !utf8.Valid(b){return fixed("plan_utf8")};d:=json.NewDecoder(bytes.NewReader(b));d.UseNumber();nodes:=0
	var walk func(int)error
	walk=func(depth int)error {
		nodes++;if depth>8||nodes>512{return fixed("plan_json_cap")};v,e:=d.Token();if e!=nil||v==nil{return fixed("plan_json")}
		x,ok:=v.(json.Delim);if !ok{return nil}
		switch x {
		case '{':
			var keys []string
			for d.More(){v,e:=d.Token();k,ok:=v.(string);if e!=nil||!ok||len(keys)>64{return fixed("plan_keys")};for _,c:=range k{if !(c>='a'&&c<='z')&&!(c>='0'&&c<='9')&&c!='_'{return fixed("plan_keys")}};for _,old:=range keys{if old==k{return fixed("plan_duplicate_key")}};keys=append(keys,k);if walk(depth+1)!=nil{return fixed("plan_json")}}
			v,e=d.Token();if e!=nil||v!=json.Delim('}')||(depth==0&&len(keys)!=18){return fixed("plan_json")}
		case '[':
			for d.More(){if walk(depth+1)!=nil{return fixed("plan_json")}};v,e=d.Token();if e!=nil||v!=json.Delim(']'){return fixed("plan_json")}
		default:return fixed("plan_json")
		};return nil
	}
	if walk(0)!=nil{return fixed("plan_json")};if _,e:=d.Token();e!=io.EOF{return fixed("plan_json_tail")};return nil
}

func run(args []string)error {

	planPath:=f.String("plan","","");planSHA:=f.String("plan-sha256","","");out:=f.String("attempt-dir","","")
	if f.Parse(args)!=nil||f.NArg()!=0||*planPath==""||*out==""||!validSHA(*planSHA){return fixed("arguments")}

	b,e:=read(*planPath,65536);if e!=nil{return fixed("plan_read")};h:=sha256.Sum256(b);if hex.EncodeToString(h[:])!=*planSHA{return fixed("plan_sha")}

}
