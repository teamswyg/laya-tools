	var counters Counters
	checkpoint := func(Event) error { checkpoints++; return errors.New("fake_checkpoint_failure") }

		t.Fatal("synthetic_reservation_refusal")

	if err == nil || calls != 0 || checkpoints != 1 || counters.CandidateReserved != 1 {
		t.Fatal("synthetic_sticky_stop")
	}
