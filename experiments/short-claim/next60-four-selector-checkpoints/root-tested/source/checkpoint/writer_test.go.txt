func TestDurableReturnSnapshotPrecedesErrorMethod(t *testing.T) {
	j, r := &fakeTarget{}, &fakeTarget{}
	w := toyWriter(j, r)
	calls := 0
	var counters batchprep.Counters
	errObject := &directFakeError{calls: &calls, beforeCall: func() {
		if j.syncs != 2 {
			t.Fatal("Error called before second Sync")
		}
		fs := frames(t, j.data)
		p := fs[1].Event.Partial
		if !p.Int64Decimal.Known || p.Int64Decimal.Value != "7" || !p.ErrorPresent.Known || !p.ErrorPresent.Value || p.ErrorText.Known || p.ErrorKind.Known {
			t.Fatal("return/pending-text snapshot missing")
		}
	}}
	ops := batchprep.OriginalOps{ParseComma: func(string) (int64, error) {
		if j.syncs != 1 {
			t.Fatal("candidate called before initial Sync")
		}
		return 7, errObject
	}}
	o, e := batchprep.Observe(batchprep.Schedule()[0], toyFixture(), ops, w.Checkpoint, &counters)
	if e != nil || calls != 1 || j.syncs != 4 || w.State().EventsSynced != 4 || w.State().CompletedDispatches != 1 || o.ErrorText == nil || *o.ErrorText != "fake_error" {
		t.Fatal("phase completion")
	}
	fs := frames(t, j.data)
	lines := bytes.Split(bytes.TrimSuffix(j.data, []byte{'\n'}), []byte{'\n'})
	previous := ""
	for i, f := range fs {
		if f.Sequence != i+1 || f.PreviousRecordSHA256 != previous {
			t.Fatal("sequence/digest link")
		}
		sum := sha256.Sum256(append(append([]byte{}, lines[i]...), '\n'))
		previous = hex.EncodeToString(sum[:])
	}
	if w.State().LastRecordSHA256 != previous {
		t.Fatal("terminal digest")
	}
	if fs[2].Event.Counters.ErrorMethodReturned != 1 || !fs[2].Event.Partial.ErrorText.Known {
		t.Fatal("post-error snapshot")
	}
	copy := w.State()
	copy.LastEvent.Partial.ErrorText.Value = "changed"
	if w.State().LastEvent.Partial.ErrorText.Value == "changed" {
		t.Fatal("State escaped mutable payload")
	}
}
