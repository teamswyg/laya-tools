type stream struct {file *os.File;h hash.Hash;seen,retained,cap int64;over bool;failed bool;events chan<-string}
func newStream(f *os.File,cap int64,c chan<-string)*stream{return &stream{file:f,h:sha256.New(),cap:cap,events:c}}
func (s *stream)notify(code string){select{case s.events<-code:default:}}
func (s *stream)Write(b []byte)(int,error){
	if s.seen>math.MaxInt64-int64(len(b)){s.over=true;s.notify("stream_count_overflow");return len(b),nil};s.seen+=int64(len(b));s.h.Write(b)
	if s.seen>s.cap&&!s.over{s.over=true;s.notify("stream_cap")};left:=s.cap-s.retained;if left>0&&!s.failed{n:=len(b);if int64(n)>left{n=int(left)};w,e:=s.file.Write(b[:n]);s.retained+=int64(w);if e!=nil||w!=n{s.failed=true;s.notify("stream_write")}}
	return len(b),nil // continue draining bounded retained prefix after overflow
}

type Process struct {StartAttempts int `json:"time_wrapper_start_attempts"`;StartSucceeded bool `json:"start_succeeded"`;WaitInvocations int `json:"wait_invocations"`;WaitJoined bool `json:"wait_joined"`;ExitZero bool `json:"exit_zero"`;Timeout bool `json:"timed_out"`;Interrupted bool `json:"interrupted"`;Failure string `json:"failure"`;WallSeconds float64 `json:"outside_child_wall_seconds"`;NativeStartupCalls *int `json:"native_startup_calls_measured"`}
func once(p lifecycle,events <-chan string,interrupt <-chan os.Signal,monitor func()error,budget,grace time.Duration)Process{
	r:=Process{StartAttempts:1};start:=time.Now();if p.Start()!=nil{r.Failure="start_failed_no_wait";return r};r.StartSucceeded=true;r.WaitInvocations=1
	done:=make(chan error,1);go func(){done<-p.Wait()}();timer:=time.NewTimer(budget);defer timer.Stop();ticker:=time.NewTicker(20*time.Millisecond);defer ticker.Stop()
	stop:="";running:=true
	for running{select{case e:=<-done:r.WaitJoined=true;r.ExitZero=e==nil;if e!=nil{r.Failure="child_wait_failed"};running=false
	case code:=<-events:stop=code;running=false
	case <-timer.C:r.Timeout=true;stop="outside_timeout";running=false
	case <-interrupt:r.Interrupted=true;stop="outside_interrupted";running=false
	case <-ticker.C:if monitor!=nil&&monitor()!=nil{stop="worker_file_contract";running=false}}}
	if stop!=""{r.Failure=stop;if p.KillGroup()!=nil{r.Failure="kill_group_failed"};cleanup:=time.NewTimer(grace);select{case <-done:r.WaitJoined=true;case <-cleanup.C:r.Failure="cleanup_wait_unjoined"};cleanup.Stop()}
	r.WallSeconds=time.Since(start).Seconds();return r
}
