type fakeProcess struct{startFail bool;starts,waits,kills int;release chan struct{}}
func(f *fakeProcess)Start()error{f.starts++;if f.startFail{return errors.New("fake_start")};return nil}
func(f *fakeProcess)Wait()error{f.waits++;if f.release!=nil{<-f.release};return nil}
func(f *fakeProcess)KillGroup()error{f.kills++;if f.release!=nil{close(f.release)};return nil}
func TestFailedStartNeverWaits(t *testing.T){f:=&fakeProcess{startFail:true};r:=once(f,nil,nil,nil,time.Hour,time.Second);if f.starts!=1||f.waits!=0||f.kills!=0||r.StartSucceeded||r.WaitInvocations!=0||r.WaitJoined{t.Fatal("failed Start accounting")}}
func TestTimeoutKillsAndJoinsSingleWait(t *testing.T){f:=&fakeProcess{release:make(chan struct{})};r:=once(f,nil,nil,nil,0,time.Second);if f.starts!=1||f.waits!=1||f.kills!=1||!r.WaitJoined||!r.Timeout||r.ExitZero{t.Fatal("timeout/reap conservation")}}
func TestStreamOverflowStopsWithSingleJoinedWait(t *testing.T){f:=&fakeProcess{release:make(chan struct{})};events:=make(chan string,1);events<-"stream_cap";r:=once(f,events,nil,nil,time.Hour,time.Second);if f.waits!=1||f.kills!=1||!r.WaitJoined||r.Failure!="stream_cap"{t.Fatal("overflow lifecycle")}}
func TestStrictArgumentShape(t *testing.T){s:=strings.Repeat("0",64);if _,_,e:=arguments([]string{"--binding","/synthetic/binding.json","--binding-sha256",s});e!=nil{t.Fatal(e)};cases:=[][]string{{"--binding","/synthetic/x","--binding","/synthetic/y"},{"--binding=/synthetic/x","--binding-sha256",s},{"--binding","/synthetic/x","--binding-sha256",s,"extra"}};for _,a:=range cases{if _,_,e:=arguments(a);e==nil{t.Fatal("invalid arguments accepted")}}}

}
