type State string

const (
	True    State = "satisfied"
	False   State = "unsatisfied"
	Unknown State = "unknown"
)

type Count struct {
	True    int `json:"satisfied"`
	False   int `json:"known_mismatches"`
	Unknown int `json:"unknown"`
}

func (c *Count) add(s State) {
	switch s {
	case True:
		c.True++
	case False:
		c.False++
	default:
		c.Unknown++
	}
}
func (c Count) state() State {
	if c.False > 0 {
		return False
	}
	if c.Unknown > 0 {
		return Unknown
	}
	return True
}
