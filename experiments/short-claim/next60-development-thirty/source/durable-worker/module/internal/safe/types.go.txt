const (
	FixtureCount   = 4
	CandidateCount = 3
	MaxBytes       = 5 // Historical common4 is not used to truncate the abcde read vector.
	MaxCalls       = 24
)

type Bytes struct {
	Known bool
	Nil   bool
	N     int

}
type ErrorView struct {
	Known      bool
	Nil        bool
	TypeKnown  bool
	Type       string
	PathKnown  bool
	Op         string
	Path       string
	ExistKnown bool
	Exist      bool
	CopyKnown  bool
	Copy       bool
	CloseKnown bool
	Close      bool
}

func presence(e error) ErrorView {
	v := ErrorView{Known: true, Nil: e == nil}
	if e != nil {
		v.TypeKnown = true
		v.Type = reflect.TypeOf(e).String()
		if p, ok := e.(*os.PathError); ok {
			v.PathKnown = true
			v.Op = p.Op
			v.Path = p.Path
		}
	}
	return v // No Error/String/Formatter, Is or Unwrap calls.
}

type fileInfo struct{ size int64 }

func (f fileInfo) Name() string       { return "item" }
func (f fileInfo) Size() int64        { return f.size }
func (f fileInfo) Mode() os.FileMode  { return 0666 }
func (f fileInfo) ModTime() time.Time { return time.Time{} }
func (f fileInfo) IsDir() bool        { return false }
func (f fileInfo) Sys() any           { return nil }
func sum(a, b Counters) Counters {
	return Counters{a.Reserved + b.Reserved, a.Dispatched + b.Dispatched, a.Returned + b.Returned, a.Panicked + b.Panicked, a.Completed + b.Completed, a.IsReserved + b.IsReserved, a.IsDispatched + b.IsDispatched, a.IsReturned + b.IsReturned, a.IsPanicked + b.IsPanicked, a.Stopped || b.Stopped}
}
func capture(fn func()) (pan bool, p any) {
	returned := false
	defer func() {
		if !returned {
			pan = true
			p = recover()
		}
	}()
	fn()
	returned = true
	return false, nil
}
func bytesOf(b []byte) Bytes {
	x := Bytes{Known: true, Nil: b == nil, N: len(b)}
	copy(x.Data[:], b)
	return x
}
func panicType(p any) (string, bool) {
	if t := reflect.TypeOf(p); t != nil {
		return t.String(), true
	}
	return "", false
}
