func ack(Event) error { return nil }

// A declared stdlib synthetic only, never original SafeWriteReader execution
// or an independent proof of exact baseline parity/whole-package semantics.
func syntheticLegacy(f *FakeFS, n string, r io.Reader) (ret error) {
	if ret = f.MkdirAll("data/", 0777); ret != nil {
		return ret
	}
	_, e := f.Stat(n)
	if e == nil {
		return fmt.Errorf("%s already exists", n)
	}
	if !os.IsNotExist(e) {
		return e
	}
	d, e := f.Create(n)
	if e != nil {
		return e
	}
	defer func() {
		if ce := d.Close(); ce != nil {
			ret = errors.Join(ret, ce)
		}
	}()
	_, ret = io.Copy(d, r)
	return ret
}
func text(b Bytes) string { return string(b.Data[:b.N]) }
func TestReferenceFrozenFakes(t *testing.T) {
	c := OwnedCandidates(syntheticLegacy)
	for i, f := range Fixtures() {
		t.Run(string(rune('A'+i)), func(t *testing.T) {
			o, k, e := Observe(f, 1, c[1], Counters{}, ack)
			if e != nil || !o.Returned || k.Returned != 1 || k.Completed != 1 || o.Panic || o.Trace.Stat != 0 || o.Trace.Unsupported {
				t.Fatal("reference boundary")
			}
			switch i {
			case 0, 2:
				if text(o.Contents) != "old" || !o.Error.ExistKnown || !o.Error.Exist || o.Trace.Truncate != 0 || o.Trace.Close != 0 {
					t.Fatal("existing identity and no replacement")
				}
			case 1:
				if text(o.Contents) != "new" || !o.Error.Nil || !o.Trace.ExclusiveKnown || !o.Trace.Exclusive || o.Trace.Close != 1 {
					t.Fatal("successful exclusive open")
				}
			case 3:
				if !o.Error.CopyKnown || !o.Error.Copy || !o.Error.CloseKnown || !o.Error.Close || o.Trace.Close != 1 || o.Trace.BytesRead != 3 || o.Trace.BytesWritten != 3 {
					t.Fatal("separate joined causes")
				}
			}
		})
	}
}

	c := OwnedCandidates(syntheticLegacy)
	for i, fn := range c {
		t.Run(string(rune('A'+i)), func(t *testing.T) {

			if e != nil || o.Trace.HookRuns != 1 || o.Trace.Truncate != 0 {
				t.Fatal("common hook once")

					t.Fatal("Create replacement diagnostics")

				t.Fatal("exclusive refusal")
			}
		})
	}
}
func TestBeforeIdentitySnapshotAndValueCopy(t *testing.T) {
	c := OwnedCandidates(syntheticLegacy)
	var before, after Event
	o, _, e := Observe(Fixtures()[3], 1, c[1], Counters{Reserved: 2, Dispatched: 2, Returned: 2, Completed: 2}, func(ev Event) error {
		if ev.Stage == "before_error_is" && ev.Step == 0 {
			before = ev
		}
		if ev.Stage == "after_candidate" {
			after = ev
		}
		return nil
	})
	if e != nil || !before.Snapshot.Returned || !before.Snapshot.Error.Known || before.Snapshot.Error.CopyKnown || before.Local.IsReserved != 1 || before.Local.IsDispatched != 0 || before.Local.Completed != 0 {
		t.Fatal("pre-identity known channels")
	}
	if !after.Snapshot.Error.Copy || !after.Snapshot.Error.Close || after.Local.IsReturned != 2 || after.Cumulative.Completed != 3 || text(after.Snapshot.Contents) != "new" {
		t.Fatal("final snapshot")
	}
	after.Snapshot.Contents.Data[0] = 'X'
	if text(o.Contents) != "new" || text(before.Snapshot.Contents) != "new" {
		t.Fatal("immutable value copy")
	}
}
func TestStickyCallbackFailureBeforeAcquisition(t *testing.T) {
	c := OwnedCandidates(syntheticLegacy)
	r := Execute(c, func(e Event) error {
		if e.Stage == "before_competition" {
			return errors.New("owned fault")
		}
		return nil
	})
	if r.N != 1 || !r.Counters.Stopped || r.Counters.Dispatched != 1 || r.Rows[0].Trace.Create != 0 || r.Rows[0].Trace.HookRuns != 0 || r.Rows[0].Trace.Close != 0 || r.Rows[0].Trace.Read != 0 {
		t.Fatal("later dispatch after failure")
	}
}

			t.Fatal("order or authority")
		}

}
