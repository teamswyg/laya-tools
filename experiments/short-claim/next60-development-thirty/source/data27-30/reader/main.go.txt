// Future actual Root reader bridge. Kept inert during pure preparation.
// Only LoadDevelopmentRow and value getters; no Features/Score/Project/Fit.
package main

import (
	"bytes"
	"errors"
	"github.com/teamswyg/laya-tools/pkg/shortclaimdata"
	"io"
	"os"
	data30 "riido.local/data30prep"
)

func projectRead(raw []byte) (data30.LoadedView, error) {
	ex, e := shortclaimdata.LoadDevelopmentRow(bytes.NewReader(raw))
	if e != nil || !ex.Valid() {
		return data30.LoadedView{}, errors.New("project_reader_validation")
	}
	m := ex.Metadata()
	p := ex.Input().Prepared()
	s := ex.Supervision()
	v := data30.LoadedView{Valid: true, ID: m.StableID, Request: p.Request, Schema: p.Schema, Provenance: p.Provenance, Role: m.Role, Policy: m.FeaturePolicy, Family: m.SourceFamily, Revision: m.SourceRevision, TextRevision: m.TextRevision, ScopeDescription: m.FiniteScope.Description, Group: m.WholeGroup, Inputs: m.FiniteScope.InputCount, Observations: m.FiniteScope.CandidateObservations, Count: p.Count, SupervisionCount: s.Count, All: m.FiniteScope.AllInputGuarantee, Unseen: m.FiniteScope.UnseenSourceGuarantee, UnusedZero: true}
	if p.Count < 1 || p.Count > len(v.Candidates) || s.Count != p.Count {
		return data30.LoadedView{}, errors.New("project_reader_validation")
	}
	for j := 0; j < len(v.Candidates); j++ {
		c := p.Candidates[j]
		if j < p.Count {
			v.Candidates[j] = data30.LoadedCandidate{ID: c.ID, Text: c.Text, Label: s.Labels[j], Weight: s.Weights[j]}
		} else if c.ID != "" || c.Text != "" || c.NormalizedText != "" || s.Labels[j] || s.Weights[j] != 0 {
			v.UnusedZero = false
		}
	}
	return v, nil
}
func main() {
	names := []string{"--mode", "--previous-data", "--subbody-adoption", "--ini-adoption", "--group-decision", "--group-decision-sha256", "--additional-adoption", "--data", "--data-sha256", "--attempt-dir"}
	v, e := data30.ParseOptions(os.Args[1:], names)
	if e != nil {
		io.WriteString(os.Stderr, "data30_arguments\n")
		os.Exit(2)
	}
	mode, e := data30.ParseMode(v[0])
	if e != nil {
		io.WriteString(os.Stderr, "data30_mode\n")
		os.Exit(2)
	}
	w, e := data30.OpenReaderRecorder(v[9])
	if e != nil {
		io.WriteString(os.Stderr, "data30_reader_reservation_failed\n")
		os.Exit(1)
	}
	s := data30.InitialSummary()
	b, e := data30.LoadInputs(data30.Paths{Mode: mode, Previous: v[1], Sub: v[2], Ini: v[3], Decision: v[4], DecisionSHA: v[5], Additional: v[6]})
	var raw []byte
	if e == nil {
		raw, e = data30.ReadPinned(v[7], v[8], 0)
	}
	if e == nil {
		e = data30.ValidateReader(b, raw, v[8], projectRead, w.Checkpoint, &s)
	}
	if e != nil {
		s.State = "reader_correspondence_failed"
		s.Failure = data30.FailureCode(e)
	}
	if w.Finish(s) != nil {
		io.WriteString(os.Stderr, "data30_reader_receipt_failed\n")
		os.Exit(1)
	}
	if e != nil {
		io.WriteString(os.Stderr, "data30_reader_validation_failed\n")
		os.Exit(1)
	}
	io.WriteString(os.Stdout, "data30_reader_correspondence_saved\n")
}
