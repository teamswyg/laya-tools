func falseValue(v *bool) bool { return v != nil && !*v }
func trueValue(v *bool) bool  { return v != nil && *v }
func equalInts(a, b []int) bool {
	if a == nil || b == nil || len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}
func bind(b Bundle) error {
	if b.Mode != 27 && b.Mode != 30 {
		return failure("target_mode")
	}
	for _, p := range []struct {
		raw  []byte
		size int
		sha  string
	}{{b.Previous, PreviousBytes, PreviousSHA}, {b.Sub, SubBytes, SubSHA}, {b.Ini, IniBytes, IniSHA}} {
		if len(p.raw) != p.size || Digest(p.raw) != p.sha {
			return failure("input_pin")
		}
	}
	if !digestOK(b.DecisionSHA) || Digest(b.Decision) != b.DecisionSHA {
		return failure("external_group_decision_pin")
	}
	return nil
}
func prefix(raw []byte) (Summary, error) {
	s := InitialSummary()
	if len(raw) < 1 || len(raw) > maxDataBytes || raw[len(raw)-1] != '\n' {
		return s, failure("prefix_bounds")
	}
	scan := bufio.NewScanner(bytes.NewReader(raw))
	scan.Buffer(make([]byte, 4096), maxRowBytes+1)
	for scan.Scan() {
		i := s.Requests
		if i >= 23 || len(scan.Bytes()) == 0 {
			return s, failure("prefix_rows")
		}
		r, e := rowDecode(scan.Bytes())
		if e != nil {
			return s, e
		}
		if r.ID != priorIDs[i] || r.Group < 76 || r.Group > 78 {
			return s, failure("prefix_identity")
		}
		if e = countRow(r, &s); e != nil {
			return s, e
		}
	}
	if scan.Err() != nil || s.Requests != 23 || s.Labels != 67 || s.Positive != 23 || s.Negative != 44 || s.Inputs != 113 || s.Selected != 331 || s.PositiveIndex1 != 20 {
		return s, failure("prefix_counts")
	}
	s.Original = 335
	s.PrefixExact = true
	return s, nil
}
func countRow(r Row, s *Summary) error {
	positive := 0
	for i, c := range r.Candidates {
		if c.Label == nil {
			return failure("row_unknown_label")
		}
		for j := 0; j < i; j++ {
			if r.Candidates[j].ID == c.ID {
				return failure("duplicate_candidate")
			}
		}
		s.Labels++
		if *c.Label {
			s.Positive++
			positive++
			if i == 1 {
				s.PositiveIndex1++
			}
		} else {
			s.Negative++
		}
	}
	if positive != 1 {
		return failure("positive_count")
	}
	s.Requests++
	s.Inputs += r.Scope.Inputs
	s.Selected += r.Scope.Observations
	return nil
}
func decision(raw []byte, mode int) (GroupDecision, error) {
	var d GroupDecision
	if e := checkedJSON(raw, &d, -1); e != nil {
		return d, e
	}
	if d.Schema != "riido-next60-Root-whole-family-group-decision-v1" || d.State != "Root_adopted" || len(d.Authority) < 4 || d.Authority[:4] != "Root" || d.Mode != mode || d.Registry != (Pin{37421, RegistrySHA}) || d.ExistingMax != 78 || !trueValue(d.OldUnchanged) || d.Previous != (Pin{PreviousBytes, PreviousSHA}) || d.Sub != (Pin{SubBytes, SubSHA}) || d.Ini != (Pin{IniBytes, IniSHA}) || len(d.Groups) != 3 || len(d.RequestOrder) != mode-23 || d.Allocated != 3 || d.Reassigned != 0 || d.Model != 0 {
		return d, failure("external_group_decision")
	}
	for i, g := range d.Groups {
		if g.Family != familyIDs[i] || g.Revision != familyRevs[i] || g.ID <= 78 || g.ID > 65535 || g.Role != "development_train" {
			return d, failure("external_group_identity")
		}
		for j := 0; j < i; j++ {
			if d.Groups[j].ID == g.ID {
				return d, failure("external_group_collision")
			}
		}
	}
	for i, id := range d.RequestOrder {
		if id != appendIDs[i] {
			return d, failure("external_append_order")
		}
	}
	if mode == 27 && d.Additional != nil || mode == 30 && (d.Additional == nil || d.Additional.Bytes < 1 || d.Additional.Bytes > maxInputBytes || !digestOK(d.Additional.SHA)) {
		return d, failure("remaining_qualification_pin")
	}
	return d, nil
}

func checkCandidate(c selected, n, base int) error {
	if c.Index == nil || *c.Index < 0 || *c.Index >= 3 || !identifier(c.ID, 64) || !textOK(c.Text) || c.Truth == nil || c.Role != "development_train" || c.Weight != 1 || !positions(c.Known, n) || !positions(c.Unknown, n) || c.Retained == nil || len(c.Vector) != n || len(c.Ordinals) != n {
		return failure("selected_shape")
	}
	known := []int{}
	unknown := []int{}
	for k, v := range c.Vector {
		switch v {
		case "satisfied":
		case "unsatisfied":
			known = append(known, k)
		case "unknown":
			unknown = append(unknown, k)
		default:
			return failure("selected_vector_state")
		}
		if c.Ordinals[k] != base+k*3+*c.Index {
			return failure("selected_ordinal")
		}
	}
	if !equalInts(c.Known, known) || !equalInts(c.Unknown, unknown) {
		return failure("selected_evidence_positions")
	}
	for i, u := range c.Retained {
		if u.Fixture < 0 || u.Fixture >= n || !textOK(u.Field) || !textOK(u.Reason) {
			return failure("retained_unknown_shape")
		}
		for j := 0; j < i; j++ {
			if c.Retained[j].Fixture == u.Fixture && c.Retained[j].Field == u.Field {
				return failure("retained_unknown_duplicate")
			}
		}
	}
	if *c.Truth {
		if len(known) != 0 || len(unknown) != 0 || len(c.Retained) != 0 {
			return failure("positive_not_all_known_true")
		}
	} else if len(known) == 0 {
		return failure("negative_without_known_counterexample")
	}
	return nil
}
func inspect(b Bundle) ([7]request, GroupDecision, Summary, error) {
	var all [7]request
	s, e := prefix(b.Previous)
	if e != nil {
		return all, GroupDecision{}, s, e
	}
	d, e := decision(b.Decision, b.Mode)
	if e != nil {
		return all, d, s, e
	}
	s.GroupDecisionSHA = b.DecisionSHA
	var a, z qualification
	if e = checkedJSON(b.Sub, &a, -1); e != nil {
		return all, d, s, e
	}
	if e = checkedJSON(b.Ini, &z, -1); e != nil {
		return all, d, s, e
	}
	if a.Schema != "riido-subbody-Root-finite-development-qualification-v2" || z.Schema != "riido-ini-two-Root-finite-development-qualification-v1" || a.State != "Root_adopted_two_finite_semantic_requests_data_materialization_separate" || z.State != a.State || !trueValue(a.UnknownNeverFalse) || !trueValue(z.UnknownNeverFalse) || len(a.Requests) != 2 || len(z.Requests) != 2 {
		return all, d, s, failure("Root_qualification_header")
	}
	copy(all[0:2], a.Requests)
	copy(all[2:4], z.Requests)
	if b.Mode == 30 {
		if len(b.Additional) != d.Additional.Bytes || Digest(b.Additional) != d.Additional.SHA {
			return all, d, s, failure("remaining_qualification_pin")
		}
		var x qualification
		if e = checkedJSON(b.Additional, &x, -1); e != nil {
			return all, d, s, e
		}
		if x.Schema != "riido-remaining-three-Root-finite-development-qualification-v1" || x.State != "Root_adopted_three_finite_semantic_requests_data_materialization_separate" || len(x.Requests) != 3 || !trueValue(x.UnknownNeverFalse) {
			return all, d, s, failure("remaining_qualification_pending_or_header")
		}
		copy(all[4:7], x.Requests)
	} else if len(b.Additional) != 0 {
		return all, d, s, failure("unexpected_remaining_qualification")
	}
	for i, r := range all[:b.Mode-23] {
		f := familyAt[i]
		if r.ID != appendIDs[i] || r.Family != familyIDs[f] || r.Revision != familyRevs[f] || r.Inputs != inputsAt[i] || r.Group != nil || !trueValue(r.Pending) || !textOK(r.Request) || !textOK(r.Scope) || !identifier(r.TextRevision, 128) || len(r.Selected) < 2 || len(r.Selected) > 3 || r.Observations != r.Inputs*len(r.Selected) || r.Excluded == nil {
			return all, d, s, failure("Root_request_binding")
		}
		if i < 4 && (len(r.Selected) != 3 || len(r.Excluded) != 0) {
			return all, d, s, failure("fixed_four_candidate_count")
		}
		if i < 2 && r.TextRevision != "subbody-finite-v1" || i >= 2 && i < 4 && r.TextRevision != "ini-two-finite-v1" {
			return all, d, s, failure("text_revision")
		}
		if i >= 4 && r.Request != futureRequests[i-4] {
			return all, d, s, failure("remaining_frozen_request")
		}
		positive := 0
		last := -1
		for j, c := range r.Selected {
			base := 0
			if i == 1 {
				base = 15
			}
			if i >= 4 {
				if r.OrdinalBase == nil || *r.OrdinalBase < 0 || *r.OrdinalBase > 1024 {
					return all, d, s, failure("remaining_evidence_ordinal_base")
				}
				base = *r.OrdinalBase
			}
			if e = checkCandidate(c, r.Inputs, base); e != nil {
				return all, d, s, e
			}
			idx := *c.Index
			if idx <= last {
				return all, d, s, failure("selected_source_order")
			}
			last = idx
			if i < 4 && idx != j {
				return all, d, s, failure("fixed_four_source_order")
			}
			if i < 2 {
				expected := [3]string{"baseline_original", "reference_authored", "negative_authored"}
				if c.ID != expected[idx] {
					return all, d, s, failure("source_candidate_ID")
				}
			} else {
				expected := [3]string{"baseline-design", "reference-design", "wrong-seed-design"}
				if c.ID != expected[idx] {
					return all, d, s, failure("source_candidate_ID")
				}
			}
			if i >= 4 && c.Text != futureCaptions[i-4][idx] {
				return all, d, s, failure("remaining_frozen_caption")
			}
			if *c.Truth {
				positive++
			}
			for _, u := range c.Retained {
				s.Unknowns = append(s.Unknowns, RetainedUnknown{r.ID, idx, u})
			}
		}
		if positive != 1 {
			return all, d, s, failure("Root_positive_count")
		}
	}
	return all, d, s, nil
}
func newRow(a request, g int) Row {
	no1, no2 := false, false
	r := Row{Schema: "riido-finite-development-request-row-v1", ID: a.ID, Request: a.Request, Role: "development_train", Group: g, Family: a.Family, Revision: a.Revision, TextRevision: a.TextRevision, Policy: "request_and_candidates_text_only", Scope: finiteScope{a.Inputs, a.Observations, a.Scope, &no1, &no2}}
	for _, c := range a.Selected {
		v := *c.Truth
		r.Candidates = append(r.Candidates, candidate{c.ID, c.Text, &v, c.Weight})
	}
	return r
}

// composeBound is private; tests supply only owned synthetic DTOs. Public callers
// always bind the original23 and both immutable Root qualifiers first.
func composeBound(b Bundle) ([]byte, Summary, error) {
	reqs, d, s, e := inspect(b)
	if e != nil {
		return nil, s, e
	}
	raw := bytes.Clone(b.Previous)
	for i, a := range reqs[:b.Mode-23] {
		r := newRow(a, d.Groups[familyAt[i]].ID)
		encoded, e := json.Marshal(r)
		if e != nil {
			return nil, s, failure("row_encode")
		}
		checked, e := rowDecode(encoded)
		if e != nil {
			return nil, s, e
		}
		if e = countRow(checked, &s); e != nil {
			return nil, s, e
		}
		if len(raw)+len(encoded)+1 > maxDataBytes {
			return nil, s, failure("data_bounds")
		}
		raw = append(raw, encoded...)
		raw = append(raw, '\n')
		s.Original += a.Inputs * 3
	}
	if s.Requests != b.Mode || s.Positive != b.Mode || s.Inputs != mapInputs(b.Mode) {
		return nil, s, failure("combined_counts")
	}
	if b.Mode == 27 && (s.Labels != 79 || s.Negative != 52 || s.Original != 392 || s.Selected != 388 || s.PositiveIndex1 != 24) {
		return nil, s, failure("combined27_counts")
	}
	s.CopiedRoot = true
	s.DataSHA = Digest(raw)
	s.DataBytes = len(raw)
	s.State = "composed_Root_adopted_data_only"
	return raw, s, nil
}
func mapInputs(mode int) int {
	if mode == 27 {
		return 132
	}
	return 146
}
func ComposePinned(b Bundle) ([]byte, Summary, error) {
	if e := bind(b); e != nil {
		return nil, InitialSummary(), e
	}
	return composeBound(b)
}
func loadedMatches(v LoadedView, r Row) bool {
	if !v.Valid || !v.UnusedZero || v.ID != r.ID || v.Request != r.Request || v.Schema != "riido-short-behavior-claim-v1" || v.Provenance != "finite-development-v1" || v.Role != r.Role || v.Policy != r.Policy || v.Family != r.Family || v.Revision != r.Revision || v.TextRevision != r.TextRevision || v.ScopeDescription != r.Scope.Description || v.Group != r.Group || v.Inputs != r.Scope.Inputs || v.Observations != r.Scope.Observations || v.Count != len(r.Candidates) || v.SupervisionCount != len(r.Candidates) || v.All || v.Unseen {
		return false
	}
	for i, c := range v.Candidates {
		if i >= len(r.Candidates) {
			if c != (LoadedCandidate{}) {
				return false
			}
			continue
		}
		w := r.Candidates[i]
		if c.ID != w.ID || c.Text != w.Text || c.Label != *w.Label || c.Weight != w.Weight {
			return false
		}
	}
	return true
}
func cloneSummary(s Summary) Summary {
	s.Unknowns = append([]RetainedUnknown{}, s.Unknowns...)
	return s
}
func readOne(reader Reader, raw []byte) (v LoadedView, e error, returned bool) {
	defer func() {
		if recover() != nil {
			e = failure("reader_panic")
			returned = false
		}
	}()
	v, e = reader(bytes.Clone(raw))
	returned = true
	return
}
func ValidateReader(b Bundle, data []byte, sha string, reader Reader, progress Progress, out *Summary) error {
	if out == nil {
		return failure("reader_arguments")
	}
	*out = InitialSummary()
	if e := bind(b); e != nil {
		return e
	}
	if !digestOK(sha) || Digest(data) != sha {
		return failure("data_pin")
	}
	return validateBound(b, data, reader, progress, out)
}
func validateBound(b Bundle, data []byte, reader Reader, progress Progress, out *Summary) error {
	if reader == nil || progress == nil || out == nil {
		return failure("reader_arguments")
	}
	expected, s, e := composeBound(b)
	if e != nil {
		return e
	}
	if !bytes.Equal(data, expected) || !bytes.HasPrefix(data, b.Previous) {
		return failure("data_correspondence")
	}
	var rows [30]Row
	var rawRows [30][]byte
	scan := bufio.NewScanner(bytes.NewReader(data))
	scan.Buffer(make([]byte, 4096), maxRowBytes+1)
	n := 0
	for scan.Scan() {
		if n >= b.Mode {
			return failure("reader_rows")
		}
		r, e := rowDecode(scan.Bytes())
		if e != nil {
			return e
		}
		rows[n] = r
		rawRows[n] = bytes.Clone(scan.Bytes())
		n++
	}
	if scan.Err() != nil || n != b.Mode {
		return failure("reader_rows")
	}
	expectedSummary := s
	s.Requests = 0
	s.Labels = 0
	s.Positive = 0
	s.Negative = 0
	s.Inputs = 0
	s.Selected = 0
	s.PositiveIndex1 = 0
	s.State = "reader_correspondence_in_progress"
	*out = s
	for i, r := range rows[:n] {
		out.ReaderAttempts++
		if e = progress(cloneSummary(*out)); e != nil {
			return failure("reader_checkpoint_failed")
		}
		v, re, returned := readOne(reader, rawRows[i])
		if returned {
			out.ReaderReturns++
		}
		if re != nil || !loadedMatches(v, r) {
			out.Failure = "project_reader_value_correspondence"
			if re != nil {
				out.Failure = FailureCode(re)
			}
			if e = progress(cloneSummary(*out)); e != nil {
				return failure("reader_checkpoint_failed")
			}
			return failure("project_reader_value_correspondence")
		}
		out.ReaderMatches++
		if e = countRow(r, out); e != nil {
			return e
		}
		if e = progress(cloneSummary(*out)); e != nil {
			return failure("reader_checkpoint_failed")
		}
	}
	expectedSummary.State = s.State
	expectedSummary.ReaderAttempts = n
	expectedSummary.ReaderReturns = n
	expectedSummary.ReaderMatches = n
	if !reflect.DeepEqual(*out, expectedSummary) {
		return failure("reader_final_counts")
	}
	out.State = "complete_reader_value_correspondence_only"
	return nil
}
func EncodeSummary(s Summary) ([]byte, error) {
	if s.LabelsAssigned || s.Qualified || s.UnknownFalse || s.Features != 0 || s.Score != 0 || s.Project != 0 || s.Fit != 0 || s.OriginalCalls != 0 || s.Model != 0 {
		return nil, failure("summary_authority")
	}
	raw, e := json.Marshal(s)
	if e != nil || len(raw)+1 > maxSummaryBytes {
		return nil, failure("summary_bounds")
	}
	return append(raw, '\n'), nil
}
