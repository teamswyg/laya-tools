func ownedPrefix() []byte {
	raw := []byte{}
	for i, id := range priorIDs {
		n := 5
		if i == 1 || i == 17 {
			n = 4
		}
		nc := 3
		if i == 1 || i == 17 {
			nc = 2
		}
		pos := 1
		if i == 0 || i == 17 || i == 20 {
			pos = 0
		}
		r := Row{Schema: "riido-finite-development-request-row-v1", ID: id, Request: "Owned synthetic request.", Role: "development_train", Group: 77, Family: "owned-synthetic-family", Revision: familyRevs[0], TextRevision: "owned-control-v1", Policy: "request_and_candidates_text_only", Scope: finiteScope{n, n * nc, "Owned synthetic finite inputs.", flag(false), flag(false)}}
		for j := 0; j < nc; j++ {
			r.Candidates = append(r.Candidates, candidate{[3]string{"owned-a", "owned-b", "owned-c"}[j], "Owned synthetic claim.", flag(j == pos), 1})
		}
		raw = append(raw, marshal(r)...)
		raw = append(raw, '\n')
	}
	return raw
}
func ownedRequest(i int) request {
	n := inputsAt[i]
	r := request{ID: appendIDs[i], Request: "Owned synthetic request.", Family: familyIDs[familyAt[i]], Revision: familyRevs[familyAt[i]], Scope: "Owned synthetic finite inputs.", Inputs: n, Pending: flag(true), TextRevision: "ini-two-finite-v1", Observations: n * 3, Excluded: []json.RawMessage{}}
	if i < 2 {
		r.TextRevision = "subbody-finite-v1"
	}
	if i >= 4 {
		r.Request = futureRequests[i-4]
		r.TextRevision = "remaining-three-finite-v1"
		r.OrdinalBase = integer(0)
	}
	base := 0
	if i == 1 {
		base = 15
	}
	for j := 0; j < 3; j++ {
		ids := [3]string{"baseline-design", "reference-design", "wrong-seed-design"}
		if i < 2 {
			ids = [3]string{"baseline_original", "reference_authored", "negative_authored"}
		}
		text := "Owned synthetic claim."
		if i >= 4 {
			text = futureCaptions[i-4][j]
		}
		c := selected{Index: integer(j), ID: ids[j], Text: text, Truth: flag(j == 1), Role: "development_train", Weight: 1, Known: []int{}, Unknown: []int{}, Retained: []unknownPredicate{}, Ordinals: []int{}, Vector: []string{}}
		for k := 0; k < n; k++ {
			c.Ordinals = append(c.Ordinals, base+k*3+j)
			v := "satisfied"
			if j != 1 && k == 0 {
				v = "unsatisfied"
				c.Known = append(c.Known, k)
			}
			c.Vector = append(c.Vector, v)
		}
		r.Selected = append(r.Selected, c)
	}
	return r
}
func ownedBundle(mode int) Bundle {
	b := Bundle{Mode: mode, Previous: ownedPrefix()}
	sub := qualification{Schema: "riido-subbody-Root-finite-development-qualification-v2", State: "Root_adopted_two_finite_semantic_requests_data_materialization_separate", Requests: []request{ownedRequest(0), ownedRequest(1)}, UnknownNeverFalse: flag(true)}
	ini := qualification{Schema: "riido-ini-two-Root-finite-development-qualification-v1", State: sub.State, Requests: []request{ownedRequest(2), ownedRequest(3)}, UnknownNeverFalse: flag(true)}
	b.Sub = marshal(sub)
	b.Ini = marshal(ini)
	d := GroupDecision{Schema: "riido-next60-Root-whole-family-group-decision-v1", State: "Root_adopted", Authority: "Root owned synthetic control", Mode: mode, Registry: Pin{37421, RegistrySHA}, ExistingMax: 78, OldUnchanged: flag(true), Previous: Pin{PreviousBytes, PreviousSHA}, Sub: Pin{SubBytes, SubSHA}, Ini: Pin{IniBytes, IniSHA}, Allocated: 3, Groups: []Group{}, RequestOrder: appendIDs[:mode-23]}
	for i, f := range familyIDs {
		d.Groups = append(d.Groups, Group{f, familyRevs[i], 79 + i, "development_train"})
	}
	if mode == 30 {
		x := qualification{Schema: "riido-remaining-three-Root-finite-development-qualification-v1", State: "Root_adopted_three_finite_semantic_requests_data_materialization_separate", Requests: []request{ownedRequest(4), ownedRequest(5), ownedRequest(6)}, UnknownNeverFalse: flag(true)}
		b.Additional = marshal(x)
		d.Additional = &Pin{len(b.Additional), Digest(b.Additional)}
	}
	b.Decision = marshal(d)
	b.DecisionSHA = Digest(b.Decision)
	return b
}
func fakeRead(raw []byte) (LoadedView, error) {
	r, e := rowDecode(raw)
	if e != nil {
		return LoadedView{}, e
	}
	v := LoadedView{Valid: true, ID: r.ID, Request: r.Request, Schema: "riido-short-behavior-claim-v1", Provenance: "finite-development-v1", Role: r.Role, Policy: r.Policy, Family: r.Family, Revision: r.Revision, TextRevision: r.TextRevision, ScopeDescription: r.Scope.Description, Group: r.Group, Inputs: r.Scope.Inputs, Observations: r.Scope.Observations, Count: len(r.Candidates), SupervisionCount: len(r.Candidates), UnusedZero: true}
	for i, c := range r.Candidates {
		v.Candidates[i] = LoadedCandidate{c.ID, c.Text, *c.Label, c.Weight}
	}
	return v, nil
}
func changeDecision(b *Bundle, f func(*GroupDecision)) {
	var d GroupDecision
	json.Unmarshal(b.Decision, &d)
	f(&d)
	b.Decision = marshal(d)
	b.DecisionSHA = Digest(b.Decision)
}
func changeQualification(b *Bundle, which int, f func(*qualification)) {
	raw := b.Sub
	if which == 1 {
		raw = b.Ini
	}
	if which == 2 {
		raw = b.Additional
	}
	var q qualification
	json.Unmarshal(raw, &q)
	f(&q)
	raw = marshal(q)
	switch which {
	case 0:
		b.Sub = raw
	case 1:
		b.Ini = raw
	case 2:
		b.Additional = raw
		changeDecision(b, func(d *GroupDecision) { d.Additional = &Pin{len(raw), Digest(raw)} })
	}
}

			}

func TestReaderRecorderRealOwnedFlowAndStickyFailure(t *testing.T) {
	b := ownedBundle(27)
	data, _, e := composeBound(b)
	if e != nil {
		t.Fatal(e)
	}
	w, e := OpenReaderRecorder(filepath.Join(t.TempDir(), "synthetic-recorder"))
	if e != nil {
		t.Fatal(e)
	}
	s := InitialSummary()
	if e = validateBound(b, data, fakeRead, w.Checkpoint, &s); e != nil {
		t.Fatal(e)
	}
	if e = w.Finish(s); e != nil {
		t.Fatal(e)
	}
	if w.Checkpoint(s) == nil {
		t.Fatal("post-final accepted")
	}
	w, e = OpenReaderRecorder(filepath.Join(t.TempDir(), "failure-recorder"))
	if e != nil {
		t.Fatal(e)
	}
	s = InitialSummary()
	e = validateBound(b, data, func([]byte) (LoadedView, error) { panic("owned") }, w.Checkpoint, &s)
	if e == nil {
		t.Fatal("owned panic accepted")
	}
	s.State = "reader_correspondence_failed"
	if e = w.Finish(s); e != nil {
		t.Fatal("failure receipt", e)
	}
	w, e = OpenReaderRecorder(filepath.Join(t.TempDir(), "closed-recorder"))
	if e != nil {
		t.Fatal(e)
	}
	w.journal.Close()
	s = InitialSummary()
	s.State = "reader_correspondence_in_progress"
	s.ReaderAttempts = 1
	if w.Checkpoint(s) == nil || w.Checkpoint(s) == nil {
		t.Fatal("closed write did not stop")
	}
	w.Close()
}
