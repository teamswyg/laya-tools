type failure string

func (e failure) Error() string { return string(e) }
func FailureCode(e error) string {
	if e == nil {
		return ""
	}
	if f, ok := e.(failure); ok {
		return string(f)
	}
	return "data30_callback_failed"
}

func ParseMode(s string) (int, error) {
	if s == "27" {
		return 27, nil
	}
	if s == "30" {
		return 30, nil
	}
	return 0, failure("target_mode")
}
