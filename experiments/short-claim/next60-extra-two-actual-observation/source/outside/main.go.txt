func failure(s string) error { return errors.New(s) }
func fixedCode(e error) string {
	if e == nil {
		return ""
	}
	s := e.Error()
	allowed := [14]string{"extratwo_outside_config_invalid", "extratwo_outside_pin_invalid", "extratwo_outside_build_invalid",
		"extratwo_outside_path_invalid", "extratwo_outside_output_invalid", "extratwo_outside_start_failed",
		"extratwo_outside_wait_failed", "extratwo_outside_deadline", "extratwo_outside_capture_cap",
		"extratwo_outside_worker_artifacts_invalid", "extratwo_outside_worker_protocol_invalid",
		"extratwo_outside_metrics_invalid", "extratwo_outside_rss_limit", "extratwo_outside_bindings_changed"}
	for _, a := range allowed {
		if s == a {
			return a
		}
	}
	return "extratwo_outside_output_invalid"
}
func path(s string) bool { return s != "" && filepath.IsAbs(s) && filepath.Clean(s) == s }
func contains(parent, child string) bool {
	rel, e := filepath.Rel(parent, child)
	return e == nil && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator))
}
func regularRead(path string, cap int) ([]byte, os.FileInfo, error) {
	st, e := os.Lstat(path)
	if e != nil || !st.Mode().IsRegular() || st.Size() < 0 || st.Size() > int64(cap) {
		return nil, nil, failure("extratwo_outside_pin_invalid")
	}
	f, e := os.Open(path)
	if e != nil {
		return nil, nil, failure("extratwo_outside_pin_invalid")
	}
	defer f.Close()
	now, e := f.Stat()
	if e != nil || !now.Mode().IsRegular() || !os.SameFile(st, now) || now.Size() != st.Size() || !now.ModTime().Equal(st.ModTime()) {
		return nil, nil, failure("extratwo_outside_pin_invalid")
	}
	raw, e := io.ReadAll(io.LimitReader(f, int64(cap)+1))
	after, se := f.Stat()
	if e != nil || se != nil || len(raw) > cap || int64(len(raw)) != st.Size() || after.Size() != st.Size() || !os.SameFile(st, after) || !after.ModTime().Equal(st.ModTime()) {
		return nil, nil, failure("extratwo_outside_pin_invalid")
	}
	return raw, st, nil
}
func sum(raw []byte) string { s := sha256.Sum256(raw); return hex.EncodeToString(s[:]) }
func verify(p Pin, cap int) (PinFact, os.FileInfo, error) {
	fact := PinFact{ID: p.ID, Bytes: p.Bytes, SHA: p.SHA}
	if p.ID == "" || !path(p.Path) || p.Bytes <= 0 || p.Bytes > cap || !protocol.SHA(p.SHA) {
		return fact, nil, failure("extratwo_outside_pin_invalid")
	}
	raw, st, e := regularRead(p.Path, cap)
	if e != nil || len(raw) != p.Bytes || sum(raw) != p.SHA {
		return fact, nil, failure("extratwo_outside_pin_invalid")
	}
	fact.Verified = true
	return fact, st, nil
}
func pin(raw []byte) (Pin, error) {
	v, e := protocol.Object(raw, []string{"id", "path", "bytes", "sha256"})
	if e != nil {
		return Pin{}, failure("extratwo_outside_config_invalid")
	}
	var p Pin
	p.ID, e = protocol.ReadString(v[0])
	if e != nil {
		return p, failure("extratwo_outside_config_invalid")
	}
	p.Path, e = protocol.ReadString(v[1])
	if e != nil {
		return p, failure("extratwo_outside_config_invalid")
	}
	p.Bytes, e = protocol.ReadInt(v[2])
	if e != nil {
		return p, failure("extratwo_outside_config_invalid")
	}
	p.SHA, e = protocol.ReadString(v[3])
	if e != nil {
		return p, failure("extratwo_outside_config_invalid")
	}
	return p, nil
}
func readConfig(file, sha, out string) (Config, error) {
	var c Config
	raw, _, e := regularRead(file, configCap)
	if e != nil || !protocol.SHA(sha) || sum(raw) != sha {
		return c, failure("extratwo_outside_config_invalid")
	}
	c.ConfigPin = Pin{"extratwo/config", file, len(raw), sha}
	v, e := protocol.Object(raw, []string{"schema", "controller_directory", "worker_attempt_directory", "family", "worker_module", "worker_package", "worker_binary", "controller_binary", "execution_plan", "time_binary", "root_receipts", "fixtures", "wants", "max_wall_seconds", "max_rss_bytes", "heap_limit_bytes", "max_stdout_bytes", "max_stderr_bytes"})
	if e != nil {
		return c, failure("extratwo_outside_config_invalid")
	}
	s, e := protocol.ReadString(v[0])
	if e != nil || s != schema {
		return c, failure("extratwo_outside_config_invalid")
	}
	c.Directory, e = protocol.ReadString(v[1])
	if e != nil || c.Directory != out || !path(c.Directory) {
		return c, failure("extratwo_outside_path_invalid")
	}
	c.Attempt, e = protocol.ReadString(v[2])
	if e != nil || !path(c.Attempt) || contains(c.Directory, c.Attempt) || contains(c.Attempt, c.Directory) {
		return c, failure("extratwo_outside_path_invalid")
	}
	c.Family, e = protocol.ReadInt(v[3])
	if e != nil || c.Family < 0 || c.Family > 1 {
		return c, failure("extratwo_outside_config_invalid")
	}
	c.WorkerModule, e = protocol.ReadString(v[4])
	if e != nil || c.WorkerModule != protocol.Modules[c.Family] {
		return c, failure("extratwo_outside_config_invalid")
	}
	c.WorkerPackage, e = protocol.ReadString(v[5])
	if e != nil || c.WorkerPackage != c.WorkerModule+"/cmd/observe" {
		return c, failure("extratwo_outside_config_invalid")
	}
	targets := [6]*Pin{&c.Worker, &c.Self, &c.Plan, &c.Time, &c.Fixtures, &c.Wants}
	fields := [6]int{6, 7, 8, 9, 11, 12}
	ids := [6]string{"extratwo/worker", "extratwo/controller", "extratwo/execution-plan", "extratwo/system-time", "extratwo/fixtures", "extratwo/wants"}
	for i, p := range targets {
		*p, e = pin(v[fields[i]])
		if e != nil || p.ID != ids[i] {
			return c, failure("extratwo_outside_config_invalid")
		}
	}
	if c.Time.Path != "/usr/bin/time" || c.Fixtures.Bytes != protocol.FixtureBytes || c.Fixtures.SHA != protocol.FixtureSHA || c.Wants.Bytes != protocol.WantsBytes || c.Wants.SHA != protocol.WantsSHA {
		return c, failure("extratwo_outside_config_invalid")
	}
	var evidence []json.RawMessage
	if json.Unmarshal(v[10], &evidence) != nil || len(evidence) != 4 {
		return c, failure("extratwo_outside_config_invalid")
	}
	required := [4]string{"extratwo/root-source", "extratwo/root-compiler", "extratwo/root-rights", "extratwo/root-resource"}
	for i, r := range evidence {
		c.Evidence[i], e = pin(r)
		if e != nil || c.Evidence[i].ID != required[i] {
			return c, failure("extratwo_outside_config_invalid")
		}
	}
	c.Wall, e = protocol.ReadInt(v[13])
	if e != nil || c.Wall != 60 {
		return c, failure("extratwo_outside_config_invalid")
	}
	limits := [4]int{heapLimit, heapLimit, stdoutCap, stderrCap}
	for i, n := range limits {
		got, e := protocol.ReadInt(v[14+i])
		if e != nil || got != n {
			return c, failure("extratwo_outside_config_invalid")
		}
	}
	executable, e := os.Executable()
	if e != nil || filepath.Clean(executable) != c.Self.Path {
		return c, failure("extratwo_outside_pin_invalid")
	}
	all := allPins(c)
	for i, p := range all {
		if contains(c.Directory, p.Path) || contains(c.Attempt, p.Path) {
			return c, failure("extratwo_outside_path_invalid")
		}
		for j := 0; j < i; j++ {
			if p.Path == all[j].Path {
				return c, failure("extratwo_outside_path_invalid")
			}
		}
	}
	return c, nil
}
func build(file, module, pkg string) error {
	m, e := macho.Open(file)
	if e != nil {
		return failure("extratwo_outside_build_invalid")
	}
	ok := m.Magic == macho.Magic64 && m.Cpu == macho.CpuArm64 && m.Type == macho.TypeExec
	m.Close()
	if !ok {
		return failure("extratwo_outside_build_invalid")
	}
	info, e := buildinfo.ReadFile(file)
	if e != nil || !validBuildInfo(info, module, pkg) {
		return failure("extratwo_outside_build_invalid")
	}
	return nil
}
func validBuildInfo(info *debug.BuildInfo, module, pkg string) bool {
	if info == nil || info.GoVersion != "go1.27.1" || info.Path != pkg || info.Main.Path != module {
		return false
	}
	required := [5][2]string{{"GOOS", "darwin"}, {"GOARCH", "arm64"}, {"CGO_ENABLED", "0"}, {"-trimpath", "true"}, {"-buildmode", "exe"}}
	for _, r := range required {
		found := 0
		for _, s := range info.Settings {
			if s.Key == r[0] {
				if s.Value != r[1] {
					return false
				}
				found++
			}
		}
		if found != 1 {
			return false
		}
	}
	return true
}
func allPins(c Config) [11]Pin {
	return [11]Pin{c.Worker, c.Self, c.Plan, c.Time, c.Evidence[0], c.Evidence[1], c.Evidence[2], c.Evidence[3], c.Fixtures, c.Wants, c.ConfigPin}
}
func verifyAll(c Config) ([]PinFact, error) {
	pins := allPins(c)
	facts := make([]PinFact, 0, len(pins))
	var physical [11]os.FileInfo
	for i, p := range pins {
		bound := binaryCap
		if i == 2 || i == 10 {
			bound = configCap
		}
		if i >= 4 && i <= 7 {
			bound = evidenceCap
		}
		if i == 8 || i == 9 {
			bound = 1 << 20
		}
		f, st, e := verify(p, bound)
		facts = append(facts, f)
		if e != nil {
			return facts, e
		}
		real, re := filepath.EvalSymlinks(p.Path)
		if re != nil || real != p.Path {
			return facts, failure("extratwo_outside_path_invalid")
		}
		for j := 0; j < i; j++ {
			if os.SameFile(st, physical[j]) {
				return facts, failure("extratwo_outside_path_invalid")
			}
		}
		physical[i] = st
	}
	return facts, nil
}
func canonicalParent(file string) bool {
	parent := filepath.Dir(file)
	st, e := os.Lstat(parent)
	if e != nil || !st.IsDir() || st.Mode()&os.ModeSymlink != 0 {
		return false
	}
	real, e := filepath.EvalSymlinks(parent)
	return e == nil && real == parent
}

// The worker only receives the frozen plan digest as its opaque identifier.
// Root's source/compiler/rights/resource conclusions are hashed, not approved
// or interpreted by this collector. No fixture Want is evaluated here.
func bindInputs(c Config) ([]PinFact, error) {
	ps := [2]Pin{c.Fixtures, c.Wants}
	facts := make([]PinFact, 0, 2)
	for _, p := range ps {
		f, _, e := verify(p, 1<<20)
		facts = append(facts, f)
		if e != nil {
			return facts, e
		}
	}
	return facts, nil
}

type capture struct {
	file     *os.File
	cap, n   int
	exceeded bool
	signal   chan<- struct{}
}

func (w *capture) Write(p []byte) (int, error) {
	if len(p) > w.cap-w.n {
		left := w.cap - w.n
		if left > 0 {
			n, e := w.file.Write(p[:left])
			w.n += n
			if e != nil {
				return n, e
			}
		}
		w.exceeded = true
		select {
		case w.signal <- struct{}{}:
		default:
		}
		return left, failure("extratwo_outside_capture_cap")
	}
	n, e := w.file.Write(p)
	w.n += n
	return n, e
}
func write(f *os.File, v any, cap int) error {
	raw, e := json.Marshal(v)
	if e != nil {
		return failure("extratwo_outside_output_invalid")
	}
	raw = append(raw, '\n')
	if len(raw) > cap {
		return failure("extratwo_outside_output_invalid")
	}
	n, e := f.Write(raw)
	if e != nil || n != len(raw) || f.Sync() != nil {
		return failure("extratwo_outside_output_invalid")
	}
	return nil
}
func syncDirectory(path string) error {
	before, e := os.Lstat(path)
	if e != nil || !before.IsDir() || before.Mode()&os.ModeSymlink != 0 {
		return failure("extratwo_outside_output_invalid")
	}
	f, e := os.Open(path)
	if e != nil {
		return failure("extratwo_outside_output_invalid")
	}
	defer f.Close()
	now, e := f.Stat()
	if e != nil || !now.IsDir() || !os.SameFile(before, now) || f.Sync() != nil {
		return failure("extratwo_outside_output_invalid")
	}
	return nil
}

type waitOutcome struct {
	Err                   error
	Deadline, Cap, Killed bool
	KillOK                *bool
}

func await(done <-chan error, deadline <-chan time.Time, capSignal <-chan struct{}, kill func() bool) waitOutcome {
	var out waitOutcome
	select {
	case out.Err = <-done:
		return out
	case <-deadline:
		out.Deadline = true
	case <-capSignal:
		out.Cap = true
	}
	out.Killed = true
	ok := kill()
	out.KillOK = &ok
	// One existing Wait goroutine reaps the started wrapper after termination.
	// SIGKILL/reaping is an OS operation, not a formal whole-kernel time proof.
	out.Err = <-done
	return out
}

// The production hooks wrap one exec.Cmd. Tests substitute callbacks; they
// never create a process or import an original package.
type processHooks struct {
	Start func() error
	Wait  func() error
	Kill  func() bool
	Exit  func() *int
}

func startWait(h processHooks, deadline <-chan time.Time, capSignal <-chan struct{}) (p Process, waitErr error) {
	began := time.Now()
	p.StartAttempts = 1
	startErr := h.Start()
	p.StartSucceeded = startErr == nil
	p.WaitAttempts = 1
	if startErr != nil {
		_ = h.Wait()
		p.WaitReturned = true
		p.WallSeconds = time.Since(began).Seconds()
		return p, failure("extratwo_outside_start_failed")
	}
	done := make(chan error, 1)
	go func() { done <- h.Wait() }()
	w := await(done, deadline, capSignal, h.Kill)
	p.DeadlineExceeded = w.Deadline
	p.CaptureCapExceeded = w.Cap
	p.GroupKillAttempted = w.Killed
	p.GroupKillSucceeded = w.KillOK
	p.WaitReturned = true
	p.WallSeconds = time.Since(began).Seconds()
	p.ExitCode = h.Exit()
	p.Reaped = p.ExitCode != nil
	return p, w.Err
}
func parseMetrics(raw []byte) Metrics {
	out := Metrics{Scope: "Darwin /usr/bin/time -l worker child process, including startup; not model or GPU"}
	rssN, timeN := 0, 0
	for _, line := range strings.Split(string(raw), "\n") {
		f := strings.Fields(line)
		if len(f) == 5 && f[1] == "maximum" && f[2] == "resident" && f[3] == "set" && f[4] == "size" {
			n, e := strconv.ParseInt(f[0], 10, 64)
			if e != nil || n <= 0 {
				return out
			}
			out.MaximumRSSBytes = &n
			rssN++
		}
		if len(f) == 6 && f[1] == "real" && f[3] == "user" && f[5] == "sys" {
			var vals [3]float64
			for i := range vals {
				n, e := strconv.ParseFloat(f[i*2], 64)
				if e != nil || math.IsNaN(n) || math.IsInf(n, 0) || n < 0 {
					return out
				}
				vals[i] = n
			}
			out.RealSeconds = &vals[0]
			out.UserSeconds = &vals[1]
			out.SystemSeconds = &vals[2]
			timeN++
		}
	}
	out.Valid = rssN == 1 && timeN == 1
	return out
}
func artifact(file, name string, cap int) ([]byte, Artifact, error) {
	raw, _, e := regularRead(file, cap)
	a := Artifact{Name: name}
	if e != nil {
		return nil, a, failure("extratwo_outside_worker_artifacts_invalid")
	}
	a.Bytes = len(raw)
	a.SHA = sum(raw)
	a.ReadValid = true
	return raw, a, nil
}
func run(configFile, configSHA, outPath string) (r Receipt, retErr error) {
	r = Receipt{Schema: "riido-extra-two-outside-receipt-v1", State: "failed", ConfigSHA: configSHA,
		IncompleteActualCountsUnknown: true,
		Controls:                      "GOMAXPROCS1; GOMEMLIMIT256MiB soft; process-group wall deadline60s within300s batch ceiling; post-run Darwin maxRSS<=256MiB"}
	if !path(outPath) || !path(configFile) || !protocol.SHA(configSHA) || !canonicalParent(outPath) {
		return r, failure("extratwo_outside_config_invalid")
	}
	// This fresh directory/tombstone is invocation-scoped; it is not a global
	// guarantee against an operator supplying a different config/attempt.
	if os.Mkdir(outPath, 0700) != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	if syncDirectory(outPath) != nil || syncDirectory(filepath.Dir(outPath)) != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	root, e := os.OpenRoot(outPath)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	defer root.Close()
	tomb, e := root.OpenFile("RESERVED.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	if e = write(tomb, struct {
		Schema    string `json:"schema"`
		ConfigSHA string `json:"config_sha256"`
		Started   bool   `json:"started"`
	}{
		"riido-extra-two-outside-reservation-v1", configSHA, false}, 4096); e != nil {
		tomb.Close()
		return r, e
	}
	tomb.Close()
	if syncDirectory(outPath) != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	report, e := root.OpenFile("result.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	defer func() {
		if retErr != nil {
			r.StructuralCompletion = false
			s := fixedCode(retErr)
			r.FailureCode = &s
			r.State = "failed"
			r.ActualObservationCount = nil
			r.IncompleteActualCountsUnknown = true
		}
		we := write(report, r, receiptCap)
		ce := report.Close()
		de := syncDirectory(outPath)
		if we != nil || ce != nil || de != nil {
			e := failure("extratwo_outside_output_invalid")
			retErr = e
			r.State = "failed"
			s := fixedCode(e)
			r.FailureCode = &s
			r.StructuralCompletion = false
			r.ActualObservationCount = nil
			r.IncompleteActualCountsUnknown = true
		}
	}()
	c, e := readConfig(configFile, configSHA, outPath)
	if e != nil {
		return r, e
	}
	r.Pins, e = verifyAll(c)
	if e != nil {
		return r, e
	}
	r.EvidenceHashesVerified = true
	r.Family = c.Family
	if !canonicalParent(c.Attempt) {
		return r, failure("extratwo_outside_path_invalid")
	}
	if build(c.Worker.Path, c.WorkerModule, c.WorkerPackage) != nil ||
		build(c.Self.Path, "example.com/riido-extra-two-outside", "example.com/riido-extra-two-outside") != nil {
		return r, failure("extratwo_outside_build_invalid")
	}
	r.StaticBuildInfoValid = true
	planFacts, e := bindInputs(c)
	r.PlanBindings = planFacts
	if e != nil {
		return r, e
	}
	if _, e = os.Lstat(c.Attempt); !os.IsNotExist(e) {
		return r, failure("extratwo_outside_path_invalid")
	}
	stdout, e := root.OpenFile("stdout.private.bin", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	defer stdout.Close()
	stderr, e := root.OpenFile("stderr-time.private.bin", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	defer stderr.Close()
	startFile, e := root.OpenFile("START-RESERVED.json", os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	if e = write(startFile, struct {
		ConfigSHA     string `json:"config_sha256"`
		StartAttempts int    `json:"start_attempts"`
		WorkerSHA     string `json:"worker_sha256"`
		PlanSHA       string `json:"plan_sha256"`
	}{
		configSHA, 1, c.Worker.SHA, c.Plan.SHA}, 4096); e != nil {
		startFile.Close()
		return r, e
	}
	startFile.Close()
	if stdout.Sync() != nil || stderr.Sync() != nil || report.Sync() != nil || syncDirectory(outPath) != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	// Repeat the pinned regular-file byte checks immediately before Start.
	if _, e = verifyAll(c); e != nil {
		return r, e
	}
	if _, e = bindInputs(c); e != nil {
		return r, e
	}
	capSignal := make(chan struct{}, 2)
	o := capture{file: stdout, cap: stdoutCap, signal: capSignal}
	d := capture{file: stderr, cap: stderrCap, signal: capSignal}
	cmd := exec.Command(c.Time.Path, "-l", c.Worker.Path, "--out", c.Attempt, "--fixtures", c.Fixtures.Path, "--fixtures-sha256", c.Fixtures.SHA, "--wants", c.Wants.Path, "--wants-sha256", c.Wants.SHA, "--plan-sha256", c.Plan.SHA)
	cmd.Env = []string{"PATH=/usr/bin:/bin", "LANG=C", "LC_ALL=C", "GOMAXPROCS=1", "GOMEMLIMIT=268435456", "GODEBUG=panicnil=0"}
	cmd.Dir = filepath.Dir(c.Worker.Path)
	cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}
	cmd.WaitDelay = 2 * time.Second // bound pipe-copy delay after the process exits
	cmd.Stdout = &o
	cmd.Stderr = &d
	timer := time.NewTimer(time.Duration(c.Wall) * time.Second)
	var waitErr error
	r.Process, waitErr = startWait(processHooks{Start: cmd.Start, Wait: cmd.Wait,
		Kill: func() bool { return syscall.Kill(-cmd.Process.Pid, syscall.SIGKILL) == nil },
		Exit: func() *int {
			if cmd.ProcessState == nil {
				return nil
			}
			n := cmd.ProcessState.ExitCode()
			return &n
		}}, timer.C, capSignal)
	timer.Stop()
	if !r.Process.StartSucceeded {
		_, pe := verifyAll(c)
		r.SourceBindingsStillMatch = pe == nil
		_, be := bindInputs(c)
		r.PlanBindingsStillMatch = be == nil
		return r, waitErr
	}
	r.CaptureStdoutBytes = o.n
	r.CaptureStderrBytes = d.n
	if o.exceeded || d.exceeded {
		r.Process.CaptureCapExceeded = true
	}
	// Command.Wait has finished both capture-copy goroutines; files/counters
	// are now read sequentially. Each writer owns its stream; no shared locks.
	if stdout.Sync() != nil || stderr.Sync() != nil {
		return r, failure("extratwo_outside_output_invalid")
	}
	savedOut, a, e := artifact(filepath.Join(outPath, "stdout.private.bin"), "stdout.private.bin", stdoutCap)
	r.Artifacts = append(r.Artifacts, a)
	if e != nil {
		return r, e
	}
	savedErr, a, e := artifact(filepath.Join(outPath, "stderr-time.private.bin"), "stderr-time.private.bin", stderrCap)
	r.Artifacts = append(r.Artifacts, a)
	if e != nil {
		return r, e
	}
	r.Metrics = parseMetrics(savedErr)
	attemptInfo, attemptErr := os.Lstat(c.Attempt)
	attemptReal, attemptRealErr := filepath.EvalSymlinks(c.Attempt)
	attemptDirectoryValid := attemptErr == nil && attemptInfo.IsDir() && attemptInfo.Mode()&os.ModeSymlink == 0 && attemptRealErr == nil && attemptReal == c.Attempt
	if !attemptDirectoryValid {
		_, pe := verifyAll(c)
		r.SourceBindingsStillMatch = pe == nil
		_, be := bindInputs(c)
		r.PlanBindingsStillMatch = be == nil
		if r.Process.DeadlineExceeded {
			return r, failure("extratwo_outside_deadline")
		}
		if r.Process.CaptureCapExceeded {
			return r, failure("extratwo_outside_capture_cap")
		}
		if waitErr != nil {
			return r, failure("extratwo_outside_wait_failed")
		}
		return r, failure("extratwo_outside_worker_artifacts_invalid")
	}
	// Preserve parseable saved prefixes on failure. They are not complete
	// actual call totals, qualification, or an acknowledgement of success.
	rawJournal, a, je := artifact(filepath.Join(c.Attempt, "journal.jsonl"), "worker/journal.jsonl", protocol.JournalCap)
	r.Artifacts = append(r.Artifacts, a)
	if je == nil {
		j, e := protocol.ReadJournal(rawJournal, c.Family)
		r.Journal = &j
		r.JournalValid = e == nil
	}
	rawResult, a, re := artifact(filepath.Join(c.Attempt, "results.json"), "worker/results.json", protocol.FinalCap)
	r.Artifacts = append(r.Artifacts, a)
	var savedResult *protocol.Result
	if re == nil {
		v, e := protocol.ReadResult(rawResult, c.Family, c.Plan.SHA)
		if e == nil {
			savedResult = &v
			s := protocol.Summarize(v)
			r.WorkerResult = &s
		}
	}
	r.WorkerStdoutMatches = re == nil && bytes.Equal(rawResult, savedOut)
	rawReservation, a, rve := artifact(filepath.Join(c.Attempt, "reservation.json"), "worker/reservation.json", protocol.ReservationCap)
	r.Artifacts = append(r.Artifacts, a)
	_, pe := verifyAll(c)
	r.SourceBindingsStillMatch = pe == nil
	_, pbe := bindInputs(c)
	r.PlanBindingsStillMatch = pbe == nil
	if r.Process.DeadlineExceeded {
		return r, failure("extratwo_outside_deadline")
	}
	if r.Process.CaptureCapExceeded {
		return r, failure("extratwo_outside_capture_cap")
	}
	if waitErr != nil || r.Process.ExitCode == nil || *r.Process.ExitCode != 0 {
		return r, failure("extratwo_outside_wait_failed")
	}
	if pe != nil || pbe != nil {
		return r, failure("extratwo_outside_bindings_changed")
	}
	if !attemptDirectoryValid || je != nil || re != nil || rve != nil || protocol.ReadReservation(rawReservation, c.Family, c.Plan.SHA) != nil || len(rawJournal)+len(rawResult)+len(rawReservation) > protocol.CombinedCap {
		return r, failure("extratwo_outside_worker_artifacts_invalid")
	}
	if !r.JournalValid || r.Journal == nil || savedResult == nil || !r.WorkerStdoutMatches ||
		!protocol.Complete(*savedResult, *r.Journal) {
		return r, failure("extratwo_outside_worker_protocol_invalid")
	}
	if !r.Metrics.Valid || r.Process.WallSeconds > float64(c.Wall) {
		return r, failure("extratwo_outside_metrics_invalid")
	}
	if *r.Metrics.MaximumRSSBytes > heapLimit {
		return r, failure("extratwo_outside_rss_limit")
	}
	r.StructuralCompletion = true
	r.State = "complete"
	n := protocol.Planned
	r.ActualObservationCount = &n
	r.IncompleteActualCountsUnknown = false
	return r, nil
}
func cliArguments(args []string) (file, digest, out string, ok bool) {
	if len(args) != 6 {
		return
	}
	keys := [3]string{"--config", "--config-sha256", "--out"}
	var values [3]string
	var seen [3]bool
	for i := 0; i < len(args); i += 2 {
		n := -1
		for j, k := range keys {
			if args[i] == k {
				n = j
				break
			}
		}
		if n < 0 || seen[n] || args[i+1] == "" {
			return
		}
		seen[n] = true
		values[n] = args[i+1]
	}
	if !path(values[0]) || !protocol.SHA(values[1]) || !path(values[2]) {
		return
	}
	return values[0], values[1], values[2], true
}
func main() {
	runtime.GOMAXPROCS(1)
	debug.SetMemoryLimit(heapLimit)
	config, sha, out, ok := cliArguments(os.Args[1:])
	if !ok {
		fmt.Fprintln(os.Stderr, "extratwo_outside_arguments_invalid")
		os.Exit(2)
	}
	r, e := run(config, sha, out)
	if e != nil {
		s := fixedCode(e)
		r.FailureCode = &s
	}
	summary := struct {
		Schema        string  `json:"schema"`
		State         string  `json:"state"`
		FailureCode   *string `json:"failure_code"`
		StartAttempts int     `json:"start_attempts"`
		WaitAttempts  int     `json:"wait_attempts"`
		Actual        *int    `json:"actual_completed_observation_count"`
	}{"riido-extra-two-outside-summary-v1", r.State, r.FailureCode, r.Process.StartAttempts, r.Process.WaitAttempts, r.ActualObservationCount}
	raw, me := json.Marshal(summary)
	if me != nil {
		os.Exit(1)
	}
	if _, we := os.Stdout.Write(append(raw, '\n')); we != nil {
		os.Exit(1)
	}
	if e != nil {
		os.Exit(1)
	}
}
