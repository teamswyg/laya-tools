func digest(raw []byte) string { h := sha256.Sum256(raw); return hex.EncodeToString(h[:]) }
func digestOK(s string) bool {
	if len(s) != 64 {
		return false
	}
	for _, c := range s {
		if !(c >= '0' && c <= '9' || c >= 'a' && c <= 'f') {
			return false
		}
	}
	return true
}
func pinned(path, sha string, exact int) ([]byte, error) {
	if path == "" || !digestOK(sha) {
		return nil, failure("validation_arguments")
	}
	before, e := os.Lstat(path)
	if e != nil || !before.Mode().IsRegular() {
		return nil, failure("input_regular")
	}
	f, e := os.Open(path)
	if e != nil {
		return nil, failure("input_open")
	}
	defer f.Close()
	st, e := f.Stat()
	if e != nil || !st.Mode().IsRegular() || !os.SameFile(before, st) {
		return nil, failure("input_regular")
	}
	if st.Size() < 1 || st.Size() > maxInputBytes || exact > 0 && st.Size() != int64(exact) {
		return nil, failure("input_bounds")
	}
	raw, e := io.ReadAll(io.LimitReader(f, maxInputBytes+1))
	if e != nil || len(raw) != int(st.Size()) || len(raw) > maxInputBytes {
		return nil, failure("input_read")
	}
	if digest(raw) != sha {
		return nil, failure("input_pin")
	}
	return raw, nil
}

// Validates Unicode scalars before Go's decoder can replace an unpaired escape.
// Derived from the project's Apache-2.0 shortclaimdata reader, source pinned in
// the preparation ledger. json.Valid is a precondition for the indexed slices.
func scalarEscapes(raw []byte) bool {
	for i := 0; i < len(raw); i++ {
		if raw[i] != '"' {
			continue
		}
		for i++; i < len(raw) && raw[i] != '"'; i++ {
			if raw[i] != '\\' {
				continue
			}
			i++
			if raw[i] != 'u' {
				continue
			}
			u := hex4(raw[i+1 : i+5])
			i += 4
			if u >= 0xdc00 && u <= 0xdfff {
				return false
			}
			if u >= 0xd800 && u <= 0xdbff {
				if i+6 >= len(raw) || raw[i+1] != '\\' || raw[i+2] != 'u' {
					return false
				}
				lo := hex4(raw[i+3 : i+7])
				if lo < 0xdc00 || lo > 0xdfff {
					return false
				}
				i += 6
			}
		}
	}
	return true
}
func hex4(b []byte) uint16 {
	var v uint16
	for _, c := range b {
		v <<= 4
		switch {
		case c >= '0' && c <= '9':
			v |= uint16(c - '0')
		case c >= 'a' && c <= 'f':
			v |= uint16(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			v |= uint16(c - 'A' + 10)
		}
	}
	return v
}

// The complete JSON is checked for decoded duplicate/case-alias keys and bounded
// shape before typed decoding. Arrays/slices track keys; no label-keyed maps.
func checkedJSON(raw []byte, dst any, exactFields int) error {
	if len(raw) < 1 || len(raw) > maxInputBytes || !utf8.Valid(raw) || !json.Valid(raw) || !scalarEscapes(raw) {
		return failure("input_json_unicode_bounds")
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.UseNumber()
	nodes := 0
	fields, e := walk(d, 0, &nodes)
	if e != nil {
		return e
	}
	if exactFields >= 0 && fields != exactFields {
		return failure("input_fields")
	}
	if _, e = d.Token(); !errors.Is(e, io.EOF) {
		return failure("input_trailing")
	}
	if json.Unmarshal(raw, dst) != nil {
		return failure("input_types")
	}
	return nil
}
func walk(d *json.Decoder, depth int, nodes *int) (int, error) {
	*nodes++
	if depth > 32 || *nodes > 65536 {
		return 0, failure("input_shape_bounds")
	}
	t, e := d.Token()
	if e != nil {
		return 0, failure("input_json")
	}
	delim, ok := t.(json.Delim)
	if !ok {
		return -1, nil
	}
	if delim == '{' {
		keys := make([]string, 0, 16)
		for d.More() {
			if len(keys) >= 128 {
				return 0, failure("input_shape_bounds")
			}
			kt, e := d.Token()
			key, ok := kt.(string)
			if e != nil || !ok {
				return 0, failure("input_json")
			}
			for _, old := range keys {
				if strings.EqualFold(old, key) {
					return 0, failure("input_duplicate")
				}
			}
			keys = append(keys, key)
			if _, e = walk(d, depth+1, nodes); e != nil {
				return 0, e
			}
		}
		end, e := d.Token()
		if e != nil || end != json.Delim('}') {
			return 0, failure("input_json")
		}
		return len(keys), nil
	}
	if delim == '[' {
		n := 0
		for d.More() {
			n++
			if n > 1024 {
				return 0, failure("input_shape_bounds")
			}
			if _, e = walk(d, depth+1, nodes); e != nil {
				return 0, e
			}
		}
		end, e := d.Token()
		if e != nil || end != json.Delim(']') {
			return 0, failure("input_json")
		}
		return -1, nil
	}
	return 0, failure("input_json")
}
func exactObject(raw []byte, keys []string) error {
	var fields []struct {
		key   string
		value json.RawMessage
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	t, e := d.Token()
	if e != nil || t != json.Delim('{') {
		return failure("row_fields")
	}
	for d.More() {
		kt, e := d.Token()
		k, ok := kt.(string)
		if e != nil || !ok {
			return failure("row_fields")
		}
		var value json.RawMessage
		if d.Decode(&value) != nil {
			return failure("row_fields")
		}
		fields = append(fields, struct {
			key   string
			value json.RawMessage
		}{k, value})
	}
	if len(fields) != len(keys) {
		return failure("row_fields")
	}
	for _, k := range keys {
		found := false
		for _, f := range fields {
			if f.key == k {
				found = true
				break
			}
		}
		if !found {
			return failure("row_fields")
		}
	}
	return nil
}
func rowDecode(raw []byte) (row, error) {
	var r row
	if len(raw) > maxRowBytes {
		return r, failure("row_bounds")
	}
	if e := checkedJSON(raw, &r, 11); e != nil {
		return row{}, e
	}
	if e := exactObject(raw, []string{"schema", "stable_id", "request", "candidates", "role", "whole_group", "source_family", "source_revision", "finite_scope", "text_revision", "feature_policy"}); e != nil {
		return row{}, e
	}
	var partial struct {
		Candidates []json.RawMessage `json:"candidates"`
		Scope      json.RawMessage   `json:"finite_scope"`
	}
	if json.Unmarshal(raw, &partial) != nil {
		return row{}, failure("row_fields")
	}
	for _, c := range partial.Candidates {
		if e := exactObject(c, []string{"metadata_id", "text", "label", "sample_weight"}); e != nil {
			return row{}, e
		}
	}
	if e := exactObject(partial.Scope, []string{"input_count", "candidate_observations", "description", "all_input_guarantee", "unseen_source_guarantee"}); e != nil {
		return row{}, e
	}
	if r.Schema != "riido-finite-development-request-row-v1" || r.Role != "development_train" || r.Policy != "request_and_candidates_text_only" || len(r.Candidates) < 1 || len(r.Candidates) > 8 || r.Scope.Inputs < 1 || r.Scope.Inputs > 4096 || r.Scope.Observations != r.Scope.Inputs*len(r.Candidates) || r.Scope.All == nil || *r.Scope.All || r.Scope.Unseen == nil || *r.Scope.Unseen {
		return row{}, failure("row_metadata_scope")
	}
	if !identifier(r.ID, 64) || !identifier(r.Family, 128) || !identifier(r.TextRevision, 128) || !revision(r.Revision) || r.Group < 1 || r.Group > 65535 || !textOK(r.Request) || !textOK(r.Scope.Description) {
		return row{}, failure("row_metadata_text")
	}
	for _, c := range r.Candidates {
		if !identifier(c.ID, 64) || !textOK(c.Text) || c.Label == nil || c.Weight != 1 {
			return row{}, failure("row_candidate")
		}
	}
	return r, nil
}
func textOK(s string) bool {
	return len(s) > 0 && len(s) <= 512 && strings.TrimSpace(s) != "" && !strings.ContainsRune(s, 0)
}
func identifier(s string, max int) bool {
	if len(s) < 1 || len(s) > max {
		return false
	}
	for i, c := range []byte(s) {
		if !(c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9') && (i == 0 || c != '.' && c != '_' && c != '-') {
			return false
		}
	}
	return true
}
func revision(s string) bool {
	if len(s) != 40 {
		return false
	}
	for _, c := range s {
		if !(c >= '0' && c <= '9' || c >= 'a' && c <= 'f') {
			return false
		}
	}
	return true
}
func intsEqual(a, b []int) bool {
	if a == nil || b == nil || len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}
func positions(a []int, n int) bool {
	if a == nil {
		return false
	}
	prev := -1
	for _, i := range a {
		if i < 0 || i >= n || i <= prev {
			return false
		}
		prev = i
	}
	return true
}

func stringsEqual(a, b []string) bool {
	if a == nil || b == nil || len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}
func literalEqual(a, b json.RawMessage) bool {
	var x, y bytes.Buffer
	return json.Compact(&x, a) == nil && json.Compact(&y, b) == nil && bytes.Equal(x.Bytes(), y.Bytes())
}
func isTrue(v *bool) bool  { return v != nil && *v }
func isFalse(v *bool) bool { return v != nil && !*v }
func loadedMatches(v loadedView, r row) bool {
	if !v.Valid || !v.UnusedZero || v.ID != r.ID || v.Request != r.Request || v.Schema != "riido-short-behavior-claim-v1" || v.Provenance != "finite-development-v1" || v.Role != r.Role || v.Policy != r.Policy || v.Group != r.Group || v.Family != r.Family || v.Revision != r.Revision || v.TextRevision != r.TextRevision || v.Inputs != r.Scope.Inputs || v.Observations != r.Scope.Observations || v.ScopeDescription != r.Scope.Description || v.All != *r.Scope.All || v.Unseen != *r.Scope.Unseen || v.Count != len(r.Candidates) || v.SupervisionCount != len(r.Candidates) {
		return false
	}
	for j, c := range r.Candidates {
		x := v.Candidates[j]
		if x.ID != c.ID || x.Text != c.Text || x.Label != *c.Label || x.Weight != c.Weight {
			return false
		}
	}
	for j := len(r.Candidates); j < len(v.Candidates); j++ {
		if v.Candidates[j] != (loadedCandidate{}) {
			return false
		}
	}
	return true
}

// Production validates every pin before invoking a reader. Synthetic controls
// exercise validateBound only and never claim actual production reader calls.
func validateBundle(b bundle, p pins, reader func([]byte) (loadedView, error), out *summary) error {
	if reader == nil || !digestOK(p.Data) || p.Adoption != adoptionSHA {
		return failure("validation_arguments")
	}
	if len(b.Data) < 1 || len(b.Data) > maxInputBytes || digest(b.Data) != p.Data || len(b.Adoption) != adoptionBytes || digest(b.Adoption) != adoptionSHA {
		return failure("prospective_input_pin")
	}
	for _, x := range []struct {
		raw  []byte
		size int
		sha  string
	}{{b.Previous, previousBytes, previousSHA}, {b.Fixtures, fixtureBytes, fixtureSHA}, {b.Wants, wantsBytes, wantsSHA}, {b.Comparison, comparisonBytes, comparisonSHA}, {b.Results, resultBytes, resultSHA}} {
		if len(x.raw) != x.size || digest(x.raw) != x.sha {
			return failure("immutable_input_pin")
		}
	}
	return validateBound(b, reader, out)
}

func validateBound(b bundle, reader func([]byte) (loadedView, error), out *summary) error {
	if reader == nil || out == nil || *out != initialSummary() {
		return failure("validation_arguments")
	}
	if len(b.Data) < 1 || len(b.Data) > maxInputBytes || len(b.Previous) < 1 || b.Previous[len(b.Previous)-1] != '\n' || len(b.Data) <= len(b.Previous) || b.Data[len(b.Data)-1] != '\n' || !bytes.HasPrefix(b.Data, b.Previous) {
		return failure("sixteen_prefix")
	}
	out.Prefix = true
	var f fixtures
	var w wantedDocument
	var a adoption
	var proof comparison
	for _, x := range []struct {
		raw []byte
		dst any
	}{{b.Fixtures, &f}, {b.Wants, &w}, {b.Adoption, &a}, {b.Comparison, &proof}} {
		if e := checkedJSON(x.raw, x.dst, -1); e != nil {
			return e
		}
	}
	if f.Schema != "riido-next-five-fixtures-v1" || len(f.Records) != 23 || w.Schema != "riido-next-five-target-Wants-v1" || len(w.Requests) != 5 || len(w.Wants) != 23 {
		return failure("fixtures_Wanted_shape")
	}
	if a.Schema != "riido-next-five-root-finite-development-qualification-v2" || a.State != "Root_adopted_five_finite_development_requests" || a.Role != "development_train" || a.Weight != 1 || a.FixtureSHA != fixtureSHA || a.WantsSHA != wantsSHA || a.ResultSHA != resultSHA || a.ComparisonSHA != comparisonSHA || a.PlanSHA != planSHA || a.Inputs != 23 || a.Observations != 69 || a.SelectedObservations != 65 || a.Positive != 5 || a.Negative != 9 || a.Labels != 14 || a.Parents != 5 || len(a.Requests) != 5 || !isTrue(a.UnknownNeverFalse) || !isTrue(a.NegativeUnknown) || !isTrue(a.ExcludedBaseline) || a.Combined != (total{Requests: 21, Labels: 61, Positive: 21, Negative: 40, Inputs: 103, Observations: 305, Original: 305, Selected: 301}) {
		return failure("Root_adoption_header")
	}
	if a.Prior.Bytes != previousBytes || a.Prior.SHA != previousSHA || a.Prior.Rows != 16 || a.Prior.Labels != 47 || a.Prior.Positive != 16 || a.Prior.Negative != 31 || a.Prior.Inputs != 80 || a.Prior.Observations != 236 {
		return failure("Root_prior_header")
	}
	if proof.Schema != "riido-next-five-finite-saved-comparison-v1" || proof.FixtureSHA != fixtureSHA || proof.WantsSHA != wantsSHA || proof.ResultSHA != resultSHA || proof.PlanSHA != planSHA || !isFalse(proof.Labels) || !isFalse(proof.Qualified) || !intsEqual(proof.Counts, []int{69, 42, 21, 6}) || !intsEqual(proof.Predicates, []int{185, 32, 11}) || len(proof.Rows) != 69 || len(proof.Candidates) != 15 {
		return failure("comparison_header")
	}
	var predCounts [3]int
	var rowCounts [3]int
	for i, id := range requestIDs {
		sr := w.Requests[i]
		ar := a.Requests[i]
		fam, rev := sourceFamily(i)
		if sr.ID != id || sr.Ordinal != draftOrdinals[i] || sr.Family != fam || sr.Count != fixtureCounts[i] || len(sr.Captions) != 3 || ar.ID != id || ar.Draft != draftOrdinals[i] || ar.Request != sr.Request || ar.Family != fam || ar.Revision != rev || ar.TextRevision != textRevision || ar.Count != fixtureCounts[i] || ar.Group != groups[i] || ar.Observations != fixtureCounts[i]*selectedCounts[i] || len(ar.Selected) != selectedCounts[i] {
			return failure("request_identity")
		}
		for j := 0; j < 3; j++ {
			if sr.Captions[j].ID != sourceCandidateIDs[j] || !textOK(sr.Captions[j].Text) {
				return failure("Wanted_caption_identity")
			}
		}
		for k := 0; k < fixtureCounts[i]; k++ {
			n := fixtureStarts[i] + k
			fr := f.Records[n]
			wr := w.Wants[n]
			if fr.Ordinal != n || fr.RequestIndex != i || fr.Draft != draftOrdinals[i] || fr.CaseIndex != k || fr.RequestID != id || fr.CaseID == "" || wr.Ordinal != n || wr.RequestID != id || wr.CaseID != fr.CaseID || !literalEqual(fr.OriginalWant, wr.Want) || !stringsEqual(wr.Order, sourceCandidateIDs[:]) {
				return failure("fixture_Wanted_binding")
			}
			for j := 0; j < 3; j++ {
				or := proof.Rows[n*3+j]
				if !intsEqual(or.Position, []int{n*3 + j, i, k, j}) || or.RequestID != id || or.CaseID != fr.CaseID || len(or.Predicates) == 0 || len(or.Counts) != 3 {
					return failure("saved_row_identity")
				}
				var c [3]int
				for _, p := range or.Predicates {
					if p.Field == "" || len(p.Want) == 0 {
						return failure("saved_predicate")
					}
					switch p.State {
					case "T":
						c[0]++
					case "F":
						c[1]++
					case "U":
						c[2]++
					default:
						return failure("saved_predicate")
					}
				}
				if !intsEqual(or.Counts, c[:]) {
					return failure("saved_predicate_counts")
				}
				state := "satisfied"
				s := 0
				if c[1] > 0 {
					state = "unsatisfied"
					s = 1
				} else if c[2] > 0 {
					state = "unknown"
					s = 2
				}
				if or.State != state {
					return failure("saved_row_state")
				}
				rowCounts[s]++
				for p := 0; p < 3; p++ {
					predCounts[p] += c[p]
				}
			}
		}
		for j := 0; j < 3; j++ {
			ev := proof.Candidates[i*3+j]
			if !intsEqual(ev.Position, []int{i, j}) || ev.RequestID != id || ev.ID != sourceCandidateIDs[j] || len(ev.Vector) != fixtureCounts[i] || len(ev.Ordinals) != fixtureCounts[i] || !positions(ev.Known, fixtureCounts[i]) || !positions(ev.Unknown, fixtureCounts[i]) {
				return failure("candidate_evidence_identity")
			}
			known := []int{}
			unknown := []int{}
			for k := 0; k < fixtureCounts[i]; k++ {
				ordinal := (fixtureStarts[i]+k)*3 + j
				or := proof.Rows[ordinal]
				if ev.Ordinals[k] != ordinal || ev.Vector[k] != or.State {
					return failure("candidate_vector")
				}
				if or.Counts[1] > 0 {
					known = append(known, k)
				}
				if or.Counts[2] > 0 {
					unknown = append(unknown, k)
				}
			}
			state := "satisfied"
			if len(known) > 0 {
				state = "unsatisfied"
			} else if len(unknown) > 0 {
				state = "unknown"
			}
			if ev.State != state || !intsEqual(ev.Known, known) || !intsEqual(ev.Unknown, unknown) {
				return failure("candidate_counterexample_unknown")
			}
		}
		for j, ac := range ar.Selected {
			si := j
			if i == 1 {
				si = j + 1
			}
			ev := proof.Candidates[i*3+si]
			if ac.Index == nil || *ac.Index != si || ac.ID != sourceCandidateIDs[si] || ac.Text != sr.Captions[si].Text || ac.Label == nil || *ac.Label != (si == 1) || ac.Role != "development_train" || ac.Weight != 1 || !stringsEqual(ac.Vector, ev.Vector) || !intsEqual(ac.Ordinals, ev.Ordinals) || !intsEqual(ac.Known, ev.Known) || !intsEqual(ac.Unknown, ev.Unknown) {
				return failure("selected_source_correspondence")
			}
			if *ac.Label {
				if ev.State != "satisfied" || len(ev.Known) != 0 || len(ev.Unknown) != 0 {
					return failure("positive_evidence")
				}
			} else if ev.State != "unsatisfied" || len(ev.Known) == 0 {
				return failure("negative_known_counterexample")
			}
		}
		if i == 1 {
			ev := proof.Candidates[i*3]
			if len(ar.Excluded) != 1 {
				return failure("unknown_writer_exclusion")
			}
			x := ar.Excluded[0]
			if x.Index == nil || *x.Index != 0 || x.ID != sourceCandidateIDs[0] || x.Reason == "" || ev.State != "unknown" || len(ev.Known) != 0 || !intsEqual(x.Unknown, ev.Unknown) || !intsEqual(ev.Unknown, []int{0, 1, 2, 3}) {
				return failure("unknown_writer_exclusion")
			}
		} else if ar.Excluded == nil || len(ar.Excluded) != 0 {
			return failure("unexpected_exclusion")
		}
	}
	if predCounts != [3]int{185, 32, 11} || rowCounts != [3]int{42, 21, 6} {
		return failure("saved_aggregate_counts")
	}
	out.Adoption = true
	out.UnknownPreserved = true
	out.BaselineExcluded = true
	oldIDs := [16]string{"next60-pflag-native-ipnet", "next60-mapstructure-native-or", "next60-humanize-precise-ftoa", "next60-humanize-strict-comma", "next60-mapstructure-key-collision", "next60-pflag-map-snapshot", "next60-humanize-fractional-bytes", "next60-go53-pflag-annotation-ownership", "next60-go53-pflag-bounded-count", "next60-go53-pflag-single-assignment", "next60-go53-pflag-sensitive-default", "next60-go53-pflag-deferred-function", "next60-go53-pflag-text-error-value", "next60-go53-mapstructure-hooks-errors-is", "next60-go53-mapstructure-remain-conflict", "next60-go53-mapstructure-tag-precedence"}
	var allIDs [21]string
	scan := bufio.NewScanner(bytes.NewReader(b.Data))
	scan.Buffer(make([]byte, 4096), maxRowBytes+1)
	for scan.Scan() {
		i := out.Requests
		if i >= 21 || len(scan.Bytes()) == 0 {
			return failure("row_count")
		}
		r, e := rowDecode(scan.Bytes())
		if e != nil {
			return e
		}
		for k := 0; k < i; k++ {
			if allIDs[k] == r.ID {
				return failure("duplicate_stable_id")
			}
		}
		allIDs[i] = r.ID
		if i < 16 {
			if r.ID != oldIDs[i] {
				return failure("previous_order")
			}
		} else {
			k := i - 16
			ar := a.Requests[k]
			sr := w.Requests[k]
			if r.ID != ar.ID || r.Request != sr.Request || r.Group != ar.Group || r.Family != ar.Family || r.Revision != ar.Revision || r.TextRevision != textRevision || r.Scope.Description != scopeDescription || r.Scope.Inputs != fixtureCounts[k] || r.Scope.Observations != ar.Observations || len(r.Candidates) != selectedCounts[k] {
				return failure("new_row_correspondence")
			}
			for j, rc := range r.Candidates {
				ac := ar.Selected[j]
				si := *ac.Index
				if rc.ID != ac.ID || rc.Text != sr.Captions[si].Text || *rc.Label != *ac.Label || rc.Weight != ac.Weight {
					return failure("new_candidate_correspondence")
				}
			}
		}
		out.ReaderAttempts++
		v, e := reader(scan.Bytes())
		if e != nil {
			return failure("project_reader_validation")
		}
		out.ReaderLoads++
		if !loadedMatches(v, r) {
			return failure("project_reader_value_correspondence")
		}
		for j, rc := range r.Candidates {
			out.Labels++
			if *rc.Label {
				out.Positive++
				if j == 1 {
					out.PositiveIndex1++
				}
			} else {
				out.Negative++
			}
		}
		out.Inputs += r.Scope.Inputs
		out.Observations += r.Scope.Observations
		out.Requests++
		if i == 15 && (out.Labels != 47 || out.Positive != 16 || out.Negative != 31 || out.Inputs != 80 || out.Observations != 236 || out.PositiveIndex1 != 14) {
			return failure("previous_counts")
		}
	}
	if scan.Err() != nil || out.Requests != 21 || out.ReaderLoads != 21 || out.Labels != 61 || out.Positive != 21 || out.Negative != 40 || out.Inputs != 103 || out.Observations != 301 || out.PositiveIndex1 != 18 {
		return failure("final_counts")
	}
	out.OriginalObservations = 305
	out.ReaderExact = true
	out.Texts = true
	out.State = "complete_validation_only"
	return nil
}

// Output reservation is exclusive. The success result is acknowledged only
// after file Sync and Close; no power-loss or storage-hardware claim follows.
func writeOutput(path string, out summary) error {
	if path == "" || !filepath.IsAbs(path) || filepath.Clean(path) != path {
		return failure("output_arguments")
	}
	raw, e := json.Marshal(out)
	if e != nil || len(raw)+1 > maxOutputBytes {
		return failure("output_bounds")
	}
	raw = append(raw, '\n')
	f, e := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		return failure("output_reserve")
	}
	n, we := f.Write(raw)
	se := f.Sync()
	ce := f.Close()
	if we != nil || n != len(raw) {
		return failure("output_write")
	}
	if se != nil {
		return failure("output_sync")
	}
	if ce != nil {
		return failure("output_close")
	}
	return nil
}

var optionNames = [10]string{"--data", "--data-sha256", "--previous-data", "--fixtures", "--wants", "--adoption", "--adoption-sha256", "--finite-comparison", "--saved-results", "--output"}

func parseOptions(argv []string) ([10]string, error) {
	var v [10]string
	var seen [10]bool
	if len(argv) != 20 {
		return v, failure("validation_arguments")
	}
	for i := 0; i < len(argv); i += 2 {
		j := -1
		for k, n := range optionNames {
			if argv[i] == n {
				j = k
				break
			}
		}
		if j < 0 || seen[j] || argv[i+1] == "" {
			return v, failure("validation_arguments")
		}
		seen[j] = true
		v[j] = argv[i+1]
	}
	if !digestOK(v[1]) || v[6] != adoptionSHA || !filepath.IsAbs(v[9]) || filepath.Clean(v[9]) != v[9] {
		return v, failure("validation_arguments")
	}
	return v, nil
}
