func bp(v bool) *bool { return &v }

func encoded(t *testing.T, v any) []byte {
	t.Helper()
	b, e := json.Marshal(v)
	if e != nil {
		t.Fatal(e)
	}
	return b
}
func fakeView(raw []byte) (loadedView, error) {
	var r row
	if json.Unmarshal(raw, &r) != nil {
		return loadedView{}, failure("fake_reader")
	}
	v := loadedView{Valid: true, ID: r.ID, Request: r.Request, Schema: "riido-short-behavior-claim-v1", Provenance: "finite-development-v1", Role: r.Role, Policy: r.Policy, Group: r.Group, Family: r.Family, Revision: r.Revision, TextRevision: r.TextRevision, Inputs: r.Scope.Inputs, Observations: r.Scope.Observations, ScopeDescription: r.Scope.Description, All: *r.Scope.All, Unseen: *r.Scope.Unseen, Count: len(r.Candidates), SupervisionCount: len(r.Candidates), UnusedZero: true}
	for i, c := range r.Candidates {
		v.Candidates[i] = loadedCandidate{ID: c.ID, Text: c.Text, Label: *c.Label, Weight: c.Weight}
	}
	return v, nil
}

func check(b bundle, reader func([]byte) (loadedView, error)) (summary, error) {
	out := initialSummary()
	e := validateBound(b, reader, &out)
	return out, e
}
func changeRow(t *testing.T, b *bundle, index int, change func(*row)) {
	t.Helper()
	ls := bytes.Split(bytes.TrimSuffix(b.Data, []byte{'\n'}), []byte{'\n'})
	var r row
	if json.Unmarshal(ls[index], &r) != nil {
		t.Fatal("row")
	}
	change(&r)
	ls[index] = encoded(t, r)
	b.Data = append(bytes.Join(ls, []byte{'\n'}), '\n')
}
func changeAdoption(t *testing.T, b *bundle, change func(*adoption)) {
	t.Helper()
	var a adoption
	if json.Unmarshal(b.Adoption, &a) != nil {
		t.Fatal("adoption")
	}
	change(&a)
	b.Adoption = encoded(t, a)
}

func TestCLIAndExclusiveOutput(t *testing.T) {
	dir := t.TempDir()
	path := filepath.Join(dir, "result.json")
	args := []string{"--data", "data", "--data-sha256", strings.Repeat("a", 64), "--previous-data", "previous", "--fixtures", "fixtures", "--wants", "wants", "--adoption", "adoption", "--adoption-sha256", adoptionSHA, "--finite-comparison", "comparison", "--saved-results", "results", "--output", path}
	if _, e := parseOptions(args); e != nil {
		t.Fatal(e)
	}
	bad := append([]string(nil), args...)
	bad[2] = "--data"
	if _, e := parseOptions(bad); e == nil {
		t.Fatal("duplicate option")
	}
	bad = append([]string(nil), args...)
	bad[0] = "--data=x"
	if _, e := parseOptions(bad); e == nil {
		t.Fatal("equals accepted")
	}
	out := initialSummary()
	if writeOutput(path, out) != nil {
		t.Fatal("write")
	}
	before, e := os.ReadFile(path)
	if e != nil {
		t.Fatal(e)
	}
	if writeOutput(path, out) == nil {
		t.Fatal("overwrite")
	}
	after, e := os.ReadFile(path)
	if e != nil || !bytes.Equal(before, after) {
		t.Fatal("reserved output changed")
	}
	if writeOutput(filepath.Join(dir, "big.json"), summary{Scope: strings.Repeat("x", maxOutputBytes)}) == nil {
		t.Fatal("oversize output")
	}
}
func TestBoundedPinnedRegularFile(t *testing.T) {
	dir := t.TempDir()
	path := filepath.Join(dir, "input.json")
	raw := []byte(`{"x":true}`)
	if os.WriteFile(path, raw, 0600) != nil {
		t.Fatal("fixture")
	}
	if _, e := pinned(path, digest(raw), len(raw)); e != nil {
		t.Fatal(e)
	}
	if _, e := pinned(path, strings.Repeat("a", 64), len(raw)); e == nil {
		t.Fatal("wrong pin")
	}
	link := filepath.Join(dir, "link.json")
	if os.Symlink(path, link) != nil {
		t.Fatal("symlink")
	}
	if _, e := pinned(link, digest(raw), len(raw)); e == nil {
		t.Fatal("symlink")
	}
}
