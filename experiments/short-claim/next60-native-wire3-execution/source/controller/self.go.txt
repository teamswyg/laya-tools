package main

import (
	"encoding/json"
	"os"
	"path/filepath"
	"runtime/debug"
)

// This pins the observed ordinary go-run scratch executable. It does not prove
// fresh compilation from the admitted source, or create a retained-artifact
// exemption. Root separately binds source, toolchain and compiler commands.
const selfReadCap = 32 << 20

func observeSelf() (path string, s workerSeal, version, buildHash string, err error) {
	p, e := os.Executable()
	if e != nil {
		err = code("self_executable_unavailable")
		return
	}
	path, e = filepath.EvalSymlinks(p)
	if e != nil {
		err = code("self_path_unavailable")
		return
	}
	s, e = sealExecutable(path, selfReadCap, "")
	if e != nil {
		err = code("self_identity_unavailable")
		return
	}
	bi, ok := debug.ReadBuildInfo()
	if !ok || bi.GoVersion != "go1.27.1" || len(bi.Deps) > 128 || len(bi.Settings) > 128 {
		err = code("self_build_identity_rejected")
		return
	}
	b, e := json.Marshal(bi)
	if e != nil || len(b) > 32768 {
		err = code("self_build_metadata_bound")
		return
	}
	version, buildHash = bi.GoVersion, digest(b)
	return
}
