// All fixtures below are public deterministic synthetic bytes, never HF assets.
func header(kind byte, scale float32, n int, size int) []byte {
	b := make([]byte, size)
	copy(b, "RIIDOH01")
	binary.LittleEndian.PutUint32(b[8:], Dimension)
	b[12] = kind
	binary.LittleEndian.PutUint32(b[16:], math.Float32bits(scale))
	binary.LittleEndian.PutUint32(b[20:], uint32(n))
	return b
}
func fpFixture(scale float32) []byte {
	b := header(0, scale, 0, headerBytes+Dimension*4)
	nz := 0
	for i := 0; i < Dimension; i++ {
		w := float32(i%17-8) * .125
		if i%29 == 0 {
			w = math.Float32frombits(1)
		}
		if i%31 == 0 {
			w = math.Float32frombits(0x80000000)
		}
		binary.LittleEndian.PutUint32(b[headerBytes+4*i:], math.Float32bits(w))
		if w != 0 {
			nz++
		}
	}
	binary.LittleEndian.PutUint32(b[20:], uint32(nz))
	return b
}
func intFixture(scale float32) []byte {
	b := header(1, scale, 0, headerBytes+Dimension)
	nz := 0
	for i := 0; i < Dimension; i++ {
		q := int8(i%255 - 127)
		b[headerBytes+i] = byte(q)
		if float64(q)*float64(scale) != 0 {
			nz++
		}
	}
	binary.LittleEndian.PutUint32(b[20:], uint32(nz))
	return b
}
func ternFixture(indices []int, scale float32) []byte {
	b := header(2, scale, len(indices), headerBytes+bitmapBytes+(len(indices)+7)/8)
	for rank, i := range indices {
		b[headerBytes+i/8] |= 1 << uint(i%8)
		if rank%3 != 0 {
			b[headerBytes+bitmapBytes+rank/8] |= 1 << uint(rank%8)
		}
	}
	return b
}

type fixture struct {
	name  string
	bytes []byte
}

func fixtures() []fixture {
	all := make([]int, Dimension)
	for i := range all {
		all[i] = i
	}
	edges := []int{0, 1, 7, 8, 63, 64, 65, 127, 128, 255, 4095, 4096, 8191}
	return []fixture{{"fp32", fpFixture(0)}, {"fp32_scale_negative_zero", fpFixture(math.Float32frombits(0x80000000))}, {"int8", intFixture(.1)}, {"int8_scale_zero", intFixture(0)}, {"int8_scale_negative_zero", intFixture(math.Float32frombits(0x80000000))}, {"ternary_edges", ternFixture(edges, .3)}, {"ternary_dense", ternFixture(all, .3)}, {"ternary_empty", ternFixture(nil, 0)}, {"ternary_empty_negative_zero", ternFixture(nil, math.Float32frombits(0x80000000))}}
}
func syntheticFeatures() []Feature {
	fs := []Feature{{8191, -.25}, {0, .5}, {64, 1}, {63, -.75}, {4096, 1e-200}, {128, 1e100}, {1, -1e100}, {8191, .125}, {0, math.Copysign(0, -1)}}
	for i := 0; i < 257; i++ {
		fs = append(fs, Feature{Index: (i * 997) % Dimension, Value: math.Sin(float64(i))})
	}
	return fs
}
func TestCoefficientAndScoreBits(t *testing.T) {
	for _, f := range fixtures() {
		name, b := f.name, f.bytes
		t.Run(name, func(t *testing.T) {
			w, e := hintlearn.Decode(b)
			if e != nil {
				t.Fatal("synthetic reference", e)
			}
			v, e := New(b)
			if e != nil {
				t.Fatal(e)
			}
			for i, x := range w {
				y, e := v.Coefficient(i)
				if e != nil || math.Float64bits(y) != math.Float64bits(x) {
					t.Fatalf("coefficient %d", i)
				}
			}
			for _, fs := range [][]Feature{nil, {}, syntheticFeatures(), {{0, math.Inf(1)}}} {
				want := hintlearn.Score(w, fs)
				got, e := v.Score(fs)
				if e != nil || math.Float64bits(got) != math.Float64bits(want) {
					t.Fatalf("score bits got=%x want=%x", math.Float64bits(got), math.Float64bits(want))
				}
			}
			payload := len(b)
			if b[12] == 2 {
				payload += 258
			}
			if v.OwnedPayloadBytes() != payload {
				t.Fatal("payload accounting")
			}
		})
	}
}
func TestOwnedBytesAndConcurrentReaders(t *testing.T) {
	b := ternFixture([]int{0, 63, 64, 128, 8191}, .3)
	v, e := New(b)
	if e != nil {
		t.Fatal(e)
	}
	want, e := v.Score(syntheticFeatures())
	if e != nil {
		t.Fatal(e)
	}
	for i := range b {
		b[i] ^= 0xff
	}
	got, e := v.Score(syntheticFeatures())
	if e != nil || math.Float64bits(got) != math.Float64bits(want) {
		t.Fatal("borrowed storage")
	}
	var wg sync.WaitGroup
	for n := 0; n < 4; n++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for j := 0; j < 32; j++ {
				x, e := v.Score(syntheticFeatures())
				if e != nil || math.Float64bits(x) != math.Float64bits(want) {
					t.Error("reader result")
				}
			}
		}()
	}
	wg.Wait()
}
func TestAcceptanceParityAndInvalidBytes(t *testing.T) {
	base := fpFixture(0)
	cases := []struct {
		name string
		b    []byte
	}{{"empty", nil}, {"short", base[:23]}, {"trailing", append(append([]byte(nil), base...), 0)}, {"truncated", base[:len(base)-1]}}
	mut := func(name string, src []byte, f func([]byte)) {
		b := append([]byte(nil), src...)
		f(b)
		cases = append(cases, struct {
			name string
			b    []byte
		}{name, b})
	}
	mut("magic", base, func(b []byte) { b[0] = 'X' })
	mut("dimension", base, func(b []byte) { b[8] = 0 })
	mut("reserved", base, func(b []byte) { b[13] = 1 })
	mut("kind", base, func(b []byte) { b[12] = 3 })
	for _, f := range []struct {
		name string
		bits uint32
	}{{"scale_nan", 0x7fc00000}, {"scale_inf", 0x7f800000}, {"scale_negative", 0xbf800000}} {
		mut(f.name, base, func(b []byte) { binary.LittleEndian.PutUint32(b[16:], f.bits) })
	}
	mut("count_overflow", base, func(b []byte) { binary.LittleEndian.PutUint32(b[20:], ^uint32(0)) })
	mut("count_mismatch", base, func(b []byte) { binary.LittleEndian.PutUint32(b[20:], 0) })
	mut("fp_nan", base, func(b []byte) { binary.LittleEndian.PutUint32(b[24:], 0x7fc00000) })
	mut("fp_inf", base, func(b []byte) { binary.LittleEndian.PutUint32(b[24:], 0x7f800000) })
	mut("int8_minus128", intFixture(.1), func(b []byte) { b[24] = 128 })
	tb := ternFixture([]int{0, 63, 64}, .3)
	mut("ternary_padding", tb, func(b []byte) { b[len(b)-1] |= 128 })
	mut("ternary_count", tb, func(b []byte) { b[24] |= 2 })
	mut("ternary_zero_scale", tb, func(b []byte) { binary.LittleEndian.PutUint32(b[16:], 0) })
	mut("ternary_presence_n0", ternFixture(nil, 0), func(b []byte) { b[24] = 1 })
	for _, f := range fixtures() {
		cases = append(cases, struct {
			name string
			b    []byte
		}{"valid_" + f.name, f.bytes})
	}
	for _, c := range cases {
		t.Run(c.name, func(t *testing.T) {
			_, re := hintlearn.Decode(c.b)
			_, pe := New(c.b)
			if (re == nil) != (pe == nil) {
				t.Fatal("acceptance differs")
			}
			if c.name[:min(6, len(c.name))] != "valid_" && re == nil {
				t.Fatal("invalid fixture accepted")
			}
		})
	}
}
func TestZeroViewAndFeatureBounds(t *testing.T) {
	var empty View
	if _, e := empty.Score(nil); e != ErrView {
		t.Fatal("zero view")
	}
	var nilv *View
	if _, e := nilv.Coefficient(0); e != ErrView {
		t.Fatal("nil view")
	}
	v, e := New(fpFixture(0))
	if e != nil {
		t.Fatal(e)
	}
	for _, i := range []int{-1, Dimension} {
		if _, e = v.Coefficient(i); e != ErrIndex {
			t.Fatal("coefficient bounds")
		}
		if _, e = v.Score([]Feature{{Index: i, Value: 1}}); e != ErrIndex {
			t.Fatal("score bounds")
		}
	}
}

// This literal test catches the forbidden optimization sum(sign*feature)*scale:
// those two computations differ by an FP64 rounding bit for these values.
func TestDoNotMoveScaleAfterAccumulation(t *testing.T) {
	scale := float32(.1)
	b := ternFixture([]int{0, 1, 2}, scale)
	v, e := New(b)
	if e != nil {
		t.Fatal(e)
	}
	fs := []Feature{{0, 1e16}, {1, 1e16 + 2}, {2, 1}}
	w, e := hintlearn.Decode(b)
	if e != nil {
		t.Fatal(e)
	}
	want := hintlearn.Score(w, fs)
	got, e := v.Score(fs)
	if e != nil || math.Float64bits(got) != math.Float64bits(want) {
		t.Fatal("order changed")
	}
	alternative := (-1e16 + (1e16 + 2) + 1) * float64(scale)
	if math.Float64bits(alternative) == math.Float64bits(want) {
		t.Fatal("literal fixture did not distinguish rearrangement")
	}
}
