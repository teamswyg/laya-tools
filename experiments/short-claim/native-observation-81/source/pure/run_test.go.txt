// No upstream production imports. Every behavioral callback below is a stub.
func frozenMetadata(t *testing.T) WantFile {
	t.Helper()
	b, e := os.ReadFile("../WANTS.v1.json")
	if e != nil {
		t.Fatal(e)
	}
	w, e := LoadWant(b)
	if e != nil {
		t.Fatal(e)
	}
	return w
}

type stubFlag struct {
	values  []string
	changed bool
}

func (s *stubFlag) Register(v []string)    { s.values = CopyStrings(v) }
func (s *stubFlag) Lookup() (bool, string) { return false, "items" }
func (s *stubFlag) GetSlice() []string     { return s.values }
func (s *stubFlag) Changed() bool          { return s.changed }
func (s *stubFlag) Operate(op Operation) error {
	switch op.Name {
	case "SliceValue.Replace":
		_ = json.Unmarshal(op.Argument, &s.values)
	case "SliceValue.Append":
		var v string
		_ = json.Unmarshal(op.Argument, &v)
		s.values = append(s.values, v)
	case "FlagSet.Set":
		s.values = []string{"stub-only"}
		s.changed = true
	}
	return nil
}
func stubAdapter() Adapter {
	return Adapter{Factory: func(int) any { return "stub-factory" }, HookExec: func(any, reflect.Value, reflect.Value) (any, error) { return "stub-only", nil }, NewFlagSet: func() FlagSession { return &stubFlag{} }, Date: func(time.Time) [3]int { return [3]int{7, 8, 9} }, Clock: func(time.Time) [3]int { return [3]int{1, 2, 3} }, Nanosecond: func(time.Time) int { return 4 }, Zone: func(time.Time) (string, int) { return "stub-zone", 5 }}
}

func TestTrueDefinedTypesAndValidZeroTargets(t *testing.T) {
	for _, pair := range []struct {
		target string
		not    reflect.Type
	}{{"defined Time", reflect.TypeOf(time.Time{})}, {"defined []string", reflect.TypeOf([]string(nil))}} {
		b, _ := json.Marshal(HookInput{Expected: json.RawMessage("null"), Literal: "7", SourceType: "int", TargetType: pair.target})
		f, to, e := HookValues(b)
		if e != nil || !to.IsValid() || to.Type() == pair.not || f.Type() != reflect.TypeOf(int(7)) || f.Int() != 7 {
			t.Fatal("defined target or literal int lost")
		}
	}
	if _, _, e := HookValues([]byte(`{"expected":null,"input_literal":"7","source_type":"named string","target_type":"time.Time"}`)); e == nil {
		t.Fatal("outside source scope accepted")
	}
}

func TestSliceSnapshotsOwnValuesAndNilness(t *testing.T) {
	var r Result
	x := runner{r: &r, save: func(Result) error { return nil }}
	backing := []string{"before"}
	f := &stubFlag{values: backing}
	var s Snapshot
	if e := x.snapshot(f, &s); e != nil {
		t.Fatal(e)
	}
	backing[0] = "after"
	if s.Slice[0] != "before" {
		t.Fatal("borrowed slice escaped")
	}
	for _, values := range [][]string{nil, make([]string, 0)} {
		f.values = values
		var next Snapshot
		if e := x.snapshot(f, &next); e != nil {
			t.Fatal(e)
		}
		if *next.SliceNil != (values == nil) || (next.Slice == nil) != (values == nil) {
			t.Fatal("nil/empty collapsed")
		}
	}
}

func TestStrictJSONBoundaries(t *testing.T) {
	for _, b := range []string{`{"x":1,"x":2}`, `{"x":1,"X":2}`, `[] []`, strings.Repeat("[", 50) + "0" + strings.Repeat("]", 50), string([]byte{0xff})} {
		if UniqueJSON([]byte(b)) == nil {
			t.Fatal("bad JSON accepted")
		}
	}
	if UniqueJSON([]byte(`{"a":[{"x":"x"}],"b":[]}`)) != nil {
		t.Fatal("valid nested JSON")
	}
	if UniqueJSON(make([]byte, Limit+1)) == nil {
		t.Fatal("oversize accepted")
	}
}
