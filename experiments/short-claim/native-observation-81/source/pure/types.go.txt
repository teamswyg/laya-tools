// This interface's methods are dispatch boundaries, not upstream guarantees.
type FlagSession interface {
	Register([]string)
	Lookup() (bool, string)
	GetSlice() []string
	Changed() bool // reads a retained public field; not a budgeted method API.
	Operate(Operation) error
}
type Adapter struct {
	Factory    func(int) any
	HookExec   func(any, reflect.Value, reflect.Value) (any, error)
	NewFlagSet func() FlagSession
	Date       func(time.Time) [3]int
	Clock      func(time.Time) [3]int
	Nanosecond func(time.Time) int
	Zone       func(time.Time) (string, int)
}
