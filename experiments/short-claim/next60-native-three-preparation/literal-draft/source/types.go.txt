// Original BitSet satisfies these read methods. Mutation wrappers are supplied later.
type Bitset interface {
    Len() uint

    Words() []uint64
}

type BitsetState struct {
    Length uint
    WordsNil bool
    WordN int
    Words [2]uint64
}
func CopyState(b Bitset) (BitsetState, error) {
    if b == nil { return BitsetState{}, ErrBinding }
    n := b.Len()
    if n > MaxBits { return BitsetState{}, ErrBounds }
    words := b.Words() // borrowed original slice; immediately copied, never retained
    if len(words) > 2 || len(words) != int((n+63)/64) { return BitsetState{}, ErrBounds }
    s := BitsetState{Length:n, WordsNil:words==nil, WordN:len(words)}
    copy(s.Words[:], words)
    return s, nil
}
