package main

import (
	"encoding/json"
	"flag"
	"fmt"
	"os"
)

func main() {
	in := flag.String("input-root", "", "read-only repository root")
	out := flag.String("output", "", "new private JSON output path")
	plan := flag.String("plan-sha256", "", "parent-reviewed PLAN67 SHA")
	flag.Parse()
	if flag.NArg() != 0 || *in == "" || *out == "" || *plan != planSHA {
		fmt.Fprintln(os.Stderr, "scope67_explicit_frozen_invocation_required")
		os.Exit(2)
	}
	// Exclusive reserve prevents accidental overwrite or automatic retry reuse.
	f, e := os.OpenFile(*out, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		fmt.Fprintln(os.Stderr, "scope67_output_reserve_failed")
		os.Exit(1)
	}
	report, ledger, err := load(*in, frozenConfig)
	status := "passed_metadata_proposed_supervision_only"
	failure := ""
	if err != nil {
		status = "failed"
		failure = err.Error()
	}
	result := struct {
		State   string `json:"state"`
		Failure string `json:"failure_code"`
		Report  Output `json:"report"`
		Ledger  Ledger `json:"ledger"`
	}{status, failure, report, ledger}
	b, e := json.MarshalIndent(result, "", "  ")
	if e == nil {
		_, e = f.Write(append(b, '\n'))
	}
	ce := f.Close()
	if e != nil || ce != nil {
		fmt.Fprintln(os.Stderr, "scope67_output_persist_failed")
		os.Exit(1)
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, failure)
		os.Exit(1)
	}
	fmt.Println("scope67 metadata inventory persisted; supervision proposed only; training_ready=false")
}
