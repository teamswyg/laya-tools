func lowerSHA(s string) bool { if len(s)!=64{return false}; for _,c:=range s { if !(c>='0'&&c<='9') && !(c>='a'&&c<='f'){return false} }; return true }
func decode(data []byte, dst any) error { d:=json.NewDecoder(bytes.NewReader(data)); d.DisallowUnknownFields(); if e:=d.Decode(dst);e!=nil{return e}; if d.Decode(new(any))!=io.EOF{return errors.New("json_tail")};return nil }
func safeRelative(s string) bool { return s!="" && !filepath.IsAbs(s) && filepath.Clean(s)==s && s!=".." && !strings.HasPrefix(s,".."+string(os.PathSeparator)) }
func noLinks(path string) error { for p:=filepath.Clean(path);;p=filepath.Dir(p){s,e:=os.Lstat(p);if e!=nil||s.Mode()&os.ModeSymlink!=0{return errors.New("path")};if filepath.Dir(p)==p{return nil}} }
func readFile(path string, cap int64) ([]byte,error) {
	if !filepath.IsAbs(path)||noLinks(path)!=nil{return nil,errors.New("path")};s,e:=os.Lstat(path);if e!=nil||!s.Mode().IsRegular()||s.Size()<=0||s.Size()>cap{return nil,errors.New("file")}
	f,e:=os.Open(path);if e!=nil{return nil,e};defer f.Close();a,e:=f.Stat();if e!=nil||!os.SameFile(s,a){return nil,errors.New("identity")}
	b,e:=io.ReadAll(io.LimitReader(f,cap+1));if e!=nil||int64(len(b))!=s.Size(){return nil,errors.New("read")};z,e:=os.Lstat(path);if e!=nil||!os.SameFile(s,z)||z.Size()!=s.Size()||!z.ModTime().Equal(s.ModTime()){return nil,errors.New("changed")};return b,nil
}
func readRef(root string,r Ref,cap int64)([]byte,error){if !safeRelative(r.Path)||r.Bytes<=0||r.Bytes>cap||!lowerSHA(r.SHA){return nil,errors.New("ref")};b,e:=readFile(filepath.Join(root,r.Path),cap);if e!=nil{return nil,e};h:=sha256.Sum256(b);if int64(len(b))!=r.Bytes||hex.EncodeToString(h[:])!=r.SHA{return nil,errors.New("pin")};return b,nil}
func syncDir(path string) error {f,e:=os.Open(path);if e!=nil{return e};defer f.Close();return f.Sync()}
func exclusive(path string,data []byte) error {f,e:=os.OpenFile(path,os.O_WRONLY|os.O_CREATE|os.O_EXCL,0600);if e!=nil{return e};_,e=f.Write(data);if e==nil{e=f.Sync()};c:=f.Close();if e==nil{e=c};if e==nil{e=syncDir(filepath.Dir(path))};return e}
func inputValues(v Vectors)([6]float64,error){
	x:=[6]float64{1.125,1.375,1.999,0.000000125,math.Float64frombits(0x8000000000000000),math.Inf(1)}
	dec:=[6]string{"1.125","1.375","1.999","0.000000125","","+Inf"};bits:=[6]string{"","","","","8000000000000000",""};digits:=[6]int{2,2,2,9,2,2};want:=[6]string{"1.12","1.38","2","0.000000125","0",""}
	if v.Schema!="riido-ftoa-six-frozen-vectors-v1"||len(v.Rows)!=6{return x,errors.New("vectors")}
	for i,r:=range v.Rows {id:=fmt.Sprintf("next60-humanize-precise-ftoa-vector-%02d",i+1);if r.ID!=id||r.Input.Decimal!=dec[i]||r.Input.Bits!=bits[i]||r.Input.Digits!=digits[i]{return x,errors.New("input")};if i<5 {if r.Want.Text==nil||*r.Want.Text!=want[i]||r.Want.ErrorKind!=""{return x,errors.New("Want")}}else if r.Want.Text!=nil||r.Want.ErrorKind!="nonfinite"{return x,errors.New("Want")}}
	return x,nil
}
func errorGot(text string,err error,precision,nonfinite error) Got {present:=err!=nil;g:=Got{Text:text,ErrorChannel:true,ErrorPresent:&present};if err!=nil{g.ErrorMessage=err.Error();if errors.Is(err,precision){g.ErrorKind="precision"}else if errors.Is(err,nonfinite){g.ErrorKind="nonfinite"}else{g.ErrorKind="other"}};return g}
func observe(fn func()Got)(g Got){defer func(){if recover()!=nil{g=Got{Panic:true,ErrorKind:"panic"}}}();return fn()}
func run(path string) error {
	if runtime.Version()!="go1.27.1"||runtime.GOOS!="darwin"||runtime.GOARCH!="arm64"||os.Getenv("GOMAXPROCS")!="1"||os.Getenv("GOMEMLIMIT")!="256MiB"{return errors.New("runtime")}
	b,e:=readFile(path,65536);if e!=nil{return e};var p Plan;if decode(b,&p)!=nil||p.Schema!="riido-next60-ftoa-slice-plan-v1"||p.Mode!="reserved_native_observation_only"||p.Scheduled!=18||p.Retry!=0||!lowerSHA(p.ReservationSHA){return errors.New("plan")}
	if p.GoVersion!="go1.27.1"||p.GoOS!="darwin"||p.GoArch!="arm64"||p.CPU!=1||p.SoftHeap!=268435456||p.RSS!=268435456||p.Seconds!=300||p.OutputBytes!=12582912||p.OutsideBytes!=1048576{return errors.New("limits")}
	if !filepath.IsAbs(p.SourceRoot)||!filepath.IsAbs(p.AttemptDir)||p.SourceRoot==p.AttemptDir||noLinks(p.SourceRoot)!=nil||noLinks(filepath.Dir(p.AttemptDir))!=nil{return errors.New("dirs")}
	if p.Closure.ID!="ftoa/closure"||p.Vectors.ID!="ftoa/vectors"||p.Readiness.ID!="root/local-observation-readiness"||compiledClosureSHA!=p.Closure.SHA||compiledVectorSHA!=p.Vectors.SHA||compiledReadinessSHA!=p.Readiness.SHA{return errors.New("compiled_bindings")}
	c,e:=readRef(p.SourceRoot,p.Closure,65536);if e!=nil{return e};var closure Closure;if decode(c,&closure)!=nil||closure.Schema!="riido-ftoa-slice-closure-v1"||closure.Boundary!="original-file-only-not-whole-humanize"||len(closure.Files)!=7{return errors.New("closure")};seen:=make([]string,0,7);for _,r:=range closure.Files{for _,s:=range seen{if s==r.Path{return errors.New("duplicate")}};if _,e=readRef(p.SourceRoot,r,65536);e!=nil{return e};seen=append(seen,r.Path)}

	f,e:=os.OpenFile(filepath.Join(p.AttemptDir,"results.json"),os.O_WRONLY|os.O_CREATE|os.O_EXCL,0600);if e!=nil{return e};defer f.Close();if e=f.Sync();e!=nil{return e};if e=syncDir(p.AttemptDir);e!=nil{return e}
	h:=sha256.Sum256(b);result:=Result{Schema:"riido-ftoa-slice-original-observation-result-v1",PlanSHA:hex.EncodeToString(h[:]),ClosureSHA:p.Closure.SHA,VectorsSHA:p.Vectors.SHA,ReadinessSHA:p.Readiness.SHA,Scheduled:18,Rows:make([]Row,0,18),OracleChecks:make([]OracleCheck,0,6),Scope:"Explicit original/authored candidate/oracle entry calls only. 	for i,x:=range values {r:=vectors.Rows[i];text,err:=oracle.RenderBits(math.Float64bits(x),r.Input.Digits);result.OracleCalls++;ok:=err==nil&&r.Want.Text!=nil&&text==*r.Want.Text&&r.Want.ErrorKind=="";if r.Want.ErrorKind=="nonfinite"{ok=errors.Is(err,oracle.ErrNonfinite)&&r.Want.Text==nil};result.OracleChecks=append(result.OracleChecks,OracleCheck{r.ID,strconv.FormatUint(math.Float64bits(x),16),ok});if !ok{result.Failure="frozen_Want_oracle_disputed"}}
	if result.Failure=="" {names:=[3]string{"original/FtoaWithDigits","authored/FtoaPrecise","authored/FlawedDoubleRound"};for k,name:=range names{for i,x:=range values{r:=vectors.Rows[i];g:=observe(func()Got{switch k{case 0:result.OriginalCalls++;return Got{Text:original.FtoaWithDigits(x,r.Input.Digits),ErrorChannel:false,ErrorKind:"unavailable"};case 1:result.PreciseCalls++;s,e:=candidate.FtoaPrecise(x,r.Input.Digits);return errorGot(s,e,candidate.ErrPrecision,candidate.ErrNonfinite);default:result.FlawedCalls++;s,e:=candidate.FlawedDoubleRound(x,r.Input.Digits);return errorGot(s,e,candidate.ErrPrecision,candidate.ErrNonfinite)}});result.Rows=append(result.Rows,Row{len(result.Rows),"next60-humanize-precise-ftoa",name,r.ID,strconv.FormatUint(math.Float64bits(x),16),r.Want,g,nil,nil,nil});result.Completed++}}}

}
